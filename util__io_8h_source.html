287static inline double rw_pairs_keep(const int rank, const int w, const int l) {
288 double q = 1.0;
289 for (int i = 0; i < w - 2; i++) {
290 if (rank - l - i <= 0) return 0.0;
291 q *= (double)(rank - l - i) / (double)(rank - i);
292 }
293 return q;
294}

307static inline int rw_pairs_min_bits(const int nvar, const int rank) {
308 const double k = (double)(nvar - rank);
309 double budget = (double)rank * (double)nvar / RW_PAIRS_WORK;
310 if (budget < RW_PAIRS_FREE) budget = RW_PAIRS_FREE;
311 int l = 1;
312 double pairs = k * k / 4.0; /* (n - r)^2 / 2^(l+1) */
313 while ((l < RW_PAIRS_MAX_BITS) && (pairs > budget)) {
314 l++;
315 pairs /= 2.0;
316 }
317 return l;
318}

332static inline int rw_pairs_bits(const params_t * const p, const int nvar, const int rank, const int w) {
333 const int k = nvar - rank;
334 if ((p->pairs == 0) || (k < 2) || (rank < 1) || (w < 2)) return 0;
335 int l = p->pairs;
336 if (l < 0) {
337 l = 1;
338 while ((l < RW_PAIRS_MAX_BITS) && (((uint64_t)1 << l) < (uint64_t)k)) l++;
339 const int lmin = rw_pairs_min_bits(nvar, rank);
340 while ((l > 1) && (l > lmin) && (rw_pairs_keep(rank, w, l) < RW_PAIRS_KEEP)) l--;
341 }
342 return (l < rank) ? l : rank;
343}

375static inline int codeword_rw_limit(const params_t * const p, const int cur_dmax, const int feed) {
376 const int lim_max = p->nvar + 1;
377 if (cur_dmax <= 0) return lim_max;
378 long long int lim = cur_dmax;
379 if ((p->outC != NULL || p->maxC > 0) && p->dW >= 0) lim = (long long int)cur_dmax + p->dW + 1;
380 if (feed > lim) lim = feed;
381 return (lim < lim_max) ? (int)lim : lim_max;
382}
