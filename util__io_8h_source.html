461static inline double rw_pairs_keep(const int rank, const int w, const int l) {
462 double q = 1.0;
463 for (int i = 0; i < w - 2; i++) {
464 if (rank - l - i <= 0) return 0.0;
465 q *= (double)(rank - l - i) / (double)(rank - i);
466 }
467 return q;
468}

481static inline int rw_pairs_min_bits(const int nvar, const int rank) {
482 const double k = (double)(nvar - rank);
483 double budget = (double)rank * (double)nvar / RW_PAIRS_WORK;
484 if (budget < RW_PAIRS_FREE) budget = RW_PAIRS_FREE;
485 int l = 1;
486 double pairs = k * k / 4.0; /* (n - r)^2 / 2^(l+1) */
487 while ((l < RW_PAIRS_MAX_BITS) && (pairs > budget)) {
488 l++;
489 pairs /= 2.0;
490 }
491 return l;
492}

506static inline int rw_pairs_bits(const params_t * const p, const int nvar, const int rank, const int w) {
507 const int k = nvar - rank;
508 if ((p->pairs == 0) || (k < 2) || (rank < 1) || (w < 2)) return 0;
509 int l = p->pairs;
510 if (l < 0) {
511 l = 1;
512 while ((l < RW_PAIRS_MAX_BITS) && (((uint64_t)1 << l) < (uint64_t)k)) l++;
513 const int lmin = rw_pairs_min_bits(nvar, rank);
514 while ((l > 1) && (l > lmin) && (rw_pairs_keep(rank, w, l) < RW_PAIRS_KEEP)) l--;
515 }
516 return (l < rank) ? l : rank;
517}
