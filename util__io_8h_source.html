239static inline int codeword_rw_limit(const params_t * const p, const int cur_dmax, const int feed) {
240 const int lim_max = p->nvar + 1;
241 if (cur_dmax <= 0) return lim_max;
242 long long int lim = cur_dmax;
243 if ((p->outC != NULL || p->maxC > 0) && p->dW >= 0) lim = (long long int)cur_dmax + p->dW + 1;
244 if (feed > lim) lim = feed;
245 return (lim < lim_max) ? (int)lim : lim_max;
246}
