101static inline int std_bitcount ( uint64_t x) {
102 uint64_t c1 = UINT64_C (0x5555555555555555 );
103 uint64_t c2 = UINT64_C (0x3333333333333333 );
104 uint64_t c4 = UINT64_C (0x0F0F0F0F0F0F0F0F );
105 x -= (x >> 1) & c1;
106 x = (( x >> 2) & c2) + (x & c2);

188 static inline int nextelement(const word * const set1, const int m, const int pos){
189 word setwd;
190 int w;
191 if (pos < 0){
192 w = 0;
193 setwd = set1[0];
194 }
195 else{
196 w = SETWD(pos);
197 if (w >= m) return -1;
198 setwd = set1[w] & (m4ri_ffff<< SETBT(pos));
199 }
200
201 for (;;){
202 if (setwd != 0)
203 return TIMESWORDSIZE(w) + FIRSTBIT(setwd);
204 if (++w == m)
205 return -1;
206 setwd = set1[w];
207 }
208 }

468static inline int gauss_one_rows(mzd_t *M, const int idx, const int begrow,
469 const rci_t nrows_eff) {
470 if (begrow >= nrows_eff) return 0;
471#ifdef __M4RI_MAX_MZD_BLOCKSIZE
472 /* The rows i are at mzd_row(M, 0) + i * rowstride, see MZD_FIRST_ROW(); with M4RI versions before 2024, unless the
473 * matrix spans several blocks */
474 if (__builtin_expect((uintptr_t)mzd_row(M, nrows_eff - 1) - (uintptr_t)mzd_row(M, 0) !=
475 (uintptr_t)(nrows_eff - 1) * (uintptr_t)M->rowstride * sizeof(word), 0)) {
476 return gauss_one_rows_blocks(M, idx, begrow, nrows_eff);
477 }
478#endif
479 const wi_t word_idx = idx >> 6;
480 const word bit_mask = (word)1 << (idx & 63);
481 const wi_t width = M->width;
482 const wi_t rowstride = M->rowstride;
483 word *row_j = mzd_row(M, begrow);
484 for (rci_t j = begrow; j < nrows_eff; ++j, row_j += rowstride) {
485 if (row_j[word_idx] & bit_mask) {
486 if (j != begrow) {
487 mzd_row_swap(M, begrow, j);
488 }
489 const word * const __restrict piv_row = mzd_row(M, begrow);
490 word * __restrict dst_row = MZD_FIRST_ROW(M);
491 for (rci_t ii = 0; ii < begrow; ++ii, dst_row += rowstride) {
492 if (dst_row[word_idx] & bit_mask) {
493 for (wi_t w = 0; w < width; ++w) {
494 dst_row[w] ^= piv_row[w];
495 }
496 }
497 }
498 dst_row += rowstride;
499 for (rci_t ii = begrow + 1; ii < nrows_eff; ++ii, dst_row += rowstride) {
500 if (dst_row[word_idx] & bit_mask) {
501 for (wi_t w = 0; w < width; ++w) {
502 dst_row[w] ^= piv_row[w];
503 }
504 }
505 }
506 return 1;
507 }
508 }
509 return 0;
510}

534static inline int sparse_syndrome_non_zero(const csr_t * const H, const int cnt, const int ee[]){
535 for(int ir=0; ir < H->rows; ir++){
536 int nz=0;
