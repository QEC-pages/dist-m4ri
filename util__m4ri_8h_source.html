82 n = COUNT(n, 0);
83 n = COUNT(n, 1);

91static inline int std_bitcount ( uint64_t x) {
92 uint64_t c1 = UINT64_C (0x5555555555555555 );
93 uint64_t c2 = UINT64_C (0x3333333333333333 );
94 uint64_t c4 = UINT64_C (0x0F0F0F0F0F0F0F0F );
95 x -= (x >> 1) & c1;
96 x = (( x >> 2) & c2) + (x & c2);
97 x = ( x + (x >> 4) ) & c4;
98 x *= UINT64_C (0x0101010101010101 );
99 return (int) (x >> 56);

159 static inline int nextelement(const word * const set1, const int m, const int pos){
160 word setwd;
161 int w;
162 if (pos < 0){
163 w = 0;
164 setwd = set1[0];
165 }
166 else{
167 w = SETWD(pos);
168 if (w >= m) return -1;
169 setwd = set1[w] & (m4ri_ffff<< SETBT(pos));
170 }
171
172 for (;;){
173 if (setwd != 0)
174 return TIMESWORDSIZE(w) + FIRSTBIT(setwd);
175 if (++w == m)
176 return -1;
177 setwd = set1[w];
178 }
179 }
