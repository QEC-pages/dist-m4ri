201 static inline int nextelement(const word * const set1, const int m, const int pos){
202 word setwd;
203 int w;
204 if (pos < 0){
205 w = 0;
206 setwd = set1[0];
207 }
208 else{
209 w = SETWD(pos);
210 if (w >= m) return -1;
211 setwd = set1[w] & (m4ri_ffff<< SETBT(pos));
212 }
213
214 for (;;){
215 if (setwd != 0)
216 return TIMESWORDSIZE(w) + FIRSTBIT(setwd);
217 if (++w == m)
218 return -1;
219 setwd = set1[w];
220 }
221 }

481static inline int gauss_one_rows(mzd_t *M, const int idx, const int begrow,
482 const rci_t nrows_eff) {
483 if (begrow >= nrows_eff) return 0;
484#ifdef __M4RI_MAX_MZD_BLOCKSIZE
485 /* The rows i are at mzd_row(M, 0) + i * rowstride, see MZD_FIRST_ROW(); with M4RI versions before 2024, unless the
486 * matrix spans several blocks */
487 if (__builtin_expect((uintptr_t)mzd_row(M, nrows_eff - 1) - (uintptr_t)mzd_row(M, 0) !=
488 (uintptr_t)(nrows_eff - 1) * (uintptr_t)M->rowstride * sizeof(word), 0)) {
489 return gauss_one_rows_blocks(M, idx, begrow, nrows_eff);
490 }
491#endif
492 const wi_t word_idx = idx >> 6;
493 const word bit_mask = (word)1 << (idx & 63);
494 const wi_t width = M->width;
495 const wi_t rowstride = M->rowstride;
496 word *row_j = mzd_row(M, begrow);
497 for (rci_t j = begrow; j < nrows_eff; ++j, row_j += rowstride) {
498 if (row_j[word_idx] & bit_mask) {
499 if (j != begrow) {
500 mzd_row_swap(M, begrow, j);
501 }
502 const word * const __restrict piv_row = mzd_row(M, begrow);
503 word * __restrict dst_row = MZD_FIRST_ROW(M);
504 for (rci_t ii = 0; ii < begrow; ++ii, dst_row += rowstride) {
505 if (dst_row[word_idx] & bit_mask) {
506 for (wi_t w = 0; w < width; ++w) {
507 dst_row[w] ^= piv_row[w];
508 }
509 }
510 }
511 dst_row += rowstride;
512 for (rci_t ii = begrow + 1; ii < nrows_eff; ++ii, dst_row += rowstride) {
513 if (dst_row[word_idx] & bit_mask) {
514 for (wi_t w = 0; w < width; ++w) {
515 dst_row[w] ^= piv_row[w];
516 }
517 }
518 }
519 return 1;
520 }
521 }
522 return 0;
523}

547static inline int sparse_syndrome_non_zero(const csr_t * const H, const int cnt, const int ee[]){
548 for(int ir=0; ir < H->rows; ir++){
549 int nz=0;
550 for(int iL = H->p[ir], iE = 0; iL < H->p[ir+1]; iL++){
551 int ic = H->i[iL];
552 while((iE < cnt) && (ee[iE] < ic))
553 iE++;
554 if(iE >= cnt)
555 break;
556 if(ee[iE]==ic)
557 nz ^= 1;
558 }
559 if (nz)
560 return 1;
561 }
