91static inline int std_bitcount ( uint64_t x) {
92 uint64_t c1 = UINT64_C (0x5555555555555555 );
93 uint64_t c2 = UINT64_C (0x3333333333333333 );
94 uint64_t c4 = UINT64_C (0x0F0F0F0F0F0F0F0F );
95 x -= (x >> 1) & c1;
96 x = (( x >> 2) & c2) + (x & c2);
97 x = ( x + (x >> 4) ) & c4;
98 x *= UINT64_C (0x0101010101010101 );
99 return (int) (x >> 56);

178 static inline int nextelement(const word * const set1, const int m, const int pos){
179 word setwd;
180 int w;
181 if (pos < 0){
182 w = 0;
183 setwd = set1[0];
184 }
185 else{
186 w = SETWD(pos);
187 if (w >= m) return -1;
188 setwd = set1[w] & (m4ri_ffff<< SETBT(pos));
189 }
190
191 for (;;){
192 if (setwd != 0)
193 return TIMESWORDSIZE(w) + FIRSTBIT(setwd);
194 if (++w == m)
195 return -1;
196 setwd = set1[w];
197 }
198 }

450static inline int gauss_one_rows(mzd_t *M, const int idx, const int begrow,
451 const rci_t nrows_eff) {
452 if (begrow >= nrows_eff) return 0;
453 const wi_t word_idx = idx >> 6;
454 const word bit_mask = (word)1 << (idx & 63);
455 const wi_t width = M->width;
456 const wi_t rowstride = M->rowstride;
457 word *row_j = mzd_row(M, begrow);
458 for (rci_t j = begrow; j < nrows_eff; ++j, row_j += rowstride) {
459 if (row_j[word_idx] & bit_mask) {
460 if (j != begrow) {
461 mzd_row_swap(M, begrow, j);
462 }
463 const word * const __restrict piv_row = mzd_row(M, begrow);
464 word * __restrict dst_row = M->data;
465 for (rci_t ii = 0; ii < begrow; ++ii, dst_row += rowstride) {
466 if (dst_row[word_idx] & bit_mask) {
467 for (wi_t w = 0; w < width; ++w) {
468 dst_row[w] ^= piv_row[w];
469 }
470 }
471 }
472 dst_row += rowstride;
473 for (rci_t ii = begrow + 1; ii < nrows_eff; ++ii, dst_row += rowstride) {
474 if (dst_row[word_idx] & bit_mask) {
475 for (wi_t w = 0; w < width; ++w) {
476 dst_row[w] ^= piv_row[w];
477 }
478 }
479 }
480 return 1;
481 }
482 }
483 return 0;
484}

508static inline int sparse_syndrome_non_zero(const csr_t * const H, const int cnt, const int ee[]){
509 for(int ir=0; ir < H->rows; ir++){
510 int nz=0;
511 for(int iL = H->p[ir], iE = 0; iL < H->p[ir+1]; iL++){
512 int ic = H->i[iL];
513 while((iE < cnt) && (ee[iE] < ic))
514 iE++;
515 if(iE >= cnt)
516 break;
517 if(ee[iE]==ic)
518 nz ^= 1;
519 }
520 if (nz)
521 return 1;
522 }
523 return 0;

537static inline int colmask_syndrome_non_zero(const colmask_t * const cm, const int cnt, const int ee[]) {
538 const int kw = cm->kw;
539 const word * const mask = cm->mask;
540 if (kw == 1) {
541 word s = 0;
542 for (int i = 0; i < cnt; i++)
543 s ^= mask[ee[i]];
544 return s != 0;
545 }
546 for (int w = 0; w < kw; w++) {
547 word s = 0;
548 for (int i = 0; i < cnt; i++)
549 s ^= mask[(size_t)ee[i] * kw + w];
550 if (s)
551 return 1;
552 }
553 return 0;
