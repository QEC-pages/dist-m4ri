1#!/usr/bin/env python3
2"""Independent GF(2) witness verification and lower-bound cross-audit for dist-m4ri benchmarks.
3
4Checks every run record (`dm4ri-bench-run/1`) outside the timed region:
5 1. Independent numpy GF(2) check of the reported witness support:
6 - classical: H e = 0 (mod 2), e != 0
7 - css_sector: H e = 0 (mod 2), e not in rowspan(G)
8 - hl: H e = 0 (mod 2), L e != 0 (mod 2)
9 - dem/circ: H_dem e = 0 (mod 2), L_dem e != 0 (mod 2)
10 2. Reference and self-consistency checks:
11 - lb <= ub (when ub > 0), lb <= verified witness weight, lb <= d_ref (or bracket_ref[1])
12 - certified distance equals d_ref (or lies inside bracket_ref)
13 - verified witness weight >= d_ref (or bracket_ref[0]); a lighter valid witness flags a conflict
14 3. Cross-run lower-bound audit (`audit_records`):
15 - across all runs of an instance, every lb <= min verified witness weight
16
17A failed check sets `result.status = "wrong"`, records the details in `verify.violations`, and marks
18`verify.lb_audit = "violation"` where a lower bound is implicated.
19
20CLI usage:
21 python3 tools/bench/verify.py --manifest benchmark/manifest/certify_v1.json --session <dir>
22 python3 tools/bench/verify.py --manifest benchmark/manifest/certify_v1.json --session <dir> --audit
23"""

45def read_mtx_gf2(path: Path) -> np.ndarray:
46 """Read a 1-based MatrixMarket coordinate file into a uint8 GF(2) matrix (duplicates XORed)."""
47 nrows: Optional[int] = None
48 ncols: Optional[int] = None
49 mat: Optional[np.ndarray] = None
50 with open(path, "r", encoding="utf-8") as f:
51 header = f.readline()
52 if not header.startswith("%%MatrixMarket"):
53 raise ValueError(f"{path}: missing %%MatrixMarket header")
54 hdr_lo = header.lower()
55 is_pattern = "pattern" in hdr_lo
56 is_sym = "symmetric" in hdr_lo
57 for line in f:
58 s = line.strip()
59 if not s or s.startswith("%"):
60 continue
61 parts = s.split()
62 if nrows is None:
63 nrows, ncols = int(parts[0]), int(parts[1])
64 mat = np.zeros((nrows, ncols), dtype=np.uint8)
65 continue
66 r = int(parts[0]) - 1
67 c = int(parts[1]) - 1
68 val = 1 if is_pattern or len(parts) < 3 else (int(float(parts[2])) & 1)
69 if val and mat is not None:
70 mat[r, c] ^= 1
71 if is_sym and r != c:
72 mat[c, r] ^= 1
73 if mat is None:
74 raise ValueError(f"{path}: empty MatrixMarket file")
75 return mat

78def _pack_rows_u64(mat: np.ndarray) -> np.ndarray:
79 """Pack a 2D uint8 GF(2) matrix into uint64 bit-words per row."""
80 arr = np.asarray(mat, dtype=np.uint8) & 1
81 if arr.ndim == 1:
82 arr = arr.reshape(1, -1)
83 m, n = arr.shape
84 if n == 0:
85 return np.zeros((m, 0), dtype=np.uint64)
86 packed_bytes = np.packbits(arr, axis=1, bitorder="little")
87 rem = packed_bytes.shape[1] % 8
88 if rem:
89 pad = np.zeros((m, 8 - rem), dtype=np.uint8)
90 packed_bytes = np.hstack([packed_bytes, pad])
91 return np.ascontiguousarray(packed_bytes).view(np.uint64)

94def _leading_bit(row_u64: np.ndarray) -> int:
95 """Return index of the lowest set bit in a uint64 word array, or -1 if zero."""
96 nz = np.flatnonzero(row_u64)
97 if nz.size == 0:
98 return -1
99 w_idx = int(nz[0])
100 word = int(row_u64[w_idx])
101 lsb = word & -word
102 return (w_idx << 6) + (lsb.bit_length() - 1)

105class GF2RowBasis:
106 """Precomputed GF(2) row-echelon basis for fast membership testing (e in rowspan(G))."""
107
108 def __init__(self, mat: np.ndarray) -> None:
109 self.ncols = int(mat.shape[1])
110 self.pivots: Dict[int, np.ndarray] = {}
111 packed = _pack_rows_u64(mat)
112 for i in range(packed.shape[0]):
113 self._insert(packed[i].copy())
114
115 def _insert(self, row: np.ndarray) -> bool:
116 while True:
117 bit = _leading_bit(row)
118 if bit < 0:
119 return False
120 piv = self.pivots.get(bit)
121 if piv is None:
122 self.pivots[bit] = row
123 return True
124 np.bitwise_xor(row, piv, out=row)
125
126 @property
127 def rank(self) -> int:
128 return len(self.pivots)
129
130 def contains_support(self, support: Sequence[int]) -> bool:
131 """Return True iff the binary vector with 1s at `support` lies in the row space."""
132 vec = np.zeros((1, self.ncols), dtype=np.uint8)
133 if support:
134 vec[0, list(support)] = 1
135 row = _pack_rows_u64(vec)[0].copy()
136 while True:
137 bit = _leading_bit(row)
138 if bit < 0:
139 return True
140 piv = self.pivots.get(bit)
141 if piv is None:
142 return False
143 np.bitwise_xor(row, piv, out=row)

146def _dem_inst_parts(line: str) -> Tuple[str, str, str]:
147 """Parse one DEM line into (instruction_name, arg_str, target_str), stripping tags and comments."""
148 s = line.strip()
149 if not s or s[0] == "#":
150 return "", "", ""
151 if s[0] == "}":
152 return "}", "", ""
153 i = 0
154 while i < len(s) and (s[i].isalnum() or s[i] == "_"):
155 i += 1
156 name, j = s[:i], i
157 if j < len(s) and s[j] == "[":
158 j += 1
159 while j < len(s) and s[j] != "]":
160 j += 2 if s[j] == "\\" else 1
161 j = min(j + 1, len(s))
162 rest = s[j:].split("#", 1)[0].strip()
163 if rest.startswith("("):
164 k = rest.find(")")
165 if k < 0:
166 return name, rest[1:].strip(), ""
167 return name, rest[1:k].strip(), rest[k + 1:].strip()
168 return name, "", rest

171def parse_dem_columns(dem_text: str) -> Tuple[List[Tuple[Tuple[int, ...], Tuple[int, ...]]], int, int]:
172 """Parse a DEM (with repeat/shift_detectors/tags/^) into (cols, num_detectors, num_observables).
173
174 Matches the dist_m4ri C binary's column numbering: every error(...) line in the unrolled DEM
175 corresponds to one column (0-based), with `^`-separated components summed mod 2.
176 """
177 lines = dem_text.splitlines()
178 cols: List[Tuple[Tuple[int, ...], Tuple[int, ...]]] = []
179 max_d = -1
180 max_o = -1
181
182 def run_block(idx: int, shift: int) -> Tuple[int, int]:
183 nonlocal max_d, max_o
184 while idx < len(lines):
185 name, _args, rest = _dem_inst_parts(lines[idx])
186 idx += 1
187 if not name:
188 continue
189 if name == "}":
190 return idx, shift
191 if name == "repeat":
192 count = int(rest.split()[0])
193 body = idx
194 end = idx
195 for _ in range(count):
196 end, shift = run_block(body, shift)
197 if count <= 0:
198 depth = 1
199 while end < len(lines) and depth > 0:
200 nm = _dem_inst_parts(lines[end])[0]
201 end += 1
202 depth += (nm == "repeat") - (nm == "}")
203 idx = end
204 elif name == "shift_detectors":
205 shift += int(rest.split()[-1])
206 elif name == "detector":
207 for tok in rest.split():
208 if tok.startswith("D"):
209 max_d = max(max_d, int(tok[1:]) + shift)
210 elif name == "logical_observable":
211 for tok in rest.split():
212 if tok.startswith("L"):
213 max_o = max(max_o, int(tok[1:]))
214 elif name == "error":
215 dets: set[int] = set()
216 obs: set[int] = set()
217 for tok in rest.split():
218 if tok == "^":
219 continue
220 if tok[0] == "D":
221 d = int(tok[1:]) + shift
222 max_d = max(max_d, d)
223 dets ^= {d}
224 elif tok[0] == "L":
225 o = int(tok[1:])
226 max_o = max(max_o, o)
227 obs ^= {o}
228 cols.append((tuple(sorted(dets)), tuple(sorted(obs))))
229 return idx, shift
230
231 run_block(0, 0)
232 return cols, max_d + 1, max_o + 1

239def verify_matrix_witness_numpy(
240 H: np.ndarray,
241 support: Sequence[int],
242 *,
243 L: Optional[np.ndarray] = None,
244 G: Optional[np.ndarray] = None,
245 G_basis: Optional[GF2RowBasis] = None,
246 classical: bool = False,
247) -> Dict[str, Any]:
248 """Verify a 0-based codeword support over GF(2) using numpy."""
249 H_u8 = np.asarray(H, dtype=np.uint8) & 1
250 ncols = int(H_u8.shape[1])
251 cols = [int(c) for c in support]
252 if not cols:
253 return {"ok": False, "reason": "empty witness", "weight": 0}
254 if len(set(cols)) != len(cols):
255 return {"ok": False, "reason": "duplicate column indices", "weight": len(cols)}
256 if any(c < 0 or c >= ncols for c in cols):
257 return {"ok": False, "reason": f"column index out of range [0, {ncols})", "weight": len(cols)}
258
259 h_syn = np.bitwise_xor.reduce(H_u8[:, cols], axis=1) if H_u8.shape[0] > 0 else np.zeros(0, dtype=np.uint8)
260 h_wt = int(np.count_nonzero(h_syn))
261 if h_wt != 0:
262 return {
263 "ok": False,
264 "reason": f"H e != 0 (syndrome weight {h_wt})",
265 "weight": len(cols),
266 "H_syndrome_weight": h_wt,
267 }
268
269 detail: Dict[str, Any] = {"ok": True, "weight": len(cols), "H_syndrome_weight": 0}
270 if classical:
271 return detail
272
273 checked_nontrivial = False
274 if L is not None:
275 L_u8 = np.asarray(L, dtype=np.uint8) & 1
276 l_syn = np.bitwise_xor.reduce(L_u8[:, cols], axis=1) if L_u8.shape[0] > 0 else np.zeros(0, dtype=np.uint8)
277 l_wt = int(np.count_nonzero(l_syn))
278 detail["L_syndrome_weight"] = l_wt
279 checked_nontrivial = True
280 if l_wt == 0:
281 detail["ok"] = False
282 detail["reason"] = "L e == 0 (mod 2)"
283 return detail
284
285 if G_basis is not None or G is not None:
286 basis = G_basis if G_basis is not None else GF2RowBasis(G) # type: ignore[arg-type]
287 in_rowspan = basis.contains_support(cols)
288 detail["independent_of_G"] = not in_rowspan
289 checked_nontrivial = True
290 if in_rowspan:
291 detail["ok"] = False
292 detail["reason"] = "e lies in rowspan(G)"
293 return detail
294
295 if not checked_nontrivial:
296 return {"ok": False, "reason": "neither L nor G provided for non-triviality check", "weight": len(cols)}
297 return detail

300def verify_dem_witness_numpy(
301 cols: Sequence[Tuple[Tuple[int, ...], Tuple[int, ...]]],
302 num_detectors: int,
303 num_observables: int,
304 support: Sequence[int],
305) -> Dict[str, Any]:
306 """Verify a 0-based DEM error-column support over GF(2) using numpy syndrome vectors."""
307 ncols = len(cols)
308 idx_list = [int(c) for c in support]
309 if not idx_list:
310 return {"ok": False, "reason": "empty witness", "weight": 0}
311 if len(set(idx_list)) != len(idx_list):
312 return {"ok": False, "reason": "duplicate column indices", "weight": len(idx_list)}
313 if any(c < 0 or c >= ncols for c in idx_list):
314 return {"ok": False, "reason": f"column index out of range [0, {ncols})", "weight": len(idx_list)}
315
316 h_syn = np.zeros(max(0, num_detectors), dtype=np.uint8)
317 l_syn = np.zeros(max(0, num_observables), dtype=np.uint8)
318 for c in idx_list:
319 dets, obs = cols[c]
320 if dets:
321 h_syn[list(dets)] ^= 1
322 if obs:
323 l_syn[list(obs)] ^= 1
324
325 h_wt = int(np.count_nonzero(h_syn))
326 l_wt = int(np.count_nonzero(l_syn))
327 if h_wt != 0:
328 return {
329 "ok": False,
330 "reason": f"H_dem e != 0 (detector syndrome weight {h_wt})",
331 "weight": len(idx_list),
332 "H_syndrome_weight": h_wt,
333 "L_syndrome_weight": l_wt,
334 }
335 if l_wt == 0:
336 return {
337 "ok": False,
338 "reason": "L_dem e == 0 (no logical observable flipped)",
339 "weight": len(idx_list),
340 "H_syndrome_weight": 0,
341 "L_syndrome_weight": 0,
342 }
343 return {
344 "ok": True,
345 "weight": len(idx_list),
346 "H_syndrome_weight": 0,
347 "L_syndrome_weight": l_wt,
348 }

355class VerifierContext:
356 """Caches loaded matrices, GF(2) row bases, and parsed DEM columns across runs."""
357
358 def __init__(self, repo_root: Path = REPO_ROOT, cache_dir: Optional[Path] = None) -> None:
359 self.repo_root = repo_root.resolve()
360 self.cache_dir = (cache_dir if cache_dir is not None else mi.default_cache_dir(self.repo_root)).resolve()
361 self._mtx_cache: Dict[Path, np.ndarray] = {}
362 self._basis_cache: Dict[Path, GF2RowBasis] = {}
363 self._dem_cache: Dict[str, Tuple[List[Tuple[Tuple[int, ...], Tuple[int, ...]]], int, int]] = {}
364
365 def _get_mtx(self, path: Path) -> np.ndarray:
366 mat = self._mtx_cache.get(path)
367 if mat is None:
368 mat = read_mtx_gf2(path)
369 self._mtx_cache[path] = mat
370 return mat
371
372 def _get_basis(self, path: Path) -> GF2RowBasis:
373 basis = self._basis_cache.get(path)
374 if basis is None:
375 basis = GF2RowBasis(self._get_mtx(path))
376 self._basis_cache[path] = basis
377 return basis
378
379 def _get_dem_file(self, path: Path) -> Tuple[List[Tuple[Tuple[int, ...], Tuple[int, ...]]], int, int]:
380 key = f"file:{path.resolve()}"
381 val = self._dem_cache.get(key)
382 if val is None:
383 val = parse_dem_columns(path.read_text(encoding="utf-8"))
384 # Avoid retaining huge DEM column lists in the harness process (preserves wait4 max_rss floor).
385 if len(val[0]) <= 50000:
386 self._dem_cache[key] = val
387 return val

389 def _get_circuit_dem(
390 self,
391 inst: Dict[str, Any],
392 dem_mode: str,
393 dem_path_hint: Optional[str] = None,
394 ) -> Tuple[List[Tuple[Tuple[int, ...], Tuple[int, ...]]], int, int]:
395 if dem_path_hint and Path(dem_path_hint).is_file():
396 return self._get_dem_file(Path(dem_path_hint))
397 key = f"circ:{inst['id']}:{dem_mode}"
398 val = self._dem_cache.get(key)
399 if val is not None:
400 return val
401 recipe = inst.get("recipe")
402 if not recipe:
403 raise ValueError(f"Circuit instance {inst['id']} has no recipe for DEM verification")
404 if dem_mode == "stim_auto":
405 # Matches dist_m4ri.py _dem_from_input -> _circuit_dem (decompose_errors=True fallback False)
406 dm = mi._import_dist_m4ri(self.repo_root)

411 if recipe.get("rounds") is not None:

414 circ = dm.add_noise(circ, p=float(recipe["add_noise"]["p"]))
415 dem, _, _ = dm._circuit_dem(circ)
416 text = str(dem.flattened())
417 val = parse_dem_columns(text)
418 else:
419 data, _ = mi.generate_canonical_dem(recipe, repo_root=self.repo_root)
420 val = parse_dem_columns(data.decode("utf-8"))
421 if len(val[0]) <= 50000:
422 self._dem_cache[key] = val
423 return val

425 def check_witness(
426 self,
427 inst: Dict[str, Any],
428 witness: Dict[str, Any],
429 dem_path_hint: Optional[str] = None,
430 ) -> Dict[str, Any]:
431 """Run the independent numpy GF(2) check for a witness dict on `inst`."""
432 support_raw = witness.get("support") or []
433 base = int(witness.get("base", 0))
434 support = [int(c) - base for c in support_raw]
435 kind = inst["kind"]
436 files = mi.resolve_instance_files(inst, repo_root=self.repo_root, cache_dir=self.cache_dir, verify_sha=False)
437
438 if kind == "classical":
439 return verify_matrix_witness_numpy(self._get_mtx(files["H"]), support, classical=True)
440 if kind == "css_sector":
441 return verify_matrix_witness_numpy(
442 self._get_mtx(files["H"]), support, G_basis=self._get_basis(files["G"])
443 )
444 if kind == "hl":
445 return verify_matrix_witness_numpy(self._get_mtx(files["H"]), support, L=self._get_mtx(files["L"]))
446 if kind == "dem":
447 cols, m, k = self._get_dem_file(files["dem"])
448 return verify_dem_witness_numpy(cols, m, k, support)
449 if kind == "circuit":
450 dem_mode = str(witness.get("dem_mode") or "canonical")
451 cols, m, k = self._get_circuit_dem(inst, dem_mode, dem_path_hint=dem_path_hint)
452 return verify_dem_witness_numpy(cols, m, k, support)
453 raise ValueError(f"Unsupported instance kind {kind!r}")

455 def verify_record(
456 self,
457 record: Dict[str, Any],
458 inst: Dict[str, Any],
459 dem_path_hint: Optional[str] = None,
460 ) -> Dict[str, Any]:
461 """Verify a run record in place, populating `record['verify']` and setting `status='wrong'` on violation."""
462 res = record["result"]
463 status_engine = res.get("status_engine") or res.get("status")
464 res["status_engine"] = status_engine
465 lb = res.get("lb")
466 ub = res.get("ub")
467 exact = bool(res.get("exact"))
468 witness = res.get("witness")
469
470 d_ref = inst.get("d_ref")
471 bracket_ref = inst.get("bracket_ref")
472
473 violations: List[str] = []
474 witness_weight: Optional[int] = None
475 witness_verified: Optional[bool] = None
476 witness_detail: Optional[Dict[str, Any]] = None
477 ub_verified: Optional[bool] = None
478 unverified_reason: Optional[str] = None
479 lb_audit = "ok" if lb is not None else "n/a"
480 matches_ref: Optional[bool] = None
481
482 # 1. Witness GF(2) verification
483 if witness is not None:
484 w_declared = witness.get("weight")
485 w_complete = bool(witness.get("complete", True))
486 w_support = witness.get("support") or []
487 if not w_complete or not w_support:
488 unverified_reason = (
489 "witness support truncated or incomplete"
490 if not w_complete
491 else "empty witness support"
492 )
493 ub_verified = False if (ub is not None and ub > 0) else None
494 else:
495 detail = self.check_witness(inst, witness, dem_path_hint=dem_path_hint)
496 witness_detail = detail
497 witness_weight = int(detail.get("weight", len(w_support)))
498 witness_verified = bool(detail.get("ok"))
499 if not witness_verified:
500 violations.append(f"witness failed GF(2) check: {detail.get('reason', 'unknown')}")
501 elif w_declared is not None and int(w_declared) != witness_weight:
502 violations.append(
503 f"witness declared weight {w_declared} != support length {witness_weight}"
504 )
505 witness_verified = False
506 if ub is not None and ub > 0:
507 ub_verified = bool(witness_verified and witness_weight == ub)
508 if witness_verified and witness_weight != ub:
509 unverified_reason = f"verified witness has weight {witness_weight} != reported ub {ub}"
510 elif ub is not None and ub > 0:
511 ub_verified = False
512 unverified_reason = str(
513 (res.get("engine_info") or {}).get("witness_unavailable_reason")
514 or "no witness reported by engine"
515 )
516
517 # 2. Bound self-consistency & witness vs lb
518 if lb is not None and ub is not None and ub > 0 and lb > ub:
519 violations.append(f"reported lb ({lb}) > reported ub ({ub})")
520 lb_audit = "violation"
521 if witness_verified is True and witness_weight is not None and lb is not None and lb > witness_weight:
522 violations.append(f"reported lb ({lb}) > verified witness weight ({witness_weight})")
523 lb_audit = "violation"
524
525 # 3. Reference check (d_ref / bracket_ref)
526 if status_engine != "error":
527 if d_ref is not None:
528 if lb is not None and lb > d_ref:
529 violations.append(f"reported lb ({lb}) > d_ref ({d_ref})")
530 lb_audit = "violation"
531 if witness_verified is True and witness_weight is not None and witness_weight < d_ref:
532 violations.append(
533 f"verified witness weight ({witness_weight}) < d_ref ({d_ref}); check reference"
534 )
535 if exact or status_engine == "certified":
536 matches_ref = (lb == d_ref and ub == d_ref)
537 if not matches_ref:
538 violations.append(f"certified [{lb}, {ub}] != d_ref ({d_ref})")
539 lb_audit = "violation" if (lb is not None and lb > d_ref) else lb_audit
540 elif bracket_ref is not None:
541 ref_lo, ref_hi = int(bracket_ref[0]), int(bracket_ref[1])
542 if lb is not None and lb > ref_hi:
543 violations.append(f"reported lb ({lb}) > bracket_ref[1] ({ref_hi})")
544 lb_audit = "violation"
545 if witness_verified is True and witness_weight is not None and witness_weight < ref_lo:
546 violations.append(
547 f"verified witness weight ({witness_weight}) < bracket_ref[0] ({ref_lo}); check reference"
548 )
549 if exact or status_engine == "certified":
550 matches_ref = (lb is not None and ref_lo <= lb <= ref_hi and lb == ub)
551 if not matches_ref:
552 violations.append(
553 f"certified [{lb}, {ub}] outside bracket_ref [{ref_lo}, {ref_hi}]"
554 )
555 if lb is not None and lb > ref_hi:
556 lb_audit = "violation"
557
558 verify_block = {
559 "witness_weight": witness_weight,
560 "witness_verified_numpy": witness_verified,
561 "witness_check_detail": witness_detail,
562 "ub_verified": ub_verified,
563 "lb_audit": lb_audit,
564 "matches_ref": matches_ref,
565 "violations": violations,
566 "unverified_reason": unverified_reason,
567 }
568 record["verify"] = verify_block
569 if violations:
570 res["status"] = "wrong"
571 else:
572 res["status"] = status_engine
573 return record

586def audit_records(
587 records: Sequence[Dict[str, Any]],
588 manifest: Dict[str, Any],
589) -> Dict[str, Any]:
590 """Cross-run integrity audit across all deduplicated records (C4 / Table C).
591
592 Checks:
593 - wrong runs (status == 'wrong')
594 - cross-run LB conflicts: any run with lb > min verified witness weight of the same instance,
595 or max lb > d_ref / bracket_ref[1]
596 - certified runs without a verified witness
597 - contaminated runs
598 """
599 dedup = deduplicate_records(records)
600 by_inst_id = {inst["id"]: inst for inst in manifest.get("instances", [])}
601 grouped: Dict[str, List[Dict[str, Any]]] = {}
602 for rec in dedup:
603 iid = rec["instance"]["id"]
604 grouped.setdefault(iid, []).append(rec)
605
606 wrong_runs: List[Dict[str, Any]] = []
607 lb_conflicts: List[Dict[str, Any]] = []
608 unverified_certified: List[Dict[str, Any]] = []
609 contaminated_runs: List[Dict[str, Any]] = []
610
611 for iid, recs in sorted(grouped.items()):
612 inst = by_inst_id.get(iid, {})
613 d_ref = inst.get("d_ref")
614 bracket_ref = inst.get("bracket_ref")
615 verified_weights = [
616 int(r["verify"]["witness_weight"])
617 for r in recs
618 if (r.get("verify") or {}).get("witness_verified_numpy") is True
619 and (r.get("verify") or {}).get("witness_weight") is not None
620 ]
621 min_ver_w = min(verified_weights) if verified_weights else None
622
623 for r in recs:
624 res = r.get("result") or {}
625 ver = r.get("verify") or {}
626 env = r.get("env") or {}
627 status = res.get("status")
628 lb = res.get("lb")
629
630 if status == "wrong":
631 wrong_runs.append({
632 "run_id": r["run_id"],
633 "instance": iid,
634 "engine": f"{r['engine']['name']}:{r['engine']['config']}@p{r['options']['threads']}",
635 "seed": r["options"]["seed"],
636 "lb": lb,
637 "ub": res.get("ub"),
638 "violations": ver.get("violations") or [],
639 })
640 if res.get("status_engine") != "error" and lb is not None:
641 conflict_reasons: List[str] = []
642 if min_ver_w is not None and lb > min_ver_w:
643 conflict_reasons.append(f"lb ({lb}) > cross-run min verified witness ({min_ver_w})")
644 if d_ref is not None and lb > d_ref:
645 conflict_reasons.append(f"lb ({lb}) > d_ref ({d_ref})")
646 elif bracket_ref is not None and lb > int(bracket_ref[1]):
647 conflict_reasons.append(f"lb ({lb}) > bracket_ref[1] ({bracket_ref[1]})")
648 if conflict_reasons:
649 lb_conflicts.append({
650 "run_id": r["run_id"],
651 "instance": iid,
652 "engine": f"{r['engine']['name']}:{r['engine']['config']}@p{r['options']['threads']}",
653 "seed": r["options"]["seed"],
654 "lb": lb,
655 "min_verified_witness": min_ver_w,
656 "reasons": conflict_reasons,
657 })
658 ub_ok = ver.get("ub_verified") if "ub_verified" in ver else ver.get("witness_verified_numpy")
659 if status == "certified" and not res.get("k0") and ub_ok is not True:
660 unverified_certified.append({
661 "run_id": r["run_id"],
662 "instance": iid,
663 "engine": f"{r['engine']['name']}:{r['engine']['config']}@p{r['options']['threads']}",
664 "seed": r["options"]["seed"],
665 "ub": res.get("ub"),
666 "reason": ver.get("unverified_reason") or "ub witness not verified",
667 })
668 if env.get("contaminated"):
669 contaminated_runs.append({
670 "run_id": r["run_id"],
671 "instance": iid,
672 "engine": f"{r['engine']['name']}:{r['engine']['config']}@p{r['options']['threads']}",
673 "seed": r["options"]["seed"],
674 "ext_cpu_s": env.get("ext_cpu_s"),
675 "wall_sec": res.get("wall_sec"),
676 })
677
678 return {
679 "total_runs": len(dedup),
680 "wrong_runs": wrong_runs,
681 "lb_conflicts": lb_conflicts,
682 "unverified_certified": unverified_certified,
683 "contaminated_runs": contaminated_runs,
684 "ok": not wrong_runs and not lb_conflicts,
685 }

688def _load_session_records(session_dir: Path) -> List[Dict[str, Any]]:
689 res_file = session_dir / "results.jsonl"
690 out: List[Dict[str, Any]] = []
691 if res_file.is_file():
692 for lineno, line in enumerate(res_file.read_text(encoding="utf-8").splitlines(), 1):
693 if not line.strip():
694 continue
695 try:
696 out.append(json.loads(line))
697 except ValueError as exc:
698 print(

700 file=sys.stderr,
701 )
702 return out
703 runs_dir = session_dir / "runs"
704 if runs_dir.is_dir():
705 for rpath in sorted(runs_dir.glob("*/run.json")):
706 try:
707 out.append(json.loads(rpath.read_text(encoding="utf-8")))
708 except (OSError, ValueError) as exc:
709 print(f"Warning: skipping unreadable {rpath}: {exc}", file=sys.stderr)
710 return out
