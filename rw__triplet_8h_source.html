64static inline int rw_t3_pair(const rw_t3_t * const t, int u, int v) {
65 if (u > v) {
66 const int x = u;
67 u = v;
68 v = x;
69 }
70 const uint64_t key = ((uint64_t)(uint32_t)u << 32) | (uint32_t)v;
71 uint32_t s = (uint32_t)((key * 0x9e3779b97f4a7c15ULL) >> 32) & t->pmask;
72 while (t->pval[s]) {
73 if (t->pkey[s] == key) return t->pval[s] - 1;
74 s = (s + 1) & t->pmask;
75 }
76 return -1;
77}

122static inline uint64_t rw_set_key(const uint64_t * const zcol, const int * const supp, const int cnt,
123 const uint64_t salt) {
124 uint64_t h = (salt + (uint64_t)cnt) * 0x9e3779b97f4a7c15ULL;
