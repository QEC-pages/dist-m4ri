1#!/usr/bin/env python3
2"""Deterministic input generation and sha256 verification for dist-m4ri benchmark manifests.
3
4Generates canonical Detector Error Models (DEMs) from tracked Stim circuits into a content-addressed cache
5outside git (default: <repo>/scratch/bench_cache/), verifies every tracked and generated file against the
6sha256 recorded in the manifest, and refuses to proceed on any mismatch.
7
8Canonical DEM format (matching scratch/bench_0.14.0/run_bench.py:228-263):
9 - remove_empty_detectors(circuit)
10 - optional set_circuit_rounds(circuit, rounds) when recipe['rounds'] is not None
11 - optional add_noise(circuit, p=recipe['add_noise']['p']) when circuit has no noise
12 - circuit.detector_error_model(decompose_errors=False, approximate_disjoint_errors=...)
13 - flattened error columns via dist_m4ri._dem_error_columns(dem, with_prob=True)
14 - one `error(<p>) D... L...` line per non-empty mechanism (sorted detector and observable indices,
15 decomposed `^` components summed mod 2, no detector coordinate lines), plus trailing sentinel
16 `detector D<m-1>` / `logical_observable L<k-1>` only when the highest index does not appear in any error.
17
18Usage:
19 python3 benchmark/manifest/make_inputs.py --manifest benchmark/manifest/certify_v1.json
20 python3 benchmark/manifest/make_inputs.py --only-small --check-overlap
21 python3 benchmark/manifest/make_inputs.py --verify-only --tier T0
22"""

47def sha256_bytes(data: bytes) -> str:
48 """Return hex sha256 digest of bytes."""
49 return hashlib.sha256(data).hexdigest()

52def sha256_file(path: Path) -> str:
53 """Return hex sha256 digest of a file, streaming in 1 MiB chunks."""
54 h = hashlib.sha256()
55 with open(path, "rb") as f:
56 for chunk in iter(lambda: f.read(1 << 20), b""):
57 h.update(chunk)
58 return h.hexdigest()

83def canonical_dem_bytes(dem: Any, dm: Any) -> Tuple[bytes, Dict[str, int]]:
84 """Serialize a stim.DetectorErrorModel into the canonical DEM form (run_bench.py:228-263)."""
85 cols = dm._dem_error_columns(dem, with_prob=True)
86 m = int(dem.num_detectors)
87 k = int(dem.num_observables)
88 lines: List[str] = []
89 max_d, max_o = -1, -1
90 for det_set, obs_set, prob in cols:
91 if not det_set and not obs_set:
92 continue
93 dets = sorted(det_set)
94 obs = sorted(obs_set)
95 if dets:
96 max_d = max(max_d, dets[-1])
97 if obs:
98 max_o = max(max_o, obs[-1])
99 lines.append(
100 f"error({prob})"
101 + "".join(f" D{d}" for d in dets)
102 + "".join(f" L{o}" for o in obs)
103 + "\n"
104 )
105 if max_d < m - 1:
106 lines.append(f"detector D{m - 1}\n")
107 if max_o < k - 1:
108 lines.append(f"logical_observable L{k - 1}\n")
109 data = "".join(lines).encode("utf-8")
110 meta = {"detectors": m, "observables": k, "errors": len(lines) - int(max_d < m - 1) - int(max_o < k - 1)}
111 return data, meta

114def generate_canonical_dem(
115 recipe: Dict[str, Any],
116 repo_root: Path = REPO_ROOT,
117 allow_stim_mismatch: bool = False,
118) -> Tuple[bytes, Dict[str, Any]]:
119 """Generate canonical DEM bytes deterministically from a manifest recipe.
120
121 Verifies the source .stim sha256 and the pinned stim version before running Stim.
122 """
123 if recipe.get("type") != "canonical_dem_v1":
124 raise ValueError(f"Unsupported recipe type: {recipe.get('type')!r}")
125
126 src_spec = recipe["source_stim"]
127 src_path = repo_root / src_spec["path"]
128 if not src_path.is_file():
129 raise FileNotFoundError(f"Source Stim circuit not found: {src_path}")
130 actual_src_sha = sha256_file(src_path)
131 expected_src_sha = src_spec.get("sha256")
132 if expected_src_sha and actual_src_sha != expected_src_sha:
133 raise ValueError(
134 f"Source .stim sha256 mismatch for {src_spec['path']}: "
135 f"expected {expected_src_sha}, got {actual_src_sha}"
136 )
137
138 dm = _import_dist_m4ri(repo_root)
139 stim = dm._get_stim()
140 stim_ver = getattr(stim, "__version__", "unknown")
141 req_stim = recipe.get("stim_version")
142 if req_stim and stim_ver != req_stim and not allow_stim_mismatch:
143 raise RuntimeError(
144 f"Stim version mismatch: manifest pins {req_stim}, installed {stim_ver}. "
145 f"Install stim=={req_stim} or pass --allow-stim-mismatch to inspect."
146 )
147
148 t0 = time.monotonic()
149 circuit = stim.Circuit.from_file(str(src_path))
150 if recipe.get("remove_empty_detectors", True):
151 circuit, _ = dm.remove_empty_detectors(circuit)
152 if recipe.get("rounds") is not None:
153 circuit = dm.set_circuit_rounds(circuit, int(recipe["rounds"]))
154
155 noise_spec = recipe.get("add_noise")
156 if noise_spec:
157 when = noise_spec.get("when", "if_noiseless")
158 p_noise = float(noise_spec["p"])
159 if when == "always":
160 circuit = dm.add_noise(circuit, p=p_noise, force=True)
161 elif when == "if_noiseless":
162 if not dm.has_noise(circuit):
163 circuit = dm.add_noise(circuit, p=p_noise)
164 else:
165 raise ValueError(f"Unknown add_noise.when={when!r}")
166
167 decomp = bool(recipe.get("decompose_errors", False))
168 approx = bool(recipe.get("approximate_disjoint_errors", False))
169 dem = circuit.detector_error_model(
170 decompose_errors=decomp,
171 approximate_disjoint_errors=approx,
172 )
173 data, meta = canonical_dem_bytes(dem, dm)
174 meta["sha256"] = sha256_bytes(data)
175 meta["bytes"] = len(data)
176 meta["gen_s"] = round(time.monotonic() - t0, 4)
177 meta["stim_version"] = stim_ver
178 meta["dist_m4ri_py_version"] = getattr(dm, "__version__", "unknown")
179 return data, meta

192def write_cache_entry(cache_dir: Path, inst_id: str, data: bytes, sha: str, ext: str = ".dem") -> Path:
193 """Write content-addressed cache entry atomically and refresh by_id symlink."""
194 dst = cache_path_for_sha(cache_dir, sha, ext=ext)
195 dst.parent.mkdir(parents=True, exist_ok=True)
196 if not dst.is_file():
197 tmp = dst.with_name(f".{dst.name}.{os.getpid()}.tmp")
198 tmp.write_bytes(data)
199 os.replace(tmp, dst)
200 else:
201 existing_sha = sha256_file(dst)
202 if existing_sha != sha:
203 raise ValueError(
204 f"Corrupted cache file {dst}: expected sha256 {sha}, got {existing_sha}"
205 )
206
207 link_dir = cache_dir / "by_id"
208 link_dir.mkdir(parents=True, exist_ok=True)
209 link_path = link_dir / f"{safe_cache_name(inst_id)}{ext}"
210 rel_target = os.path.relpath(dst, link_dir)
211 tmp_link = link_dir / f".{link_path.name}.{os.getpid()}.tmp"
212 if tmp_link.is_symlink() or tmp_link.exists():
213 tmp_link.unlink()
214 os.symlink(rel_target, tmp_link)
215 os.replace(tmp_link, link_path)
216 return dst

219def resolve_instance_files(
220 inst: Dict[str, Any],
221 repo_root: Path = REPO_ROOT,
222 cache_dir: Optional[Path] = None,
223 verify_sha: bool = True,
224) -> Dict[str, Path]:
225 """Resolve and optionally sha256-verify every input file of a manifest instance."""
226 cdir = cache_dir if cache_dir is not None else default_cache_dir(repo_root)
227 resolved: Dict[str, Path] = {}
228 for role, spec in inst.get("files", {}).items():
229 expected_sha = spec.get("sha256")
230 if spec.get("generated"):
231 if not expected_sha:
232 raise ValueError(
233 f"Instance {inst['id']} file {role!r} is generated but has no sha256 in manifest"
234 )
235 ext = Path(spec.get("cache_name", ".dem")).suffix or ".dem"
236 path = cache_path_for_sha(cdir, expected_sha, ext=ext)
237 if not path.is_file():
238 raise FileNotFoundError(
239 f"Generated input missing for {inst['id']} ({role}): {path}. "
240 f"Run benchmark/manifest/make_inputs.py first."
241 )
242 else:
243 path = repo_root / spec["path"]
244 if not path.is_file():
245 raise FileNotFoundError(f"Tracked input missing for {inst['id']} ({role}): {path}")
246 if verify_sha and expected_sha:
247 actual_sha = sha256_file(path)
248 if actual_sha != expected_sha:
249 raise ValueError(
250 f"sha256 mismatch for {inst['id']} ({role}: {path}): "
251 f"expected {expected_sha}, got {actual_sha}"
252 )
253 resolved[role] = path
254 return resolved

278def verify_manifest_inputs(
279 manifest: Dict[str, Any],
280 instances: Iterable[Dict[str, Any]],
281 repo_root: Path = REPO_ROOT,
282 cache_dir: Optional[Path] = None,
283 warm_page_cache: bool = False,
284) -> Dict[str, Any]:
285 """Verify sha256 of all files of the selected instances (and optionally warm OS page cache)."""
286 cdir = cache_dir if cache_dir is not None else default_cache_dir(repo_root)
287 inst_list = list(instances)
288 checked_files: Dict[Path, int] = {}
289 errors: List[str] = []
290 for inst in inst_list:
291 try:
292 files = resolve_instance_files(inst, repo_root=repo_root, cache_dir=cdir, verify_sha=True)
293 for p in files.values():
294 if p not in checked_files:
295 checked_files[p] = p.stat().st_size
296 if warm_page_cache:
297 # sha256_file already read the full file into the OS page cache.
298 pass
299 except Exception as exc:
300 errors.append(f"{inst['id']}: {exc}")
301 if errors:
302 raise RuntimeError(
303 "Input verification failed:\n " + "\n ".join(errors)
304 )
305 return {
306 "checked": len(inst_list),
307 "files": len(checked_files),
308 "bytes": sum(checked_files.values()),
309 "sha256_ok": True,
310 "page_cache_warmed": bool(warm_page_cache),
311 }

319def _line_stats(path: Path) -> Dict[str, int]:
320 n_err = n_det = n_obs = n_other = 0
321 with open(path, "rb") as f:
322 for line in f:
323 if line.startswith(b"error("):
324 n_err += 1
325 elif line.startswith(b"detector"):
326 n_det += 1
327 elif line.startswith(b"logical_observable"):
328 n_obs += 1
329 elif line.strip():
330 n_other += 1
331 return {"error": n_err, "detector": n_det, "logical_observable": n_obs, "other": n_other}

388def check_overlap(
389 manifest: Dict[str, Any],
390 cache_dir: Path,
391 scratch_root: Path,
392 repo_root: Path = REPO_ROOT,
393) -> List[Dict[str, Any]]:
394 """Compare cached canonical DEMs against existing DEMs in scratch/bench_0.14.0, 0.15.1_p0, 0.17."""
395 dm = _import_dist_m4ri(repo_root)
396 report: List[Dict[str, Any]] = []
397 by_id = {inst["id"]: inst for inst in manifest["instances"]}
398 for inst_id, refs in OVERLAP_MAP.items():
399 inst = by_id.get(inst_id)
400 if not inst:
401 continue
402 dem_spec = inst["files"].get("dem", {})
403 expected_sha = dem_spec.get("sha256")
404 cached = cache_path_for_sha(cache_dir, expected_sha, ".dem") if expected_sha else None
405 is_cached = cached is not None and cached.is_file()
406 for src_label, rel_path in refs:
407 ref_path = scratch_root / rel_path
408 entry: Dict[str, Any] = {
409 "id": inst_id,
410 "source": src_label,
411 "ref_path": str(ref_path),
412 "ref_exists": ref_path.is_file(),
413 "cached": is_cached,
414 "expected_sha256": expected_sha,
415 }
416 if not ref_path.is_file():
417 entry["status"] = "ref_missing"
418 report.append(entry)
419 continue
420 ref_sha = sha256_file(ref_path)
421 entry["ref_sha256"] = ref_sha
422 if not is_cached:
423 entry["status"] = "deferred_not_cached"
424 entry["manifest_sha_matches_ref"] = (ref_sha == expected_sha)
425 report.append(entry)
426 continue
427 assert cached is not None
428 gen_sha = sha256_file(cached)
429 entry["gen_sha256"] = gen_sha
430 if gen_sha == ref_sha:
431 entry["status"] = "exact_sha256_match"
432 entry["diff_summary"] = None
433 else:
434 gen_lines = _line_stats(cached)
435 ref_lines = _line_stats(ref_path)
436 # Compare error lines verbatim and as GF(2) column multisets
437 gen_err = [ln for ln in cached.read_text().splitlines() if ln.startswith("error(")]
438 ref_err = [ln for ln in ref_path.read_text().splitlines() if ln.startswith("error(")]
439 err_lines_equal = (gen_err == ref_err)
440 cols_equal = err_lines_equal or (_col_multiset(str(cached), dm) == _col_multiset(str(ref_path), dm))
441 entry["status"] = "format_diff_same_errors" if err_lines_equal else (
442 "order_or_prob_diff_same_cols" if cols_equal else "matrix_mismatch"
443 )
444 entry["gen_lines"] = gen_lines
445 entry["ref_lines"] = ref_lines
446 entry["error_lines_identical"] = err_lines_equal
447 entry["columns_identical"] = cols_equal
448 diff_parts = []
449 if ref_lines["detector"] != gen_lines["detector"]:
450 diff_parts.append(
451 f"ref has {ref_lines['detector']} detector() coordinate lines vs "
452 f"{gen_lines['detector']} in canonical DEM"
453 )
454 if ref_path.read_bytes().endswith(b"\n") != cached.read_bytes().endswith(b"\n"):
455 diff_parts.append("trailing newline differs")
456 if not err_lines_equal:
457 diff_parts.append(f"error lines differ (columns_identical={cols_equal})")
458 entry["diff_summary"] = "; ".join(diff_parts) or "byte difference"
459 report.append(entry)
460 return report

488 args = parse_args(argv)
489 repo_root = args.repo_root.resolve()

496 if args.only_small and inst.get("kind") == "dem" and (inst.get("recipe") or {}).get("large")
497 ]

499 print(
500 f"Manifest: {manifest['id']} v{manifest['version']} (sha256={manifest_sha[:16]}), "
501 f"selected={len(instances)}, skipped_large={len(skipped_large)}, cache={cache_dir}",
502 flush=True,
503 )
504
505 generated: List[Dict[str, Any]] = []
506 verified_cached = 0
507 verified_tracked = 0
508 mismatches: List[str] = []
509
510 for inst in instances:
511 inst_id = inst["id"]
512 for role, spec in inst.get("files", {}).items():
513 expected_sha = spec.get("sha256")
