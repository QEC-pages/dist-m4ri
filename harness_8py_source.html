1#!/usr/bin/env python3
2"""Exclusive, restartable certification benchmark harness for `dist-m4ri` and future engines.
3
4Implements the protocol of `scratch/bench_review/REVIEW.md` Sections 3.3 and 3.5 and writes
5`dm4ri-bench-run/1`, `dm4ri-bench-env/1`, and `dm4ri-bench-plan/1` JSON files (see `SCHEMA.md`).
6
7Key guarantees:
8 - Input provenance (C1): verifies every tracked and cached input sha256 at session start and
9 warms the OS page cache (m7).
10 - Exclusive runs (C2): acquires `flock /tmp/qec-cert-bench.lock` (shared with qec-cert) around
11 each timed run; `--no-lock` is provided strictly for isolated pinned testing.
12 - Load gate & contamination (M4, m6): waits until <= 2 busy cores before 128-thread runs, uses
13 `os.wait4` for child CPU time and max RSS, and measures `/proc/stat` busy/steal CPU-seconds
14 to flag runs where external CPU time exceeds 2% of `threads * wall_sec`.
15 - Budgets, seeds, placement, kill (M1, M7, M8, m9, m10): per-tier budgets 60/300/1800/3600 s,
16 seeds 5/5/3/1 (always >= 1), p=128 + p=64 on T1/T2 (p=1 pinned to CPU 1, not CPU 0), fixed-
17 seed random permutation per tier, periodic canary run, SIGTERM -> SIGKILL at
18 T + max(30 s, 0.05 T).
19 - Independent verification (C4): invokes `verify.py` after every run outside the timed region.
20 - Restartable & atomic: `run_id = sha1(canonical_json(key))`; finished runs are skipped on
21 restart (`--retry` retries harness errors).
22
23Usage:
24 python3 tools/bench/harness.py --manifest benchmark/manifest/certify_v1.json --session <dir>
25 python3 tools/bench/harness.py --session <dir> --plan-only
26 python3 tools/bench/harness.py --session <dir> --tiers T0 --threads 2 --seeds 1 \\
27 --taskset 126-127 --nice 19 --no-lock --no-load-gate --canary-every 0
28"""

102def read_proc_stat_jiffies() -> Tuple[int, int, int]:
103 """Return (busy_jiffies, steal_jiffies, total_jiffies) from aggregate cpu line of /proc/stat.
104
105 Linux /proc/stat fields: user, nice, system, idle, iowait, irq, softirq, steal[, guest, guest_nice].
106 `guest` and `guest_nice` are already accounted inside `user` and `nice`, so we sum only fields 0..7.
107 """
108 with open("/proc/stat", "r", encoding="utf-8") as f:
109 parts = f.readline().split()[1:]
110 vals = [int(x) for x in parts[:8]]
111 while len(vals) < 8:
112 vals.append(0)
113 user, nice, system, idle, iowait, irq, softirq, steal = vals
114 busy = user + nice + system + irq + softirq + steal
115 total = busy + idle + iowait
116 return busy, steal, total

119def measure_busy_cores(window_s: float = 2.0) -> float:
120 """Measure machine-wide busy cores over `window_s` seconds."""
121 b0, _s0, t0 = read_proc_stat_jiffies()
122 time.sleep(max(0.1, window_s))
123 b1, _s1, t1 = read_proc_stat_jiffies()
124 ncpu = os.cpu_count() or 128
125 dt = max(1, t1 - t0)
126 return round(ncpu * max(0, b1 - b0) / dt, 3)

150def _parse_lscpu(raw: str) -> Dict[str, Any]:
151 kv: Dict[str, str] = {}
152 numa: Dict[str, str] = {}
153 caches: Dict[str, str] = {}
154 for line in raw.splitlines():
155 if ":" not in line:
156 continue
157 k, v = line.split(":", 1)
158 k, v = k.strip(), v.strip()
159 kv[k] = v
160 if k.startswith("NUMA node") and k.endswith("CPU(s)"):
161 numa[k] = v
162 elif k.endswith("cache"):
163 caches[k] = v
164 flags = kv.get("Flags", "")
165 flags_sha1 = hashlib.sha1(flags.encode("utf-8")).hexdigest() if flags else None
166 return {
167 "model": kv.get("Model name"),
168 "architecture": kv.get("Architecture"),
169 "cpus": int(kv["CPU(s)"]) if kv.get("CPU(s)", "").isdigit() else os.cpu_count(),
170 "sockets": int(kv["Socket(s)"]) if kv.get("Socket(s)", "").isdigit() else None,
171 "cores_per_socket": int(kv["Core(s) per socket"]) if kv.get("Core(s) per socket", "").isdigit() else None,
172 "threads_per_core": int(kv["Thread(s) per core"]) if kv.get("Thread(s) per core", "").isdigit() else None,
173 "numa": numa,
174 "caches": caches,
175 "hypervisor": kv.get("Hypervisor vendor"),
176 "flags_sha1": flags_sha1,
177 "lscpu": raw,
178 }

350def run_process_wait4(
351 argv: Sequence[str],
352 *,
353 cwd: str,
354 env_extra: Optional[Dict[str, str]],
355 out_path: Path,
356 err_path: Path,
357 kill_after_s: float,
358 taskset_cpus: Optional[str] = None,
359 nice_val: Optional[int] = None,
360 sigterm_grace_s: float = 10.0,
361) -> Dict[str, Any]:
362 """Run one subprocess in its own session, wait with `os.wait4`, and measure CPU/RSS/contamination."""
363 cmd: List[str] = []
364 if taskset_cpus:
365 cmd.extend(["taskset", "-c", taskset_cpus])
366 if nice_val is not None:
367 cmd.extend(["nice", "-n", str(nice_val)])
368 cmd.extend(argv)
369
370 env = dict(os.environ)
371 if env_extra:
372 env.update(env_extra)
373
374 clk_tck = float(os.sysconf("SC_CLK_TCK"))
375 out_path.parent.mkdir(parents=True, exist_ok=True)
376 err_path.parent.mkdir(parents=True, exist_ok=True)
377
378 started_at = utc_now_iso()
379 busy0, steal0, _tot0 = read_proc_stat_jiffies()
380 ru_self0 = resource.getrusage(resource.RUSAGE_SELF)
381 t0 = time.monotonic()
382
383 killed = False
384 term_sent_at: Optional[float] = None
385 kill_sent = False
386 status_raw = 0
387 rusage = None
388
389 with open(out_path, "wb") as out_f, open(err_path, "wb") as err_f:
390 proc = subprocess.Popen(
391 cmd,
392 cwd=cwd,
393 env=env,
394 stdout=out_f,
395 stderr=err_f,
396 start_new_session=True,
397 )
398 pid = proc.pid
399 sleep_s = 0.01
400 while True:
401 wpid, st, ru = os.wait4(pid, os.WNOHANG)
402 if wpid == pid:
403 status_raw = st
404 rusage = ru
405 break
406 elapsed = time.monotonic() - t0
407 if not killed and elapsed >= kill_after_s:
408 killed = True
409 term_sent_at = time.monotonic()
410 try:
411 os.killpg(pid, signal.SIGTERM)
412 except ProcessLookupError:
413 pass
414 elif killed and not kill_sent and term_sent_at is not None:
415 if (time.monotonic() - term_sent_at) >= sigterm_grace_s:
416 kill_sent = True
417 try:
418 os.killpg(pid, signal.SIGKILL)
419 except ProcessLookupError:
420 pass
421 time.sleep(sleep_s)
422 sleep_s = min(0.2, sleep_s * 1.25)
423
424 wall_sec = time.monotonic() - t0
425 ended_at = utc_now_iso()
426 busy1, steal1, _tot1 = read_proc_stat_jiffies()
427 ru_self1 = resource.getrusage(resource.RUSAGE_SELF)
428
429 # Ensure Python's Popen object knows the child was reaped
430 proc.returncode = os.waitstatus_to_exitcode(status_raw)
431 if os.WIFEXITED(status_raw):
432 rc: Optional[int] = os.WEXITSTATUS(status_raw)
433 sig: Optional[int] = None
434 elif os.WIFSIGNALED(status_raw):
435 rc = None
436 sig = os.WTERMSIG(status_raw)
437 else:
438 rc = proc.returncode
439 sig = None
440
441 cpu_user_s = float(rusage.ru_utime) if rusage is not None else 0.0
442 cpu_sys_s = float(rusage.ru_stime) if rusage is not None else 0.0
443 max_rss_kb = int(rusage.ru_maxrss) if rusage is not None else 0
444
445 busy_cpu_s = max(0.0, (busy1 - busy0) / clk_tck)
446 steal_s = max(0.0, (steal1 - steal0) / clk_tck)
447 child_cpu_s = cpu_user_s + cpu_sys_s
448 harness_cpu_s = max(
449 0.0, (ru_self1.ru_utime + ru_self1.ru_stime) - (ru_self0.ru_utime + ru_self0.ru_stime)
450 )
451 ext_cpu_s = max(0.0, busy_cpu_s - child_cpu_s - harness_cpu_s)
452
453 return {
454 "cmd": cmd,
455 "rc": rc,
456 "signal": sig,
457 "killed": killed,
458 "wall_sec": round(wall_sec, 6),
459 "cpu_user_s": round(cpu_user_s, 6),
460 "cpu_sys_s": round(cpu_sys_s, 6),
461 "max_rss_kb": max_rss_kb,
462 "busy_cpu_s": round(busy_cpu_s, 4),
463 "child_cpu_s": round(child_cpu_s, 4),
464 "harness_cpu_s": round(harness_cpu_s, 4),
465 "ext_cpu_s": round(ext_cpu_s, 4),
466 "steal_s": round(steal_s, 4),
467 "started_at": started_at,
468 "ended_at": ended_at,
469 }

472def wait_for_load_gate(
473 *,
474 max_busy: float,
475 window_s: float,
476 poll_s: float,
477 max_wait_s: Optional[float],
478 log_fn: Any,
479) -> Dict[str, Any]:
480 """Wait until external machine busy cores <= max_busy (two consecutive readings after pausing)."""
481 t0 = time.monotonic()
482 busy = measure_busy_cores(window_s)
483 if busy <= max_busy:
484 return {"busy_cores": busy, "max_busy": max_busy, "waited_s": 0.0, "passed": True}
485 log_fn(f"LOAD GATE: busy_cores={busy:.2f} > {max_busy:.2f}; waiting")
486 consec_ok = 0
487 while True:
488 waited = time.monotonic() - t0
489 if max_wait_s is not None and waited >= max_wait_s:
490 log_fn(f"LOAD GATE: exceeded max wait ({waited:.1f}s); busy_cores={busy:.2f}")
491 return {"busy_cores": busy, "max_busy": max_busy, "waited_s": round(waited, 2), "passed": False}
492 time.sleep(poll_s if consec_ok == 0 else min(10.0, poll_s))
493 busy = measure_busy_cores(window_s)
494 if busy <= max_busy:
495 consec_ok += 1
496 if consec_ok >= 2:
497 waited = time.monotonic() - t0
498 log_fn(f"LOAD GATE: cleared after {waited:.1f}s (busy_cores={busy:.2f})")
499 return {"busy_cores": busy, "max_busy": max_busy, "waited_s": round(waited, 2), "passed": True}
500 else:
501 consec_ok = 0

578 budget = float(budget_override) if budget_override is not None else float(tier_cfg["budget_s"])

580 if any(s < 1 for s in seeds):
581 raise ValueError(f"Harness rule m9: all seeds must be >= 1 (got {seeds})")
582
583 tier_runs: List[Dict[str, Any]] = []
584 for inst in tier_insts:
585 for ad in adapters:
586 if not ad.supports(inst):
587 continue
588 if threads_override is not None:
589 p_list = [int(x) for x in threads_override]
590 elif ad.name == "dist_m4ri" and ad.config == "default":
591 # 128 on all tiers + 64 on T1/T2 (from manifest tiers table)
592 p_list = [int(x) for x in tier_cfg["threads"]]
593 else:
594 p_list = [int(tier_cfg["threads"][0])]
595 for p in p_list:
596 for seed in seeds:
597 key = {
598 "manifest": f"{manifest['id']}/{manifest['version']}",
599 "instance": inst["id"],
600 "engine": ad.name,
601 "config": ad.config,
602 "league": league,
603 "threads": int(p),
604 "budget_s": budget,
605 "seed": int(seed),
606 "canary": None,
607 "harness_major": HARNESS_MAJOR,
608 }
609 tier_runs.append({
610 "run_id": compute_run_id(key),
611 "key": key,
612 "tier": tier,
613 "instance": inst["id"],
614 "engine": ad.name,
615 "config": ad.config,
616 "league": league,
617 "threads": int(p),
618 "budget_s": budget,
619 "seed": int(seed),
620 })
621 rng.shuffle(tier_runs)
622 planned_runs.extend(tier_runs)
623
624 return {
625 "schema": "dm4ri-bench-plan/1",
626 "created": utc_now_iso(),
627 "order_seed": order_seed,
628 "track": track,
629 "manifest": {"id": manifest["id"], "version": manifest["version"]},
630 "options": {
631 "tiers": tier_order,
632 "only_small": only_small,
633 "engines": [f"{a.name}:{a.config}" for a in adapters],
634 "threads_override": list(threads_override) if threads_override else None,
635 "seeds_override": list(seeds_override) if seeds_override else None,
636 "budget_override": budget_override,
637 },
638 "total_runs": len(planned_runs),
639 "runs": planned_runs,

679def execute_single_run(
680 *,
681 item: Dict[str, Any],
682 inst: Dict[str, Any],
683 adapter: Adapter,
684 verifier: vf.VerifierContext,
685 session_dir: Path,
686 session_id: str,
687 manifest: Dict[str, Any],
688 manifest_sha: str,
689 env_id: str,
690 lock_file: Optional[str],
691 use_load_gate: bool,
692 max_busy_cores: float,
693 load_window_s: float,
694 load_poll_s: float,
695 load_max_wait_s: Optional[float],
696 taskset_override: Optional[str],
697 nice_val: Optional[int],
698 log_fn: Any,
699) -> Dict[str, Any]:
700 """Execute one run (lock -> load gate -> prepare -> wait4 subprocess -> parse -> verify -> record)."""
701 run_id = item["run_id"]
702 key = item["key"]
703 threads = int(item["threads"])
704 budget_s = float(item["budget_s"])
705 seed = int(item["seed"])
706 canary = key.get("canary")
707 kill_s = kill_after_seconds(budget_s)
708 run_dir = session_dir / "runs" / run_id
709 out_path = run_dir / "out.txt"
710 err_path = run_dir / "err.txt"
711 taskset_cpus = taskset_override if taskset_override is not None else default_taskset_for_threads(threads)
712
713 lock_fd = None
714 lock_wait_s = 0.0
715 load_gate_info: Dict[str, Any] = {
716 "busy_cores": None,
717 "max_busy": max_busy_cores if use_load_gate else None,
718 "waited_s": 0.0,
719 "passed": True,
720 }
721
722 try:
723 if lock_file:
724 t_lk0 = time.monotonic()
725 lock_fd = os.open(lock_file, os.O_CREAT | os.O_RDWR, 0o666)
726 fcntl.flock(lock_fd, fcntl.LOCK_EX)
727 lock_wait_s = round(time.monotonic() - t_lk0, 4)
728
729 if use_load_gate:
730 load_gate_info = wait_for_load_gate(
731 max_busy=max_busy_cores,
732 window_s=load_window_s,
733 poll_s=load_poll_s,
734 max_wait_s=load_max_wait_s,
735 log_fn=log_fn,
736 )
737
738 prep = adapter.prepare(inst)
739 prep_s = float(prep.get("prep_s", 0.0))
740 cmd_spec = adapter.command(inst, threads, budget_s, seed, {"prep": prep})
741
742 proc_info = run_process_wait4(
743 cmd_spec["argv"],
744 cwd=cmd_spec.get("cwd") or str(REPO_ROOT),
745 env_extra=cmd_spec.get("env"),
746 out_path=out_path,
747 err_path=err_path,
748 kill_after_s=kill_s,
749 taskset_cpus=taskset_cpus,
750 nice_val=nice_val,
751 )
752 except Exception as exc:
753 # Harness-level error: record it so `--retry` can pick it up cleanly
754 now = utc_now_iso()
755 record = _assemble_error_record(
756 item=item,
757 inst=inst,
758 adapter=adapter,
759 session_id=session_id,
760 manifest=manifest,
761 manifest_sha=manifest_sha,
762 env_id=env_id,
763 taskset_cpus=taskset_cpus,
764 nice_val=nice_val,
765 lock_file=lock_file,
766 lock_wait_s=lock_wait_s,
767 load_gate_info=load_gate_info,
768 kill_s=kill_s,
769 run_dir=run_dir,
770 session_dir=session_dir,
771 err_msg=f"{type(exc).__name__}: {exc}",
772 timestamp=now,
773 )
774 atomic_write_json(run_dir / "run.json", record)
775 append_jsonl(session_dir / "results.jsonl", record)
776 return record
777 finally:
778 if lock_fd is not None:
779 try:
780 fcntl.flock(lock_fd, fcntl.LOCK_UN)
781 finally:
782 os.close(lock_fd)
783
784 # Parse and verify OUTSIDE the lock and timed region
785 stdout_text = out_path.read_text(encoding="utf-8", errors="replace") if out_path.is_file() else ""
786 stderr_text = err_path.read_text(encoding="utf-8", errors="replace") if err_path.is_file() else ""
787 parsed = adapter.parse(
788 proc_info["rc"],
789 stdout_text,
790 stderr_text,
791 killed=proc_info["killed"],
792 prep=prep,
793 )
794
795 wall_sec = proc_info["wall_sec"]
796 contaminated = bool(proc_info["ext_cpu_s"] > 0.02 * threads * max(wall_sec, 0.001))
797 exclusive = bool(lock_file is not None and (not use_load_gate or load_gate_info.get("passed")))
798
799 result_block = dict(parsed)
800 result_block.update(
801 wall_sec=wall_sec,
802 cap_sec=budget_s,
803 prep_s=round(prep_s, 6),
804 cpu_user_s=proc_info["cpu_user_s"],
805 cpu_sys_s=proc_info["cpu_sys_s"],
806 max_rss_kb=proc_info["max_rss_kb"],
807 rc=proc_info["rc"],
808 signal=proc_info["signal"],
809 killed=proc_info["killed"],
810 )
811
812 record: Dict[str, Any] = {
813 "schema": "dm4ri-bench-run/1",
814 "run_id": run_id,
815 "key": key,
816 "session": session_id,
817 "harness": {"version": HARNESS_VERSION},
818 "manifest": {"id": manifest["id"], "version": manifest["version"], "sha256": manifest_sha},
819 "instance": {
820 "id": inst["id"],
821 "kind": inst["kind"],
822 "tier": inst["tier"],
823 "d_ref": inst.get("d_ref"),
824 "bracket_ref": inst.get("bracket_ref"),
825 "compare3_id": inst.get("compare3_id"),
826 },
827 "engine": adapter.describe(),
828 "options": {
829 "league": key["league"],
830 "threads": threads,
831 "budget_s": budget_s,
832 "seed": seed,
833 "argv_extra": cmd_spec.get("argv_extra") or [],
834 "hints": {},
835 "canary": canary,
836 "kill_after_s": round(kill_s, 3),
837 "cpus": taskset_cpus,
838 },
839 "placement": {
840 "taskset": taskset_cpus,
841 "nice": nice_val,
842 "exclusive": exclusive,
843 "lock": lock_file,
844 "lock_wait_sec": lock_wait_s,
845 "load_gate": load_gate_info,
846 },
847 "result": result_block,
848 "verify": {},
849 "env": {
850 "env_id": env_id,
851 "busy_cpu_s": proc_info["busy_cpu_s"],
852 "child_cpu_s": proc_info["child_cpu_s"],
853 "harness_cpu_s": proc_info["harness_cpu_s"],
854 "ext_cpu_s": proc_info["ext_cpu_s"],
855 "steal_s": proc_info["steal_s"],
856 "contaminated": contaminated,
857 "started_at": proc_info["started_at"],
858 "ended_at": proc_info["ended_at"],
859 },
860 "raw": {
861 "dir": str(run_dir.relative_to(session_dir)),
862 "stdout": str(out_path.relative_to(session_dir)),
863 "stderr": str(err_path.relative_to(session_dir)),
864 "argv": proc_info["cmd"],
865 },
866 }
867
868 verifier.verify_record(record, inst, dem_path_hint=prep.get("dem_path"))
869 atomic_write_json(run_dir / "run.json", record)
870 append_jsonl(session_dir / "results.jsonl", record)
871 return record
