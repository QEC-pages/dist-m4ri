75/* Is {a, b, c} a triplet: h_a + h_b = h_c (distinct sorted rows ci[cp[j]..cp[j+1]-1]) and l_a + l_b = l_c? */
76static int t3_match(const int * const cp, const int * const ci, const colmask_t * const cm, const int a, const int b,
77 const int c) {
78 const int kw = cm->kw;
79 const word * const la = cm->mask + (size_t)a * kw, * const lb = cm->mask + (size_t)b * kw;
80 const word * const lc = cm->mask + (size_t)c * kw;
81 for (int w = 0; w < kw; w++)
82 if ((la[w] ^ lb[w]) != lc[w]) return 0;
83 int ia = cp[a], ib = cp[b], ic = cp[c];
84 const int ea = cp[a + 1], eb = cp[b + 1], ec = cp[c + 1];
85 while (ia < ea || ib < eb) {
86 int r;
87 if (ib >= eb) r = ci[ia++];
88 else if (ia >= ea) r = ci[ib++];
89 else if (ci[ia] < ci[ib]) r = ci[ia++];
90 else if (ci[ib] < ci[ia]) r = ci[ib++];
91 else { /* a common row cancels */
92 ia++;
93 ib++;
94 continue;
95 }
96 if (ic >= ec || ci[ic] != r) return 0;
97 ic++;
98 }
99 return ic == ec;
100}

102static void t3_pair_insert(rw_t3_t * const t, int u, int v, const int w) {
103 if (u > v) {
104 const int x = u;
105 u = v;
106 v = x;
107 }
108 const uint64_t key = ((uint64_t)(uint32_t)u << 32) | (uint32_t)v;
109 uint32_t s = (uint32_t)((key * 0x9e3779b97f4a7c15ULL) >> 32) & t->pmask;
110 while (t->pval[s]) {
111 if (t->pkey[s] == key) { /* (repeated columns w: the smallest one, whatever the order of the rows of H) */
112 if (w + 1 < t->pval[s]) t->pval[s] = w + 1;
113 return;
114 }
115 s = (s + 1) & t->pmask;
116 }
117 t->pkey[s] = key;
118 t->pval[s] = w + 1;
119}

121rw_t3_t *rw_t3_build(const csr_t * const HT, const int m, const colmask_t * const cm, const long long budget) {
122 const double t0 = rw_wall();
123 const int n = HT->rows;
124 rw_t3_t * const t = rw_xcalloc(1, sizeof(rw_t3_t));
125 t->n = n;
126 t->complete = 1;
127 if (!cm || n <= 0 || m <= 0) {
128 t->ms = 1e3 * (rw_wall() - t0);
129 return t;
130 }
131 const int kw = cm->kw;
132 /* the distinct rows of each column (a repeated entry counts once, as in mzd_from_csr()) */
133 int * const cp = rw_xmalloc(((size_t)n + 1) * sizeof(int));
134 int * const ci = rw_xmalloc(((size_t)HT->p[n] + 1) * sizeof(int));
135 int nz = 0;
136 for (int j = 0; j < n; j++) {
137 cp[j] = nz;
138 for (int k = HT->p[j]; k < HT->p[j + 1]; k++)
139 if (nz == cp[j] || ci[nz - 1] != HT->i[k]) ci[nz++] = HT->i[k];
140 }
141 cp[n] = nz;
142 /* the Zobrist hash of each column (random keys of the rows of H and L), and the columns by their smallest row */
143 uint64_t st = 0x517cc1b727220a95ULL;
144 uint64_t * const zr = rw_xmalloc((size_t)m * sizeof(uint64_t));
145 for (int r = 0; r < m; r++) zr[r] = rw_splitmix64(&st);
146 uint64_t * const zl = rw_xmalloc((size_t)kw * 64 * sizeof(uint64_t));
147 for (int r = 0; r < kw * 64; r++) zl[r] = rw_splitmix64(&st);
148 uint64_t * const h = rw_xmalloc((size_t)n * sizeof(uint64_t));
149 int * const bb = rw_xcalloc((size_t)m + 2, sizeof(int)); /* bucket sizes, then their starts */
150 int npure = 0;
151 for (int j = 0; j < n; j++) {
152 const word * const lm = cm->mask + (size_t)j * kw;
153 if (cp[j] == cp[j + 1]) { /* a zero column of H: a codeword of weight 1 if l_j != 0 */
154 word any = 0;
155 for (int w = 0; w < kw; w++) any |= lm[w];
156 npure += (any != 0);
157 h[j] = 0;
158 continue;
159 }
160 uint64_t x = 0;
161 for (int k = cp[j]; k < cp[j + 1]; k++) x ^= zr[ci[k]];
162 for (int w = 0; w < kw; w++)
163 for (word b = lm[w]; b; b &= b - 1) x ^= zl[w * 64 + __builtin_ctzll(b)];
164 h[j] = x;
165 bb[ci[cp[j]] + 1]++;
166 }
167 if (npure > 0) {
168 t->pure = rw_xmalloc((size_t)npure * sizeof(int));
169 for (int j = 0; j < n; j++) {
170 if (cp[j] != cp[j + 1]) continue;
171 const word * const lm = cm->mask + (size_t)j * kw;
172 word any = 0;
173 for (int w = 0; w < kw; w++) any |= lm[w];
174 if (any) t->pure[t->npure++] = j;
175 }
176 }
177 free(zr);
178 free(zl);
179 long long probes = 0;
180 for (int r = 0; r < m; r++) probes += (long long)bb[r + 1] * (bb[r + 1] - 1) / 2;
181 t->probes = probes;
182 if (probes > budget) { /* skipped: too many pairs of columns with the same smallest row */
183 t->complete = 0;
184 free(cp);
185 free(ci);
186 free(h);
187 free(bb);
188 t->ms = 1e3 * (rw_wall() - t0);
189 return t;
190 }
191 for (int r = 0; r <= m; r++) bb[r + 1] += bb[r];
192 int * const bcol = rw_xmalloc((size_t)n * sizeof(int));
193 int * const bpos = rw_xmalloc(((size_t)m + 1) * sizeof(int));
194 memcpy(bpos, bb, ((size_t)m + 1) * sizeof(int));
195 for (int j = 0; j < n; j++)
196 if (cp[j] != cp[j + 1]) bcol[bpos[ci[cp[j]]]++] = j;
197 free(bpos);
198 /* the columns by their hash (chained) */
199 uint32_t hsz = 16;
200 while ((uint64_t)hsz < 2ULL * (uint64_t)n) hsz <<= 1;
201 int * const head = rw_xmalloc((size_t)hsz * sizeof(int));
202 int * const nxt = rw_xmalloc((size_t)n * sizeof(int));
203 for (uint32_t s = 0; s < hsz; s++) head[s] = -1;
204 for (int j = 0; j < n; j++) {
205 if (cp[j] == cp[j + 1]) continue;
206 const uint32_t s = (uint32_t)(h[j] >> 32) & (hsz - 1);
207 nxt[j] = head[s];
208 head[s] = j;
209 }
210 /* the triplets: for each pair {a, b} of columns with the same smallest row, the columns c with h_c = h_a + h_b */
211 long long cap = 1024, nt = 0;
212 int *tr = rw_xmalloc((size_t)cap * 3 * sizeof(int));
213 for (int r = 0; r < m; r++) {
214 for (int i1 = bb[r]; i1 < bb[r + 1]; i1++) {
215 const int a = bcol[i1];
216 for (int i2 = i1 + 1; i2 < bb[r + 1]; i2++) {
217 const int b = bcol[i2];
218 const uint64_t th = h[a] ^ h[b];
219 if (!th) continue; /* equal columns a, b: c would be a zero column */
220 for (int c = head[(uint32_t)(th >> 32) & (hsz - 1)]; c >= 0; c = nxt[c]) {
221 if (h[c] != th || c == a || c == b || !t3_match(cp, ci, cm, a, b, c)) continue;
222 if (nt == cap) {
223 cap *= 2;
224 tr = realloc(tr, (size_t)cap * 3 * sizeof(int));
225 if (!tr) ERROR("memory allocation");

227 tr[3 * nt] = a;
228 tr[3 * nt + 1] = b;
229 tr[3 * nt + 2] = c;
230 nt++;
231 }
232 }
233 }
234 }
235 free(head);
236 free(nxt);
237 free(bcol);
238 free(bb);
239 free(h);
240 free(cp);
241 free(ci);
242 if (nt == 0 || 6 * nt > ((long long)1 << 30)) { /* none (or far too many for the tables) */
243 if (nt > 0) t->complete = 0;
244 free(tr);
245 t->ms = 1e3 * (rw_wall() - t0);
246 return t;
247 }
248 t->ntrip = nt;
249 /* the pair table (at most half full) */
250 uint32_t ps = 16;
251 while ((uint64_t)ps < 6ULL * (uint64_t)nt + 32) ps <<= 1;
252 t->pmask = ps - 1;
253 t->pkey = rw_xcalloc(ps, sizeof(uint64_t));
254 t->pval = rw_xcalloc(ps, sizeof(int));
255 for (long long k = 0; k < nt; k++) {
256 const int a = tr[3 * k], b = tr[3 * k + 1], c = tr[3 * k + 2];
257 t3_pair_insert(t, a, b, c);
258 t3_pair_insert(t, a, c, b);
259 t3_pair_insert(t, b, c, a);
260 }
261 /* the triplets of each column */
262 t->tp = rw_xcalloc((size_t)n + 1, sizeof(int));
263 for (long long k = 0; k < 3 * nt; k++) t->tp[tr[k] + 1]++;
264 for (int u = 0; u < n; u++) t->tp[u + 1] += t->tp[u];
265 t->tq = rw_xmalloc((size_t)nt * 6 * sizeof(int));
266 int * const cur = rw_xmalloc(((size_t)n + 1) * sizeof(int));
267 memcpy(cur, t->tp, ((size_t)n + 1) * sizeof(int));
268 for (long long k = 0; k < nt; k++) {
269 const int a = tr[3 * k], b = tr[3 * k + 1], c = tr[3 * k + 2];
270 t->tq[2 * cur[a]] = b;
271 t->tq[2 * cur[a] + 1] = c;
272 cur[a]++;
273 t->tq[2 * cur[b]] = a;
274 t->tq[2 * cur[b] + 1] = c;
275 cur[b]++;
276 t->tq[2 * cur[c]] = a;
277 t->tq[2 * cur[c] + 1] = b;
278 cur[c]++;
279 }
280 free(tr);
281 /* each list of triplets in a canonical order (the pairs of other columns, smaller first, sorted), so that the
282 * stages do not depend on the order of the rows of H, which gives the order of the enumeration */
283 for (int u = 0; u < n; u++) {
284 for (int k = t->tp[u]; k < t->tp[u + 1]; k++) {
285 if (t->tq[2 * k] > t->tq[2 * k + 1]) {
286 const int x = t->tq[2 * k];
287 t->tq[2 * k] = t->tq[2 * k + 1];
288 t->tq[2 * k + 1] = x;
289 }
290 }
291 if (t->tp[u + 1] - t->tp[u] > 1)
292 qsort(t->tq + 2 * t->tp[u], (size_t)(t->tp[u + 1] - t->tp[u]), 2 * sizeof(int), rw_cmp_pair);
293 }
294 /* the triplet graph G_3: the distinct neighbors of each column */
295 t->gp = rw_xmalloc(((size_t)n + 1) * sizeof(int));
296 t->gi = rw_xmalloc((size_t)nt * 6 * sizeof(int));
297 int ng = 0;
298 for (int u = 0; u < n; u++) {
299 t->gp[u] = ng;
300 const int k0 = 2 * t->tp[u], k1 = 2 * t->tp[u + 1];
301 memcpy(t->gi + ng, t->tq + k0, (size_t)(k1 - k0) * sizeof(int));
302 rw_sort(t->gi + ng, k1 - k0);
303 int q = 0;
304 for (int k = 0; k < k1 - k0; k++)
305 if (q == 0 || t->gi[ng + q - 1] != t->gi[ng + k]) t->gi[ng + q++] = t->gi[ng + k];
306 ng += q;
307 }
308 t->gp[n] = ng;
309 free(cur);
310 t->ms = 1e3 * (rw_wall() - t0);
311 return t;
312}

327int rw_t3_contract(const rw_t3_t * const t, int * const supp, int cnt) {
328 if (!t || t->ntrip <= 0) return cnt;
329 int changed = 1;
330 while (changed && cnt >= 2) {
331 changed = 0;
332 for (int i = 0; i < cnt && !changed; i++) {
333 const int u = supp[i];
334 for (int j = i + 1; j < cnt; j++) {
335 const int w = rw_t3_pair(t, u, supp[j]);
336 if (w < 0) continue;
337 supp[j] = supp[--cnt]; /* (j > i: u stays at i) */
338 int k = -1;
339 for (int q = 0; q < cnt; q++) {
340 if (q != i && supp[q] == w) {
341 k = q;
342 break;
343 }
344 }
345 if (k >= 0) { /* 3 -> 0: remove u (at i) and w (at k), the larger position first */
346 const int hi = (i > k) ? i : k, lo = (i > k) ? k : i;
347 supp[hi] = supp[--cnt];
348 supp[lo] = supp[--cnt];
349 } else { /* 2 -> 1 */
350 supp[i] = w;
351 }
352 changed = 1;
353 break;
354 }
355 }
356 }
357 return cnt;
358}

505int rw_gdb_shorten(const rw_gdb_t * const g, rw_tws_t * const ws, int * const supp, int cnt) {
506 if (!g || cnt <= 0) return cnt;
507 const uint32_t gen = tws_gen(ws);
508 const uint32_t rg = tws_rgen(ws, g->nrows);
509 for (int i = 0; i < cnt; i++) {
510 const int u = supp[i];
511 ws->mark[u] = gen;
512 ws->pos[u] = i;
513 tws_overlap(g, ws, u, 1, rg);
514 }
515 for (;;) {
516 int best = -1, bgain = 0;
517 for (int k = 0; k < ws->nrows_t; k++) {
518 const int r = ws->rows[k];
519 const int gain = 2 * ws->ov[r] - (g->rp[r + 1] - g->rp[r]);
520 if (gain > 0 && (gain > bgain || (gain == bgain && r < best))) {
521 best = r;
522 bgain = gain;
523 }
524 }
525 if (best < 0) break;
526 for (int k = g->rp[best]; k < g->rp[best + 1]; k++) {
527 const int u = g->ri[k];
528 if (ws->mark[u] == gen) { /* remove u */
529 const int i = ws->pos[u], last = supp[--cnt];
530 supp[i] = last;
531 ws->pos[last] = i;
532 ws->mark[u] = 0;
533 tws_overlap(g, ws, u, -1, rg);
534 } else { /* add u */
535 supp[cnt] = u;
536 ws->pos[u] = cnt++;
537 ws->mark[u] = gen;
538 tws_overlap(g, ws, u, 1, rg);
539 }
540 }
541 }
542 return cnt;
543}

545int rw_gdb_is_row(const rw_gdb_t * const g, rw_tws_t * const ws, const int * const supp, const int cnt,
546 const uint64_t key) {
547 if (!g || cnt <= 0 || cnt > g->wmax || !g->wpres[cnt]) return 0;
548 int marked = 0;
549 uint32_t gen = 0;
550 for (uint32_t s = rw_hash_slot(key, g->hmask); g->hval[s]; s = (s + 1) & g->hmask) {
551 if (g->hkey[s] != key) continue;
552 const int r = g->hval[s] - 1;
553 if (g->rp[r + 1] - g->rp[r] != cnt) continue;
554 if (!marked) {
555 gen = tws_gen(ws);
556 for (int i = 0; i < cnt; i++) ws->mark[supp[i]] = gen;
557 marked = 1;
558 }
559 int k = g->rp[r];
560 while (k < g->rp[r + 1] && ws->mark[g->ri[k]] == gen) k++;
561 if (k == g->rp[r + 1]) return 1;
562 }
563 return 0;
564}

570static int rw_cmp_ent(const void *a, const void *b) { /* triples (hub, column, partner) by hub, then column */
571 const int * const x = a, * const y = b;
572 if (x[0] != y[0]) return (x[0] > y[0]) - (x[0] < y[0]);
573 if (x[1] != y[1]) return (x[1] > y[1]) - (x[1] < y[1]);
574 return (x[2] > y[2]) - (x[2] < y[2]);
575}

577/* The 2 <-> 2 exchanges of the support c (marked with `gen`): the triples (z, u, v) for each column u of c and each
578 * triplet {u, v, z} with v not in c (both choices of the hub z), sorted by z; two triples (z, u_a, v_a), (z, u_b, v_b)
579 * with u_a != u_b and v_a != v_b give the exchange c + {u_a, v_a, u_b, v_b} of the same weight. Returns the number of
580 * triples (in ws->ent). */
581static size_t t3_exchanges(const rw_t3_t * const t, rw_tws_t * const ws, const int * const c, const int cnt,
582 const uint32_t gen) {
583 size_t need = 0;
584 for (int a = 0; a < cnt; a++) need += 2 * (size_t)(t->tp[c[a] + 1] - t->tp[c[a]]);
585 if (need > ws->entcap) {
586 free(ws->ent);
587 ws->entcap = 2 * need + 64;
588 ws->ent = rw_xmalloc(ws->entcap * 3 * sizeof(int));
589 }
590 size_t ne = 0;
591 int * const e = ws->ent;
592 for (int a = 0; a < cnt; a++) {
593 const int ua = c[a];
594 for (int k = t->tp[ua]; k < t->tp[ua + 1]; k++) {
595 const int x = t->tq[2 * k], y = t->tq[2 * k + 1];
596 if (ws->mark[y] != gen) {
597 e[3 * ne] = x;
598 e[3 * ne + 1] = ua;
599 e[3 * ne + 2] = y;
600 ne++;
601 }
602 if (ws->mark[x] != gen) {
603 e[3 * ne] = y;
604 e[3 * ne + 1] = ua;
605 e[3 * ne + 2] = x;
606 ne++;
607 }
608 }
609 }
610 if (ne > 1) qsort(e, ne, 3 * sizeof(int), rw_cmp_ent);
611 return ne;
612}

614/* Is the support c + Q1 lexicographically smaller than c + Q2 (Q1, Q2: sets of four columns; `inc`: the membership
615 * marks of c with `gen`)? The smallest column m in exactly one of Q1, Q2 decides: c + Q1 contains m iff m is added by
616 * Q1 (not in c) or removed by Q2 (in c). */
617static int q4_less(const rw_tws_t * const ws, const uint32_t gen, const int * const q1, const int * const q2) {
618 int m = INT_MAX, in1 = 0;
619 for (int i = 0; i < 4; i++) {
620 int k = 0;
621 while (k < 4 && q2[k] != q1[i]) k++;
622 if (k == 4 && q1[i] < m) {
623 m = q1[i];
624 in1 = 1;
625 }
626 k = 0;
627 while (k < 4 && q1[k] != q2[i]) k++;
628 if (k == 4 && q2[i] < m) {
629 m = q2[i];
630 in1 = 0;
631 }
632 }
633 if (m == INT_MAX) return 0;
634 const int in_c = (ws->mark[m] == gen);
635 return in1 ? !in_c : in_c;
636}

638/* canon_T3 (Proposition 4): the greedy lexicographic 2 <-> 2 canonicalization (c sorted, kept sorted) */
639static void canon_t3(const rw_t3_t * const t, rw_tws_t * const ws, int * const c, const int cnt) {
640 const int max_iter = 4 * cnt + 16;
641 for (int it = 0; it < max_iter; it++) {
642 const uint32_t gen = tws_gen(ws);
643 for (int a = 0; a < cnt; a++) ws->mark[c[a]] = gen;
644 const size_t ne = t3_exchanges(t, ws, c, cnt, gen);
645 const int * const e = ws->ent;
646 int best[4] = {0, 0, 0, 0}, found = 0;
647 for (size_t g0 = 0; g0 < ne;) {
648 size_t g1 = g0 + 1;
649 while (g1 < ne && e[3 * g1] == e[3 * g0]) g1++;
650 for (size_t i = g0; i < g1; i++) {
651 for (size_t j = i + 1; j < g1; j++) {
652 const int ua = e[3 * i + 1], va = e[3 * i + 2], ub = e[3 * j + 1], vb = e[3 * j + 2];
653 if (ua == ub || va == vb) continue;
654 /* smaller than c iff the smallest of the four columns is added */
655 const int mu = (ua < ub) ? ua : ub, mv = (va < vb) ? va : vb;
656 if (mv > mu) continue;
657 const int q[4] = {ua, ub, va, vb};
658 if (!found || q4_less(ws, gen, q, best)) {
659 memcpy(best, q, sizeof(best));
660 found = 1;
661 }
662 }
663 }
664 g0 = g1;
665 }
666 if (!found) break;
667 for (int a = 0; a < cnt; a++) { /* replace u_a, u_b by v_a, v_b */
668 if (c[a] == best[0]) c[a] = best[2];
669 else if (c[a] == best[1]) c[a] = best[3];
670 }
671 rw_sort(c, cnt);
672 }
673}

675/* Is the support c + r1 lexicographically smaller than c + r2 for two rows r1 != r2 of G (sorted)? */
676static int row_less(const rw_gdb_t * const g, const rw_tws_t * const ws, const uint32_t gen, const int r1,
677 const int r2) {
678 int i = g->rp[r1], j = g->rp[r2];
679 const int ie = g->rp[r1 + 1], je = g->rp[r2 + 1];
680 while (i < ie || j < je) {
681 if (j >= je || (i < ie && g->ri[i] < g->ri[j])) {
682 const int in_c = (ws->mark[g->ri[i]] == gen); /* in r1 only */
683 return !in_c;
684 }
685 if (i >= ie || g->ri[j] < g->ri[i]) {
686 const int in_c = (ws->mark[g->ri[j]] == gen); /* in r2 only */
687 return in_c;
688 }
689 i++;
690 j++;
691 }
692 return 0; /* equal rows */
693}

695/* canon_E (Proposition 6): the greedy lexicographic half-overlap exchanges with the even rows of G (c sorted) */
696static void canon_e(const rw_gdb_t * const g, rw_tws_t * const ws, int * const c, const int cnt) {
697 const int max_iter = 4 * cnt + 16;
698 for (int it = 0; it < max_iter; it++) {
699 const uint32_t gen = tws_gen(ws);
700 const uint32_t rg = tws_rgen(ws, g->nrows);
701 for (int a = 0; a < cnt; a++) {
702 ws->mark[c[a]] = gen;
703 tws_overlap(g, ws, c[a], 1, rg);
704 }
705 int best = -1;
706 for (int k = 0; k < ws->nrows_t; k++) {
707 const int r = ws->rows[k], w = g->rp[r + 1] - g->rp[r];
708 if ((w & 1) || 2 * ws->ov[r] != w) continue;
709 if (ws->mark[g->ri[g->rp[r]]] == gen) continue; /* smaller iff the smallest column of the row is added */
710 if (best < 0 || row_less(g, ws, gen, r, best)) best = r;
711 }
712 if (best < 0) break;
713 int q = 0;
714 for (int a = 0; a < cnt; a++) { /* the columns of c not in the row (the row is sorted) */
715 const int u = c[a];
716 int lo = g->rp[best], hi = g->rp[best + 1];
717 while (lo < hi) {
718 const int mid = (lo + hi) >> 1;
719 if (g->ri[mid] < u) lo = mid + 1;
720 else hi = mid;
721 }
722 if (lo < g->rp[best + 1] && g->ri[lo] == u) continue;
723 ws->tmp[q++] = u;
724 }
725 for (int k = g->rp[best]; k < g->rp[best + 1]; k++)
726 if (ws->mark[g->ri[k]] != gen) ws->tmp[q++] = g->ri[k];
727 if (q != cnt) break; /* (not reached: a half-overlap exchange keeps the weight) */
728 memcpy(c, ws->tmp, (size_t)cnt * sizeof(int));
729 rw_sort(c, cnt);
730 }
731}

733void rw_canon(const rw_t3_t * const t, const rw_gdb_t * const g, rw_tws_t * const ws, int * const supp,
734 const int cnt) {
735 if (cnt < 2) return;
736 if (t && t->ntrip > 0) canon_t3(t, ws, supp, cnt);
737 else if (g && g->even > 0) canon_e(g, ws, supp, cnt);
738}

740int rw_cw_reduce(const csr_t * const HT, const colmask_t * const maskL, rw_tws_t * const ws, int * const arr, int w) {
741 int k;
742 while ((k = csr_cols_dependency(HT, w, arr, ws->irr, ws->sub)) > 0) {
743 int nr = 0;
744 for (int i = 0, j = 0; i < w; i++) {
745 if (j < k && ws->sub[j] == arr[i]) j++;
746 else ws->rest[nr++] = arr[i];
747 }
748 const int nt_sub = !maskL || colmask_syndrome_non_zero(maskL, k, ws->sub);
749 const int nt_rest = !maskL || colmask_syndrome_non_zero(maskL, nr, ws->rest);
750 const int keep_sub = nt_sub && (!nt_rest || k <= nr);
751 memcpy(arr, keep_sub ? ws->sub : ws->rest, (size_t)(keep_sub ? k : nr) * sizeof(int));
752 w = keep_sub ? k : nr;
753 }
754 return w;
755}

791 *is_new = 0;
792 size_t s = tab_slot(key, tab->cap - 1);
793 while (tab->key[s]) {

797 if (tab->n >= RW_CANON_MAX) {

811 }

813 free(tab->inc);

815 tab->inc = i2;
816 tab->cap = cap;
817 s = tab_slot(key, cap - 1);
818 while (tab->key[s]) s = (s + 1) & (cap - 1);
819 }
820 tab->key[s] = key;
821 tab->inc[s] = 0;
822 tab->n++;
823 *is_new = 1;
824 return &tab->inc[s];
825}

861/* The queue of the breadth-first search: the supports of the codewords to expand */
862typedef struct {
863 int *pool;
864 size_t len, cap; /* ints used and allocated in pool */
865 size_t head; /* the next codeword: its weight at pool[head], then its columns */
866} rw_bfs_t;
867
868static void bfs_push(rw_bfs_t * const q, const int * const c, const int w) {
869 if (q->len + (size_t)w + 1 > q->cap) {
870 q->cap = 2 * (q->len + (size_t)w + 1) + 1024;
871 q->pool = realloc(q->pool, q->cap * sizeof(int));
872 if (!q->pool) ERROR("memory allocation");
873 }
874 q->pool[q->len++] = w;
875 memcpy(q->pool + q->len, c, (size_t)w * sizeof(int));
876 q->len += (size_t)w;
877}

879/* Add the codeword c (w columns, any order; sorted in place) to the hash if new and irreducible (irred=1); returns 1 if
880 * added (and queued), -1 if maxC codewords are collected, 0 otherwise */
881static int bfs_try(params_t * const p, const csr_t * const HT, rw_tws_t * const ws, rw_bfs_t * const q, int * const c,
882 const int w) {
883 rw_sort(c, w);
884 cw_vec_t *known = NULL;
885 HASH_FIND(hh, p->codewords, c, (size_t)w * sizeof(int), known);
886 if (known) return 0;
887 if (codeword_maxc_reached(p)) return -1;
888 if (p->irred && w >= 3 && csr_cols_rank(HT, w, c, ws->irr) != w - 1) return 0;
889 p->codewords = codeword_add_maybe(p, c, w, CW_SRC_OTHER);
890 HASH_FIND(hh, p->codewords, c, (size_t)w * sizeof(int), known);
891 if (!known) return 0;
892 bfs_push(q, c, w);
893 return 1;
894}

896/* The identical columns of [H; L] (the same nonzero column of H and the same column of L), for the 1 <-> 1 swaps of the
897 * expansion: twin[j] is the next column of the class of j (a cycle, in increasing order; j itself without an identical
898 * column), NULL if no two columns are identical; *pcnt: the number of columns in the classes of two or more */
899typedef struct {
900 uint64_t h;
901 int j;
902} t2_key_t;
903
904static int t2_key_cmp(const void *a, const void *b) {
905 const t2_key_t *x = a, *y = b;
906 if (x->h != y->h) return (x->h < y->h) ? -1 : 1;
907 return (x->j > y->j) - (x->j < y->j);
908}
909
910static int t2_cols_equal(const csr_t * const HT, const colmask_t * const mL, const int i, const int j) {
911 const int li = HT->p[i + 1] - HT->p[i];
912 if (li != HT->p[j + 1] - HT->p[j] || memcmp(HT->i + HT->p[i], HT->i + HT->p[j], (size_t)li * sizeof(int)))
913 return 0;
914 return !mL || !memcmp(mL->mask + (size_t)i * mL->kw, mL->mask + (size_t)j * mL->kw, (size_t)mL->kw * sizeof(word));
915}
916
917static int *t2_twins(const params_t * const p, const csr_t * const HT, int * const pcnt) {
918 const int n = HT->rows;
919 const colmask_t * const mL = p->maskL;
920 *pcnt = 0;
921 t2_key_t * const key = rw_xmalloc((size_t)(n > 0 ? n : 1) * sizeof(t2_key_t));
922 int nz = 0;
923 for (int j = 0; j < n; j++) {
924 if (HT->p[j + 1] == HT->p[j]) continue; /* (a zero column of H: a codeword of weight 1, or a zero column) */
925 uint64_t h = (uint64_t)(HT->p[j + 1] - HT->p[j]);
926 for (int k = HT->p[j]; k < HT->p[j + 1]; k++) h = rw_mix64(h ^ (uint64_t)HT->i[k]);
927 for (int k = 0; mL && k < mL->kw; k++) h = rw_mix64(h ^ (uint64_t)mL->mask[(size_t)j * mL->kw + k]);
928 key[nz].h = h;
929 key[nz++].j = j;
930 }
931 qsort(key, nz, sizeof(t2_key_t), t2_key_cmp);
932 int *twin = NULL;
933 for (int a = 0; a < nz;) {
934 int b = a + 1;
935 while (b < nz && key[b].h == key[a].h) b++;
936 for (int s = a; s < b && b - a > 1; s++) { /* the classes in a run of equal hashes (the columns in order) */
937 const int j = key[s].j;
938 if (j < 0) continue; /* (already in a class) */
939 int last = j;
940 for (int u = s + 1; u < b; u++) {
941 if (key[u].j < 0 || !t2_cols_equal(HT, mL, j, key[u].j)) continue;
942 if (!twin) {
943 twin = rw_xmalloc((size_t)n * sizeof(int));
944 for (int v = 0; v < n; v++) twin[v] = v;
945 }
946 twin[last] = key[u].j;
947 last = key[u].j;
948 key[u].j = -1;
949 (*pcnt)++;
950 }
951 if (last != j) {
952 twin[last] = j;
953 (*pcnt)++;
954 }
955 }
956 a = b;
957 }
958 free(key);
959 return twin;
960}

962long long rw_t3_expand(params_t * const p, const rw_t3_t * const t, const csr_t * const HT, rw_tws_t * const ws,
963 const long long limit, int * const ntwin) {
964 if (ntwin) *ntwin = 0;
965 if (p->min_w == INT_MAX || !p->codewords) return 0;
966 const int have_t3 = t && t->ntrip > 0;
967 int nt = 0;
968 int * const twin = t2_twins(p, HT, &nt);
969 if (ntwin) *ntwin = nt;
970 if (!have_t3 && !twin) return 0;
971 const int wtop = p->min_w + (p->dW > 0 ? p->dW : 0);
972 rw_bfs_t q = {NULL, 0, 0, 0};
973 for (cw_vec_t *cw = p->codewords; cw != NULL; cw = (cw_vec_t *)(cw->hh.next))
974 if (cw->weight <= wtop) bfs_push(&q, cw->arr, cw->weight);
975 int * const c = rw_xmalloc(((size_t)wtop + 2) * sizeof(int));
976 int * const nb = rw_xmalloc(((size_t)wtop + 2) * sizeof(int));
977 long long added = 0;
978 int stop = 0;
979 while (q.head < q.len && !stop) {
980 const int w = q.pool[q.head];
981 memcpy(c, q.pool + q.head + 1, (size_t)w * sizeof(int));
982 q.head += (size_t)w + 1;
983 const uint32_t gen = tws_gen(ws);
984 for (int a = 0; a < w; a++) ws->mark[c[a]] = gen;
985 /* the equal-weight 2 <-> 2 exchanges (the triples are kept in ws->ent until the next generation) */
986 const size_t ne = have_t3 ? t3_exchanges(t, ws, c, w, gen) : 0;
987 for (size_t g0 = 0; g0 < ne && !stop;) {
988 size_t g1 = g0 + 1;
989 while (g1 < ne && ws->ent[3 * g1] == ws->ent[3 * g0]) g1++;
990 for (size_t i = g0; i < g1 && !stop; i++) {
991 for (size_t j = i + 1; j < g1 && !stop; j++) {
992 const int ua = ws->ent[3 * i + 1], va = ws->ent[3 * i + 2];
993 const int ub = ws->ent[3 * j + 1], vb = ws->ent[3 * j + 2];
994 if (ua == ub || va == vb) continue;
995 for (int a = 0; a < w; a++) nb[a] = (c[a] == ua) ? va : ((c[a] == ub) ? vb : c[a]);
996 const int r = bfs_try(p, HT, ws, &q, nb, w);
997 if (r < 0) stop = 1;
998 else if (r > 0 && ++added >= limit) stop = 2;
999 }
1000 }
1001 g0 = g1;
1002 }
1003 /* the equal-weight 1 <-> 1 swaps of identical columns (the Stern filter skips a column identical to a pivot
1004 * column); all combinations through the search */
1005 for (int a = 0; twin && a < w && !stop; a++) {
