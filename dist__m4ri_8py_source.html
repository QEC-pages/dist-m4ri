2dist_m4ri.py: Python wrapper for the multithreaded dist_m4ri distance calculator.

74def get_sparse_array_state(A) -> Any:
75 """Returns a hashable state representing a numpy/scipy matrix for cache keys."""
76 if A is None:
77 return None
78 if isinstance(A, (str, Path)):
79 return str(A)
80 if isinstance(A, np.ndarray):
81 return (A.shape, A.dtype.str, A.tobytes())
82 if issparse(A):
83 csr = A.tocsr()
84 return (csr.shape, csr.data.tobytes(), csr.indices.tobytes(), csr.indptr.tobytes())
85 if hasattr(A, 'tobytes'):
86 return A.tobytes()
87 return str(A)

98def read_sparse_vectors(filepath: str) -> List[List[int]]:
99 """
100 Reads a list of sparse vectors from a text file in NZLIST format,
101 converting from 1-based indexing (in the file) to 0-based indexing (in Python).
102
103 Args:
104 filepath (str): The path to the text file.
105
106 Returns:
107 list of list of int: A list where each element is a 0-based sparse vector.
108 """
109 sparse_vectors = []
110 if not os.path.exists(filepath):
111 return sparse_vectors
112
113 with open(filepath, 'r') as f:
114 first_line = f.readline().strip()
115 if first_line != '%% NZLIST':
116 raise ValueError(f"Invalid file format in {filepath}: Missing '%% NZLIST' header.")
117
118 for line_num, line in enumerate(f, start=2):
119 line = line.strip()
120 if not line or line.startswith('%'):
121 continue
122 try:
123 parts = list(map(int, line.split()))
124 except ValueError:
125 raise ValueError(f"Non-integer data found on line {line_num}: {line}")
126
127 stated_length = parts[0]
128 vector_elements = [x - 1 for x in parts[1:]]
129 if len(vector_elements) != stated_length:
130 raise ValueError(
131 f"Length mismatch on line {line_num}. "
132 f"Expected {stated_length} elements, but found {len(vector_elements)}."
133 )
134 sparse_vectors.append(vector_elements)
135
136 return sparse_vectors

168def parse_dist_m4ri_output(stdout: str) -> Tuple[int, int, int]:
169 """
170 Parses the standard output of dist_m4ri.
171 Expected format on stdout: "dmin dmax rw_steps", "dmin dmax", or a single integer.
172
173 Returns:
174 tuple (dmin, dmax, rw_steps)
175 """
176 lines = stdout.strip().split('\n')
177 for line in reversed(lines):
178 line = line.strip()
179 if not line or line.startswith('#'):
180 continue
181 parts = line.split()
182 if len(parts) >= 3:
183 try:
184 return int(parts[0]), int(parts[1]), int(parts[2])
185 except ValueError:
186 continue
187 elif len(parts) == 2:
188 try:
189 return int(parts[0]), int(parts[1]), 0
190 except ValueError:
191 continue
192 elif len(parts) == 1:
193 try:
194 val = int(parts[0])
195 return val, val, 0
196 except ValueError:
197 continue
198
199 raise RuntimeError(f"Could not parse dist_m4ri output: {stdout}")

202def run_dist_m4ri(
203 dist_m4ri_path: Optional[str] = None,
204 method: int = 3,
205 finH: Optional[str] = None,
206 finG: Optional[str] = None,
207 finL: Optional[str] = None,
208 fdem: Optional[str] = None,
209 wmax: int = 0,
210 wmin: int = 1,
211 dexp: int = 0,
212 dest: int = 0,
213 steps: Optional[int] = None,
214 threads: Optional[int] = None,
215 timeout: float = 60.0,
216 smax: Optional[int] = None,
217 noscan: int = 0,
218 classical: int = -1,
219 dW: int = -1,
220 maxC: int = 0,
221 pmin: float = 0.0,
222 outC: Optional[str] = None,
223 seed: int = 0,
224 debug: int = 0,
225 stop_event: Optional[threading.Event] = None
226) -> Tuple[int, int, int]:
227 """
228 Low-level invocation of the multithreaded dist_m4ri binary.
229
230 Returns:
231 tuple (dmin, dmax, rw_steps)
232 """
233 exec_path = find_dist_m4ri_binary(dist_m4ri_path)
234 cmd = [exec_path, f"debug={debug}", f"method={method}"]
235
236 if finH: cmd.append(f"finH={finH}")
237 if finG: cmd.append(f"finG={finG}")
238 if finL: cmd.append(f"finL={finL}")
239 if fdem: cmd.append(f"fdem={fdem}")
240 if wmax > 0: cmd.append(f"wmax={wmax}")
241 if wmin > 1: cmd.append(f"wmin={wmin}")
242 if dexp > 0: cmd.append(f"dexp={dexp}")
243 elif dest > 0: cmd.append(f"dest={dest}")
244 if steps is not None and steps > 0: cmd.append(f"steps={steps}")
245 if threads is not None and threads > 0: cmd.append(f"threads={threads}")
246 if timeout > 0: cmd.append(f"timeout={timeout}")
247 if smax is not None: cmd.append(f"smax={smax}")
248 if noscan: cmd.append(f"noscan={noscan}")
249 if classical >= 0: cmd.append(f"classical={classical}")
250 if dW >= 0: cmd.append(f"dW={dW}")
251 if maxC > 0: cmd.append(f"maxC={maxC}")
252 if pmin > 0.0: cmd.append(f"pmin={pmin}")
253 if outC: cmd.append(f"outC={outC}")
254 if seed != 0: cmd.append(f"seed={seed}")
255
256 if debug & 2:
257 print(f"[dist_m4ri] Running: {' '.join(cmd)}")
258
259 proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
260
261 if stop_event is not None:
262 while proc.poll() is None:
263 if stop_event.is_set():
264 proc.terminate()
265 try:
266 proc.wait(timeout=1.0)
267 except subprocess.TimeoutExpired:
268 proc.kill()
269 raise RuntimeError("dist_m4ri execution cancelled by stop_event")
270 time.sleep(0.05)
271 stdout, stderr = proc.communicate()
272 else:
273 stdout, stderr = proc.communicate()
274
275 if proc.returncode != 0:
276 raise RuntimeError(f"dist_m4ri failed with exit code {proc.returncode}:\n{stderr}")
277
278 return parse_dist_m4ri_output(stdout)

281def _matrix_to_file(matrix, extension: str = ".mtx", temp_dir: str = "tmp") -> str:
282 """Helper to convert a matrix (numpy or scipy sparse) or file path to an MTX file path."""
283 if isinstance(matrix, (str, Path)):
284 return str(matrix)
285
286 path = create_unique_file(directory=temp_dir, extension=extension)
287 if issparse(matrix):
288 csr = matrix.astype(np.int8)
289 mmwrite(path, csr, symmetry='general')
290 else:
291 mat_arr = np.asarray(matrix, dtype=np.int8)
292 csr = csr_matrix(mat_arr)
293 mmwrite(path, csr, symmetry='general')
294 return path

297def compute_classical_distance(
298 H: Any,
299 dist_m4ri: Optional[str] = None,
300 method: int = 3,
301 threads: Optional[int] = None,
302 timeout: float = 60.0,
303 num_steps: Optional[int] = None,
304 d_exp: int = 0,
305 wmax: int = 0,
306 dW: int = -1,
307 maxC: int = 0,
308 do_cws: bool = False,
309 solver: str = "dist_m4ri",
310 codedistance_method: str = "QDistEvol",
311 codedistance_params: Optional[Dict[str, Any]] = None,
312 seed: int = 0,
313 debug: int = 0
314) -> Union[int, Tuple[int, List[List[int]]]]:
315 """
316 Computes the minimum distance of a classical linear code given parity check matrix H.
317
318 Args:
319 H: Parity check matrix (numpy array, scipy sparse matrix, or file path).
320 dist_m4ri: Path to dist_m4ri executable (optional).
321 method: Solver method (1=RW, 2=CC, 3=Bracketing default).
322 threads: Number of worker threads.
323 timeout: Execution timeout in seconds.
324 num_steps: Maximum RW steps.
325 d_exp: Expected distance estimate.
326 wmax: Maximum weight to search in CC.
327 dW: Extra weight window above dmin to collect codewords.
328 maxC: Maximum number of codewords to collect.
329 do_cws: Whether to return extracted codewords.
330 solver: "dist_m4ri" or "codedistance".
331 codedistance_method: Method if using codedistance library.
332 codedistance_params: Extra parameters for codedistance library.
333 seed: Random seed.
334 debug: Debug level flags.
335
336 Returns:
337 dist or (dist, cws) if do_cws is True
338 """
339 global _distance_cache, _use_distance_cache
340 cache_key = None
341 if _use_distance_cache:
342 try:
343 h_state = get_sparse_array_state(H)
344 cache_key = ("classical", h_state, method, d_exp, wmax, dW, maxC, do_cws, solver, codedistance_method)
345 if cache_key in _distance_cache:
346 if debug & 4:
347 print("[dist_m4ri] Cache hit for classical distance!")
348 return _distance_cache[cache_key]
349 except Exception:
350 cache_key = None
351
352 if solver == "codedistance":
353 if not _HAS_CODEDISTANCE:
354 raise ImportError("codedistance library requested but not installed.")
355 params = dict(codedistance_params or {})
356 if num_steps is not None and "iterCount" not in params:
357 params["iterCount"] = num_steps
358
359 H_mat = H.toarray() if issparse(H) else (np.asarray(H, dtype=np.int8) if isinstance(H, np.ndarray) else None)
360 res = codedistance.codeDistance(
361 H_mat, None, tB=1, method=codedistance_method, params=params,
362 seed=seed if seed != 0 else None
363 )
364 d = res.get("d", -1)
365 cws = []
366 if do_cws and res.get("L") is not None and np.sum(res["L"]) > 0:
367 cws = [list(np.where(res["L"])[0])]
368 res_val = (d, cws) if do_cws else d
369 if _use_distance_cache and cache_key is not None:
370 _distance_cache[cache_key] = res_val
371 return res_val
372
373 # Use native multithreaded dist_m4ri
374 temp_files = []
375 try:
376 if isinstance(H, (str, Path)) and os.path.exists(str(H)):
377 file_H = str(H)
378 else:
379 file_H = _matrix_to_file(H, extension="_H.mtx")
380 temp_files.append(file_H)
381
382 outC = None
383 if do_cws:
384 outC = create_unique_file(extension="_cws.nz")
385 temp_files.append(outC)
386
387 dmin, dmax, _ = run_dist_m4ri(
388 dist_m4ri_path=dist_m4ri,
389 method=method,
390 finH=file_H,
391 classical=1,
392 wmax=wmax,
393 dexp=d_exp,
394 steps=num_steps,
395 threads=threads,
396 timeout=timeout,
397 dW=dW,
398 maxC=maxC,
399 outC=outC,
400 seed=seed,
401 debug=debug
402 )
403
404 dist = dmin if (dmin == dmax or dmax == 0) else dmax
405 cws = []
406 if do_cws and outC and os.path.exists(outC):
407 cws = read_sparse_vectors(outC)
408 cws.sort(key=len)
409
410 res_val = (dist, cws) if do_cws else dist
411 if _use_distance_cache and cache_key is not None:
412 _distance_cache[cache_key] = res_val
413 return res_val
414
415 finally:
416 for f in temp_files:
417 if os.path.exists(f):
418 try: os.remove(f)
419 except OSError: pass

422def compute_css_distance(
423 Hx: Any,
424 Hz: Any,
425 Lx: Optional[Any] = None,
426 Lz: Optional[Any] = None,
427 dist_m4ri: Optional[str] = None,
428 method: int = 3,
429 threads: Optional[int] = None,
430 timeout: float = 60.0,
431 num_steps: Optional[int] = None,
432 d_exp: int = 0,
433 d_min: int = 0,
434 wmax: int = 0,
435 dW: int = -1,
436 maxC: int = 0,
437 do_cws: bool = False,
438 solver: str = "dist_m4ri",
439 codedistance_method: str = "QDistEvol",
440 codedistance_params: Optional[Dict[str, Any]] = None,
441 seed: int = 0,
442 debug: int = 0,
443 **kwargs
444) -> Tuple[Any, ...]:
445 """
446 Computes CSS quantum code distance d = min(d_X, d_Z).
447
448 Args:
449 Hx: X-stabilizer parity check matrix.
450 Hz: Z-stabilizer parity check matrix.
451 Lx: Optional X-logical operator matrix (alternative to Hz as finG).
452 Lz: Optional Z-logical operator matrix (alternative to Hx as finG).
453 dist_m4ri: Path to dist_m4ri executable (optional).
454 method: Solver method (1=RW, 2=CC, 3=Bracketing default).
455 threads: Number of worker threads.
456 timeout: Execution timeout in seconds.
457 num_steps: Maximum RW steps.
458 d_exp: Expected distance estimate.
459 d_min: Minimum target distance.
460 wmax: Maximum weight to search in CC.
461 dW: Extra weight window above dmin to collect codewords.
462 maxC: Maximum number of codewords to collect.
463 do_cws: Whether to return extracted X and Z codewords.
464 solver: "dist_m4ri" or "codedistance".
465 codedistance_method: Method if using codedistance library.
466 codedistance_params: Extra parameters for codedistance library.
467 seed: Random seed.
468 debug: Debug level flags.
469
470 Returns:
471 tuple (dist, dist_list_X, dist_list_Z, cws_X, cws_Z) if do_cws
472 else (dist, dist_list_X, dist_list_Z)
473 """
474 can_compute_Z = Hx is not None and (hasattr(Hx, 'shape') and Hx.shape[0] > 0 if not isinstance(Hx, str) else True)
475 can_compute_X = Hz is not None and (hasattr(Hz, 'shape') and Hz.shape[0] > 0 if not isinstance(Hz, str) else True)
476
477 if not can_compute_Z and not can_compute_X:
478 raise ValueError("Cannot compute CSS distance: Both Hx and Hz are empty.")
479
480 global _distance_cache, _use_distance_cache
481 cache_key = None
482 if _use_distance_cache:
483 try:
484 hx_state = get_sparse_array_state(Hx) if can_compute_Z else None
485 hz_state = get_sparse_array_state(Hz) if can_compute_X else None
486 cache_key = (
487 "css", hx_state, hz_state, method, d_exp, d_min, wmax, dW, maxC,
488 do_cws, solver, codedistance_method
489 )
490 if cache_key in _distance_cache:
491 if debug & 4:
492 print("[dist_m4ri] Cache hit for CSS distance!")
493 return _distance_cache[cache_key]
494 except Exception:
495 cache_key = None
496
497 if solver == "codedistance":
498 if not _HAS_CODEDISTANCE:
499 raise ImportError("codedistance library requested but not installed.")
500 params = dict(codedistance_params or {})
501 if num_steps is not None and "iterCount" not in params:
502 params["iterCount"] = num_steps
503
504 cws_Z, cws_X = [], []
505 dist_Z, dist_X = None, None
506 dist_list_Z, dist_list_X = [], []
507
508 Hx_mat = Hx.toarray() if issparse(Hx) else (np.asarray(Hx, dtype=np.int8) if isinstance(Hx, np.ndarray) else None)
509 Hz_mat = Hz.toarray() if issparse(Hz) else (np.asarray(Hz, dtype=np.int8) if isinstance(Hz, np.ndarray) else None)
510
511 if can_compute_Z:
512 res_Z = codedistance.CSScodeDistance(
513 Hx_mat, Hz_mat, method=codedistance_method, params=params.copy(),
514 component="Z", seed=seed if seed != 0 else None
515 )
516 dist_Z = res_Z.get("d", -1)
517 dist_list_Z = [dist_Z] if dist_Z > 0 else []
518 if do_cws and res_Z.get("L") is not None and np.sum(res_Z["L"]) > 0:
519 cws_Z = [list(np.where(res_Z["L"])[0])]
520
521 if can_compute_X:
522 res_X = codedistance.CSScodeDistance(
523 Hx_mat, Hz_mat, method=codedistance_method, params=params.copy(),
524 component="X", seed=seed if seed != 0 else None
525 )
526 dist_X = res_X.get("d", -1)
527 dist_list_X = [dist_X] if dist_X > 0 else []
528 if do_cws and res_X.get("L") is not None and np.sum(res_X["L"]) > 0:
529 cws_X = [list(np.where(res_X["L"])[0])]
530
531 if dist_X is not None and dist_Z is not None:
532 dist = min(dist_Z, dist_X) if (dist_Z > 0 and dist_X > 0) else max(dist_Z, dist_X)
533 elif dist_X is not None:
534 dist = dist_X
535 else:
536 dist = dist_Z
537
538 res_tuple = (dist, dist_list_X, dist_list_Z, cws_X, cws_Z) if do_cws else (dist, dist_list_X, dist_list_Z)
539 if _use_distance_cache and cache_key is not None:
540 _distance_cache[cache_key] = res_tuple
541 return res_tuple
542
543 # Native multithreaded dist_m4ri
544 temp_files = []
545 try:
546 file_Hx = _matrix_to_file(Hx, extension="_Hx.mtx") if can_compute_Z else None
547 file_Hz = _matrix_to_file(Hz, extension="_Hz.mtx") if can_compute_X else None
548 file_Lx = _matrix_to_file(Lx, extension="_Lx.mtx") if Lx is not None else None
549 file_Lz = _matrix_to_file(Lz, extension="_Lz.mtx") if Lz is not None else None
550
551 for f in (file_Hx, file_Hz, file_Lx, file_Lz):
552 if f and not isinstance(f, (str, Path)) or (f and not os.path.exists(f)):
553 pass
554 elif f and f.startswith(tempfile.gettempdir()):
555 temp_files.append(f)
556
557 outZ = create_unique_file(extension="_Z.nz") if (do_cws and can_compute_Z) else None
558 outX = create_unique_file(extension="_X.nz") if (do_cws and can_compute_X) else None
559 if outZ: temp_files.append(outZ)
560 if outX: temp_files.append(outX)
561
562 dist_Z, dist_X = None, None
563 dist_list_Z, dist_list_X = [], []
564 cws_Z, cws_X = [], []
565
566 # Z-distance: Hx as finH, Hz as finG (or Lz as finL)
567 if can_compute_Z:
568 dmin_z, dmax_z, _ = run_dist_m4ri(
569 dist_m4ri_path=dist_m4ri,
570 method=method,
571 finH=file_Hx,
572 finG=file_Hz if file_Lz is None else None,
573 finL=file_Lz,
574 wmax=wmax,
575 wmin=d_min,
576 dexp=d_exp,
577 steps=num_steps,
578 threads=threads,
579 timeout=timeout,
580 dW=dW,
581 maxC=maxC,
582 outC=outZ,
583 seed=seed,
584 debug=debug
585 )
586 dist_Z = dmin_z if (dmin_z == dmax_z or dmax_z == 0) else dmax_z
587 dist_list_Z = [dist_Z] if dist_Z > 0 else []
588 if do_cws and outZ and os.path.exists(outZ):
589 cws_Z = read_sparse_vectors(outZ)
590 cws_Z.sort(key=len)
591
592 # X-distance: Hz as finH, Hx as finG (or Lx as finL)
593 if can_compute_X:
594 dmin_x, dmax_x, _ = run_dist_m4ri(
595 dist_m4ri_path=dist_m4ri,
596 method=method,
597 finH=file_Hz,
598 finG=file_Hx if file_Lx is None else None,
599 finL=file_Lx,
600 wmax=wmax,
601 wmin=d_min,
602 dexp=d_exp,
603 steps=num_steps,
604 threads=threads,
605 timeout=timeout,
606 dW=dW,
607 maxC=maxC,
608 outC=outX,
609 seed=seed,
610 debug=debug
611 )
612 dist_X = dmin_x if (dmin_x == dmax_x or dmax_x == 0) else dmax_x
613 dist_list_X = [dist_X] if dist_X > 0 else []
614 if do_cws and outX and os.path.exists(outX):
615 cws_X = read_sparse_vectors(outX)
616 cws_X.sort(key=len)
617
618 if dist_X is not None and dist_Z is not None:
619 dist = min(dist_Z, dist_X) if (dist_Z > 0 and dist_X > 0) else max(dist_Z, dist_X)
620 elif dist_X is not None:
621 dist = dist_X
622 else:
623 dist = dist_Z
624
625 res_tuple = (dist, dist_list_X, dist_list_Z, cws_X, cws_Z) if do_cws else (dist, dist_list_X, dist_list_Z)
626 if _use_distance_cache and cache_key is not None:
627 _distance_cache[cache_key] = res_tuple
628 return res_tuple
629
630 finally:

659) -> Tuple[Any, ...]:
660 """
661 Computes minimum graph/hypergraph distance of a Stim Detector Error Model (DEM).
662
663 Args:
664 dem: stim.DetectorErrorModel object or path to .dem file.
665 circuit: stim.Circuit object (converted to DEM).
666 dist_m4ri: Path to dist_m4ri executable (optional).
667 method: Solver method (1=RW, 2=CC, 3=Bracketing default).
668 threads: Number of worker threads.
669 timeout: Execution timeout in seconds.
670 num_steps: Maximum RW steps.
671 d_exp: Expected distance estimate.
672 d_min: Minimum target distance.
673 wmax: Maximum weight to search in CC.
674 dW: Extra weight window above dmin to collect codewords.
675 maxC: Maximum number of codewords to collect.
676 pmin: Probability cutoff for error mechanisms in DEM.
677 noscan: Skip CC scan loop if 1.
678 do_cws: Whether to return extracted error mechanisms / codewords.
679 solver: "dist_m4ri" or "codedistance".
680 codedistance_method: Method if using codedistance library.
681 codedistance_params: Extra parameters for codedistance library.
682 seed: Random seed.
683 debug: Debug level flags.
684
685 Returns:
686 tuple (dist, dist_list, cws) if do_cws else (dist, dist_list)
687 """
688 if dem is None and circuit is not None:
689 if hasattr(circuit, 'detector_error_model'):
690 dem = circuit.detector_error_model(decompose_errors=True)
691 else:
692 raise ValueError("Provided circuit object does not have detector_error_model() method.")
693
694 if dem is None:
695 raise ValueError("Either 'dem' or 'circuit' must be provided.")
696
697 if solver == "codedistance":
698 if not _HAS_CODEDISTANCE:
699 raise ImportError("codedistance library requested but not installed.")

701 params.setdefault("filterCircuit", False)
702 if num_steps is not None and "iterCount" not in params:
703 params["iterCount"] = num_steps
704

706 res = codedistance.circuitDistance(
707 circuit, method=codedistance_method, params=params,
708 seed=seed if seed != 0 else None
709 )
710 else:
711 H, L, priors = codedistance.StimDEM2HL(dem)
712 if "priors" not in params and len(priors) > 0:
713 params["priors"] = priors
714 res = codedistance.codeDistance(
715 H, L, tB=1, method=codedistance_method, params=params,
716 seed=seed if seed != 0 else None
717 )
718 d = res.get("d", -1)
719 dist_list = [d] if d > 0 else []
720 if do_cws:
721 cws = [list(np.where(res["L"])[0])] if (res.get("L") is not None and np.sum(res["L"]) > 0) else []
722 return d, dist_list, cws
723 return d, dist_list

760 maxC=maxC,

774 return dist, dist_list, cws
