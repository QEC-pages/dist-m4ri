167def format_bounds_list(dmin: int, dmax: int, num_rw: int) -> List[int]:
168 """
169 Returns [dmin, dmax, num_rw] according to:
170 - [d, d, 0] if known exactly
171 - [dmin, 0, 0] if there is no upper bound (dmax == 0)
172 - [0, dmax, num_rw] if there is no lower bound (dmin <= 1)
173 - [dmin, dmax, num_rw] otherwise
174 """
175 eff_dmin = dmin if dmin > 1 else 0
176 eff_dmax = dmax if dmax > 0 else 0
177 eff_rw = num_rw if num_rw > 0 else 0
178 if eff_dmin > 0 and eff_dmin == eff_dmax:
179 return [eff_dmin, eff_dmax, 0]
180 elif eff_dmax == 0:
181 return [eff_dmin, 0, 0]
182 elif eff_dmin == 0:
183 return [0, eff_dmax, eff_rw]
184 else:
185 return [eff_dmin, eff_dmax, eff_rw]

189 """Formats a bounds list [dmin, dmax, num_rw] as a string, stating '(exact)' if bounds coincide."""

196def get_cached_distance(
197 H: Optional[Any] = None,

199 Hz: Optional[Any] = None,
200 Lx: Optional[Any] = None,
201 Lz: Optional[Any] = None,

208 Retrieves the cached distance entry (including bounds and cumulative rw_steps)

220 key = f"classical:{get_sparse_array_state(H)}"

222 if entry:

225 return entry
226 elif Hx is not None or Hz is not None:
227 hx_st = get_sparse_array_state(Hx) if Hx is not None else "none"
228 hz_st = get_sparse_array_state(Hz) if Hz is not None else "none"
229 key = f"css:X={hx_st}:Z={hz_st}"

231 lx_st = get_sparse_array_state(Lx) if Lx is not None else "none"
232 lz_st = get_sparse_array_state(Lz) if Lz is not None else "none"
233 key = f"{key}:Lx={lx_st}:Lz={lz_st}"

235 if entry:
236 entry = dict(entry)
237 if "dmin_X" in entry:

240 entry["dZ"] = format_bounds_list(entry.get("dmin_Z", 0), entry.get("dmax_Z", 0), entry.get("rw_steps_Z", 0))
241 return entry
242 elif dem is not None or circuit is not None:
243 if dem is None and circuit is not None:

245 obj = circuit.detector_error_model(decompose_errors=True)
246 else:
247 obj = circuit
248 else:
249 obj = dem

266def get_sparse_array_state(A) -> str:
267 """Returns a deterministic string representation for JSON-compatible cache keys."""
268 if A is None:
269 return "none"
270 if isinstance(A, (str, Path)):
271 path_str = str(Path(A).resolve())
272 if os.path.isfile(path_str):
273 try:
274 with open(path_str, "rb") as f:
275 content_h = hashlib.sha256(f.read()).hexdigest()
276 return f"file:{path_str}:{content_h}"
277 except Exception:
278 return f"file:{path_str}"
279 return path_str
280 if hasattr(A, 'shape') and hasattr(A, 'dtype') and hasattr(A, 'tobytes'):
281 h = hashlib.sha256(A.tobytes()).hexdigest()
282 dtype_str = getattr(A.dtype, 'str', str(A.dtype))
283 return f"ndarray:{A.shape}:{dtype_str}:{h}"
284 if hasattr(A, 'tocsr'):
285 csr = A.tocsr()
286 h = hashlib.sha256(csr.data.tobytes() + csr.indices.tobytes() + csr.indptr.tobytes()).hexdigest()
287 return f"csr:{csr.shape}:{h}"
288 if hasattr(A, 'tobytes'):
289 h = hashlib.sha256(A.tobytes()).hexdigest()
290 return f"bytes:{h}"
291 h = hashlib.sha256(str(A).encode('utf-8')).hexdigest()
292 return f"str_sha256:{h}"

303def read_sparse_vectors(filepath: str) -> List[List[int]]:
304 """
305 Reads a list of sparse vectors from a text file in NZLIST format,
306 converting from 1-based indexing (in the file) to 0-based indexing (in Python).
307
308 Args:
309 filepath (str): The path to the text file.
310
311 Returns:
312 list of list of int: A list where each element is a 0-based sparse vector.
313 """
314 sparse_vectors = []
315 if not os.path.exists(filepath) or os.path.getsize(filepath) == 0:
316 return sparse_vectors
317
318 with open(filepath, 'r') as f:
319 first_line = f.readline().strip()
320 if not first_line:
321 return sparse_vectors
322 if first_line != '%% NZLIST':
323 raise ValueError(f"Invalid file format in {filepath}: Missing '%% NZLIST' header.")
324
325 for line_num, line in enumerate(f, start=2):
326 line = line.strip()
327 if not line or line.startswith('%'):
328 continue
329 try:
330 parts = list(map(int, line.split()))
331 except ValueError:
332 raise ValueError(f"Non-integer data found on line {line_num}: {line}")
333
334 stated_length = parts[0]
335 vector_elements = [x - 1 for x in parts[1:]]
336 if len(vector_elements) != stated_length:
337 raise ValueError(
338 f"Length mismatch on line {line_num}. "
339 f"Expected {stated_length} elements, but found {len(vector_elements)}."
340 )
341 sparse_vectors.append(vector_elements)
342
343 return sparse_vectors

376 """
377 Parses the standard output of dist_m4ri.
378 Expected format on stdout: "dmin dmax rw_steps", "dmin dmax", or a single integer.
379
380 Returns:
381 tuple (dmin, dmax, rw_steps)
382 """
383 lines = stdout.strip().split('\n')
384 for line in reversed(lines):
385 line = line.strip()

387 continue
388 parts = line.split()

391 return int(parts[0]), int(parts[1]), int(parts[2])

396 return int(parts[0]), int(parts[1]), 0

401 val = int(parts[0])
402 return val, val, 0
403 except ValueError:
404 continue
405
406 raise RuntimeError(f"Could not parse dist_m4ri output: {stdout}")

409class DistanceResult:
410 """
411 Structured result for code distance calculations containing:
412 - dmin: lower bound on distance
413 - dmax: upper bound on distance (minimum non-trivial codeword weight found)
414 - rw_steps: cumulative number of completed random window information sets
415 - cws: discovered codewords (if requested)
416 """
417 def __init__(
418 self,
419 dmin: int,
420 dmax: int,
421 rw_steps: int = 0,
422 cws: Optional[List[List[int]]] = None,
423 cws_X: Optional[List[List[int]]] = None,
424 cws_Z: Optional[List[List[int]]] = None,
425 dmin_X: Optional[int] = None,
426 dmax_X: Optional[int] = None,
427 rw_steps_X: Optional[int] = None,
428 dmin_Z: Optional[int] = None,
429 dmax_Z: Optional[int] = None,
430 rw_steps_Z: Optional[int] = None,
431 ):
432 self.dmin = dmin
433 self.dmax = dmax
434 self.rw_steps = rw_steps
435 self.cws = cws
436 self.cws_X = cws_X
437 self.cws_Z = cws_Z
438 self.dmin_X = dmin_X
439 self.dmax_X = dmax_X
440 self.rw_steps_X = rw_steps_X
441 self.dmin_Z = dmin_Z
442 self.dmax_Z = dmax_Z
443 self.rw_steps_Z = rw_steps_Z
444
445 @property
446 def is_exact(self) -> bool:
447 return self.dmin > 0 and self.dmin == self.dmax
448
449 @property
450 def dist(self) -> int:
451 return self.dmin if self.is_exact else (self.dmax if self.dmax > 0 else self.dmin)
452
453 def __int__(self) -> int:
454 return self.distdist
455
456 def __index__(self) -> int:
457 return self.distdist
458
459 def __eq__(self, other: Any) -> bool:
460 if isinstance(other, DistanceResult):
461 return (self.dmin, self.dmax, self.rw_steps) == (other.dmin, other.dmax, other.rw_steps)
462 if isinstance(other, (tuple, list)):
463 return tuple(self) == tuple(other)
464 if isinstance(other, (int, np.integer)):
465 return self.distdist == other
466 return False
467
468 def __iter__(self):
469 if self.cws_X is not None or self.cws_Z is not None:
470 return iter((self.dmin, self.dmax, self.rw_steps, self.cws_X or [], self.cws_Z or []))
471 if self.cws is not None:
472 return iter((self.dmin, self.dmax, self.rw_steps, self.cws))
473 return iter((self.dmin, self.dmax, self.rw_steps))
474
475 def __getitem__(self, index: int):
476 return tuple(self)[index]
477
478 def __len__(self) -> int:
479 return len(tuple(self))
480
481 def __str__(self) -> str:
482 if self.is_exact:
483 return f"{self.dmin} {self.dmax} {self.rw_steps} (exact)"
484 return f"{self.dmin} {self.dmax} {self.rw_steps}"
485
486 def __repr__(self) -> str:
487 if self.is_exact:
488 return f"DistanceResult({self.dmin} {self.dmax} {self.rw_steps} (exact))"
489 return f"DistanceResult(dmin={self.dmin}, dmax={self.dmax}, rw_steps={self.rw_steps})"

492def run_dist_m4ri(
493 dist_m4ri_path: Optional[str] = None,
494 method: int = 3,
495 finH: Optional[str] = None,
496 finG: Optional[str] = None,
497 finL: Optional[str] = None,
498 fdem: Optional[str] = None,
499 dmin: int = 0,
500 dmax: int = 0,
501 wmax: int = 0,
502 wmin: int = 1,
503 dexp: int = 0,
504 dest: int = 0,
505 steps: Optional[int] = None,
506 threads: Optional[int] = None,
507 timeout: float = 60.0,
508 smax: Optional[int] = None,
509 noscan: int = 0,
510 classical: int = -1,
511 dW: int = -1,
512 maxC: int = 0,
513 pmin: float = 0.0,
514 outC: Optional[str] = None,
515 seed: int = 0,
516 debug: int = 0,
517 stop_event: Optional[threading.Event] = None
518) -> Tuple[int, int, int]:
519 """
520 Low-level invocation of the multithreaded dist_m4ri binary.
521
522 Returns:
523 tuple (dmin, dmax, rw_steps)
524 """
525 exec_path = find_dist_m4ri_binary(dist_m4ri_path)
526
527 if method == 2 and wmax <= 0:
528 if dmax > 0:
529 wmax = dmax
530 elif timeout <= 0.0:
531 raise ValueError("either parameter wmax>0 or timeout>0 should be specified for CC method=2.")
532
533 cmd = [exec_path, f"debug={debug}", f"method={method}"]
534
535 if finH: cmd.append(f"finH={finH}")
536 if finG: cmd.append(f"finG={finG}")
537 if finL: cmd.append(f"finL={finL}")
538 if fdem: cmd.append(f"fdem={fdem}")
539 if dmin > 0: cmd.append(f"dmin={dmin}")
540 if dmax > 0: cmd.append(f"dmax={dmax}")
541 if wmax > 0: cmd.append(f"wmax={wmax}")
542 if wmin > 1: cmd.append(f"wmin={wmin}")
543 if dexp > 0: cmd.append(f"dexp={dexp}")
544 elif dest > 0: cmd.append(f"dest={dest}")
545 if steps is not None and steps > 0: cmd.append(f"steps={steps}")
546 if threads is not None and threads > 0: cmd.append(f"threads={threads}")
547 if timeout > 0: cmd.append(f"timeout={timeout}")
548 if smax is not None: cmd.append(f"smax={smax}")
549 if noscan: cmd.append(f"noscan={noscan}")
550 if classical >= 0: cmd.append(f"classical={classical}")
551 if dW >= 0: cmd.append(f"dW={dW}")
552 if maxC > 0: cmd.append(f"maxC={maxC}")
553 if pmin > 0.0: cmd.append(f"pmin={pmin}")
554 if outC: cmd.append(f"outC={outC}")
555 if seed != 0: cmd.append(f"seed={seed}")
556
557 if debug & 2:
558 print(f"[dist_m4ri] Running: {' '.join(cmd)}")
559
560 proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
561
562 if stop_event is not None:
563 while proc.poll() is None:
564 if stop_event.is_set():
565 proc.terminate()
566 try:
567 proc.wait(timeout=1.0)
568 except subprocess.TimeoutExpired:
569 proc.kill()
570 raise RuntimeError("dist_m4ri execution cancelled by stop_event")
571 time.sleep(0.05)
572 stdout, stderr = proc.communicate()
573 else:
574 stdout, stderr = proc.communicate()
575
576 if proc.returncode != 0:
577 raise RuntimeError(f"dist_m4ri failed with exit code {proc.returncode}:\n{stderr}")
578
579 return parse_dist_m4ri_output(stdout)
