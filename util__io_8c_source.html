406static dem_program_t *read_dem_to_program(const char *fnam) {
407 FILE *f = fopen(fnam, "r");
408 if (f == NULL) {
409 printf("FILE I/O ERROR: %s\n", strerror(errno));
410 ERROR("can't open the (DEM) file %s for reading\n", fnam);
411 }
412
413 dem_program_t *prog = malloc(sizeof(dem_program_t));
414 prog->size = 0;
415 prog->capacity = 100;
416 prog->lines = malloc(prog->capacity * sizeof(char *));
417
418 char *buf = NULL;
419 size_t bufsiz = 0;
420 ssize_t linelen;
421
422 while ((linelen = getline(&buf, &bufsiz, f)) >= 0) {
423 if (linelen > 0 && buf[linelen - 1] == '\n') {
424 buf[linelen - 1] = '\0';
425 }
426 if (prog->size >= prog->capacity) {
427 prog->capacity *= 2;
428 prog->lines = realloc(prog->lines, prog->capacity * sizeof(char *));
429 }
430 prog->lines[prog->size++] = strdup(buf);
431 }
432 if (buf) free(buf);
433 fclose(f);
434 return prog;
435}

438 for (int i = 0; i < prog->size; i++) {
439 free(prog->lines[i]);
440 }
441 free(prog->lines);
442 free(prog);
443}

445static void parse_instructions(dem_program_t *prog, int *p_line_idx,
446 int *p_iD, int_pair **p_inH, int *p_maxH, int *p_r,
447 int *p_iL, int_pair **p_inL, int *p_maxL, int *p_k,
448 int *p_n, double pmin, int *p_detector_shift, int debug) {
449 while (*p_line_idx < prog->size) {
450 char *line = prog->lines[*p_line_idx];
451 (*p_line_idx)++;
452
453 char *c = line;
454 while (isspace(*c)) c++;
455
456 if (*c == '\0' || *c == '#') continue;
457
458 if (*c == '}') {
459 return;
460 }
461
462 int num = 0;
463 int val = 0;
464 double prob = 0.0;
465
466 // Parse repeat
467 if (sscanf(c, "repeat %d { %n", &val, &num) == 1) {
468 int start_idx = *p_line_idx;
469 int temp_idx = start_idx;
470 for (int r = 0; r < val; r++) {
471 temp_idx = start_idx;
472 parse_instructions(prog, &temp_idx,
473 p_iD, p_inH, p_maxH, p_r,
474 p_iL, p_inL, p_maxL, p_k,
475 p_n, pmin, p_detector_shift, debug);
476 }
477 if (val > 0) {
478 *p_line_idx = temp_idx;
479 } else {
480 int depth = 1;
481 while (*p_line_idx < prog->size && depth > 0) {
482 char *s = prog->lines[*p_line_idx];
483 (*p_line_idx)++;
484 while (isspace(*s)) s++;
485 if (strncmp(s, "repeat", 6) == 0 && strchr(s, '{')) depth++;
486 if (*s == '}') depth--;
487 }
488 }
489 continue;
490 }
491
492 // Parse shift_detectors
493 int shift_val = 0;
494 if (sscanf(c, "shift_detectors ( %*[^)] ) %d %n", &shift_val, &num) == 1) {
495 *p_detector_shift += shift_val;
496 continue;
497 } else if (sscanf(c, "shift_detectors %d %n", &shift_val, &num) == 1) {
498 *p_detector_shift += shift_val;
499 continue;
500 }
501
502 // Parse error
503 if (sscanf(c, "error( %lg ) %n", &prob, &num) == 1) {
504 if ((prob <= 0) || (prob >= 1))
505 ERROR("probability should be in (0,1) exclusive p=%g\n"
506 "line %d: '%s'\n", prob, *p_line_idx, line);
507 c += num;
508
509 if (prob < pmin) {
510 continue;
511 }
512
513 while (1) {
514 while (isspace(c[0])) c++;
515 if (c[0] == '\0' || c[0] == '#' || c[0] == '\n') break;
516
517 num = 0;
518 if (sscanf(c, "D%d%n", &val, &num) == 1) {
519 c += num;
520 assert(val >= 0);
521 int shifted_val = val + *p_detector_shift;
522 if (shifted_val >= *p_r)
523 *p_r = shifted_val + 1;
524 if (*p_iD >= *p_maxH) {
525 *p_maxH = 2 * (*p_maxH);
526 *p_inH = realloc(*p_inH, (*p_maxH) * sizeof(**p_inH));
527 }
528 (*p_inH)[*p_iD].a = shifted_val;
529 (*p_inH)[*p_iD].b = *p_n;
530 (*p_iD)++;
531 } else if (sscanf(c, "L%d%n", &val, &num) == 1) {
532 c += num;
533 assert(val >= 0);
534 if (val >= *p_k)
535 *p_k = val + 1;
536 if (*p_iL >= *p_maxL) {
537 *p_maxL = 2 * (*p_maxL);
538 *p_inL = realloc(*p_inL, (*p_maxL) * sizeof(**p_inL));
539 }
540 (*p_inL)[*p_iL].a = val;
541 (*p_inL)[*p_iL].b = *p_n;
542 (*p_iL)++;
543 } else if (c[0] == '^') {
544 c++;
545 } else {
546 ERROR("unrecognized entry %s in error line %d: '%s'\n", c, *p_line_idx, line);
547 }
548 }
549 (*p_n)++;
550 continue;
551 }
552
553 if (strncmp(c, "detector", 8) == 0) {
554 continue;
555 }
556
557 if (strncmp(c, "logical_observable", 18) == 0) {
558 continue;
559 }
560
561 ERROR("unrecognized DEM entry in line %d: '%s'\n", *p_line_idx, line);
562 }
563}

565void read_dem_file(char *fnam, csr_t **p_spaH, csr_t **p_spaL, double pmin, int debug){
566 dem_program_t *prog = read_dem_to_program(fnam);
567
568 int maxH=100, maxL=100;
569 int_pair * inH = malloc(maxH*sizeof(int_pair));
570 int_pair * inL = malloc(maxL*sizeof(int_pair));
571 if ((!inH)||(!inL))
572 ERROR("memory allocation failed\n");
573
574 int r=-1, k=-1, n=0;
575 int iD=0, iL=0;
576 int detector_shift = 0;
577 int line_idx = 0;
578
579 parse_instructions(prog, &line_idx,
580 &iD, &inH, &maxH, &r,
581 &iL, &inL, &maxL, &k,
582 &n, pmin, &detector_shift, debug);
583
584 if (line_idx < prog->size) {
585 ERROR("Unmatched '}' in DEM file %s at line %d\n", fnam, line_idx);
586 }
587
588 if(debug & 1)
589 printf("# read DEM %s: rows_H=%d rows_L=%d cols=%d; nz_H=%d nz_L=%d\n",fnam,r,k,n,iD,iL);
590 if((r<=0)||(k<=0)||(n<=0))
591 ERROR("invalid DEM file %s: rows_H=%d rows_L=%d cols=%d; nz_H=%d nz_L=%d\n",
592 fnam,r,k,n,iD,iL);
593
594 *p_spaH = csr_from_pairs(*p_spaH, iD, inH, r, n);
595 *p_spaL = csr_from_pairs(*p_spaL, iL, inL, k, n);
596
597 free(inH);
598 free(inL);
599 free_dem_program(prog);
600}

684 ERROR("expected an integer i=%d of %d",i,w);

699cw_vec_t * codeword_add_maybe(params_t * const p, const int arr[], int weight) {
700 if (p->maxC && p->num_cws >= p->maxC) {
701 return p->codewords;
702 }
703 // Check if weight is within the current limit: min_w + dW
704 if (p->min_w != INT_MAX && p->dW >= 0 && weight > p->min_w + p->dW) {
705 return p->codewords;
706 }
707
708 const size_t keylen = weight * sizeof(int);
709 cw_vec_t *pvec = NULL;
710 HASH_FIND(hh, p->codewords, arr, keylen, pvec);
711 if (!pvec) {
712 cw_vec_t *entry = malloc(sizeof(cw_vec_t) + keylen);
713 if (!entry) ERROR("memory allocation");
714 entry->weight = weight;
715 entry->cnt = 1;
716 for (int i = 0; i < weight; i++) {
717 entry->arr[i] = arr[i];
718 }
719 HASH_ADD(hh, p->codewords, arr, keylen, entry);
720 p->num_cws++;
721
722 // Update min_w
723 if (weight < p->min_w) {
724 p->min_w = weight;
725 if (p->dW >= 0) {
726 cw_vec_t *cw, *tmp;
727 HASH_ITER(hh, p->codewords, cw, tmp) {
728 if (cw->weight > p->min_w + p->dW) {
729 HASH_DEL(p->codewords, cw);
730 free(cw);
731 p->num_cws--;
732 }
733 }
734 }
735 }
736 } else {
737 pvec->cnt++;
738 }
739 return p->codewords;
740}

742long long int nzlist_read(const char fnam[], params_t *p){
743 long long int count = 0, lineno;
744 long long int skipped_invalid = 0;
745 assert(fnam);
746 FILE * f=nzlist_r_open(fnam, &lineno);
747 if(!f){
748 if ((p->outC ==NULL) || (strcmp(fnam,p->outC)!=0)){
749 printf("codeword input file I/O ERROR: %s, outC=%s\n", strerror(errno),p->outC);
750 ERROR("can't open file %s for reading",fnam);
751 }
752 else
753 return 0;
754 }
755 cw_vec_t *entry=NULL;
756 while((entry=nzlist_r_one(f,NULL, fnam, &lineno))){
757 if((p->maxC) && (p->num_cws >= p->maxC)) {
758 free(entry);
759 break;
760 }
761 int valid = 1;
762 if (p->spaH) {
763 if (sparse_syndrome_non_zero(p->spaH, entry->weight, entry->arr)) {
764 valid = 0;
765 }
766 }
767 if (valid && p->spaL) {
768 if (!sparse_syndrome_non_zero(p->spaL, entry->weight, entry->arr)) {
769 valid = 0;
770 }
771 }
772 if (!valid) {
773 skipped_invalid++;
774 free(entry);
775 continue;
776 }
777 if((p->wmax==0) ||((p->wmax) && (entry->weight <= p->wmax))){
778 long long int old_num = p->num_cws;
779 p->codewords = codeword_add_maybe(p, entry->arr, entry->weight);
780 if (p->num_cws > old_num) {
781 count++;
782 }
783 }
784 free(entry);
785 }
786 fclose(f);
787 if (skipped_invalid > 0) {
788 printf("# Warning: skipped %lld invalid codewords (not orthogonal to H or orthogonal to L)\n", skipped_invalid);
789 }
790 if(p->debug&1)
791 printf("# read %lld codewords from %s, total %lld\n",count, fnam, p->num_cws);
792 return count;
793}
