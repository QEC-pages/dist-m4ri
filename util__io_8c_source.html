520/* The edit (Levenshtein) distance of the strings a and b (each of at most HELP_NAME_LEN - 1 characters). */
521static int help_edit_distance(const char * const a, const char * const b) {
522 const size_t la = strlen(a), lb = strlen(b);
523 int d[HELP_NAME_LEN]; /* one row of the table: d[j] for b[0..j-1] */
524 for (size_t j = 0; j <= lb; j++)
525 d[j] = (int)j;
526 for (size_t i = 1; i <= la; i++) {
527 int diag = d[0];
528 d[0] = (int)i;
529 for (size_t j = 1; j <= lb; j++) {
530 const int up = d[j];
531 int best = diag + (a[i - 1] != b[j - 1]);
532 if (up + 1 < best)
533 best = up + 1;
534 if (d[j - 1] + 1 < best)
535 best = d[j - 1] + 1;
536 d[j] = best;
537 diag = up;
538 }
539 }
540 return d[lb];
541}

543/* Writes to stderr that the topic s (n characters, normalized: key) has no help entry, with up to 5 of the closest
544 * names (by the edit distance of the normalized names: at most (len + 2) / 3, and at least 1, for a key of length
545 * len), the closest first. */
546static void help_no_entry(const char * const prog, const char * const s, const int n, const char * const key,
547 const help_entry_t * const ent, const int ne) {

551 for (int i = 0; i < ne; i++)
552 for (int j = 0; j < ent[i].nnames; j++) {
553 const int d = help_edit_distance(key, ent[i].key[j]);
554 int pos = nb;
555 if (d > maxd)
556 continue;

558 pos--;
559 if (pos >= 5)

564 best[k] = best[k - 1];

566 }

568 bestd[pos] = d;
569 }

576/* Prints the entries of the full help (MORE_HELP) for the ntopics topics (--help TOPIC, --morehelp TOPIC): each is
577 * one or more names separated by commas, compared without case, '-', '_', and '=VALUE' with the names of the entries
578 * and their aliases (see help_entries()). The entries are printed in the order of the full help, under their section
579 * headers. Returns 0, 1 if some topic has no entry (then the closest names are suggested on stderr), or -1 if there
580 * is no topic (e.g., --help=), and nothing is printed. */
581static int print_help_topics(const char * const prog, const int ntopics, const char * const topics[]) {
582 const int len = snprintf(NULL, 0, MORE_HELP, prog, DIST_M4RI_VERSION, prog);
583 char * const text = malloc((size_t)len + 1);
584 if (!text)
585 ERROR("memory allocation");
586 snprintf(text, (size_t)len + 1, MORE_HELP, prog, DIST_M4RI_VERSION, prog);
587 char **lines;
588 int nlines;
589 help_entry_t *ent;
590 const int ne = help_entries(text, &lines, &nlines, &ent);
591 char * const chosen = calloc((size_t)ne + 1, 1);
592 if (!chosen)
593 ERROR("memory allocation");
594 int status = 0, nkeys = 0;
595 for (int t = 0; t < ntopics; t++)
596 for (const char *s = topics[t]; *s; ) {
597 const char * const comma = strchr(s, ',');
598 int n = comma ? (int)(comma - s) : (int)strlen(s);
599 const char * const next = comma ? comma + 1 : s + n;
600 while ((n > 0) && isspace((unsigned char)*s)) {
601 s++;
602 n--;
603 }
604 while ((n > 0) && isspace((unsigned char)s[n - 1]))
605 n--;
606 char key[HELP_NAME_LEN];
607 help_key(key, s, (size_t)n);
608 if (key[0]) {
609 int found = 0;
610 nkeys++;
611 for (int i = 0; i < ne; i++)
612 for (int j = 0; j < ent[i].nnames; j++)
613 if (strcmp(ent[i].key[j], key) == 0)
614 chosen[i] = found = 1;
615 if (!found) {
616 help_no_entry(prog, s, n, key, ent, ne);
617 status = 1;
618 }
619 }
620 s = next;
621 }
622 int sec = -2, any = 0;
623 for (int i = 0; i < ne; i++) {
624 if (!chosen[i])
625 continue;
626 if (ent[i].sec != sec) { /* the section header, once (a whole section: its first line) */
627 if (any)
628 printf("\n");
629 if ((ent[i].sec >= 0) && !ent[i].is_sec)
630 printf("%s\n", lines[ent[i].sec]);
631 sec = ent[i].sec;
632 }
633 for (int k = ent[i].beg; k < ent[i].end; k++)
634 printf("%s\n", lines[k]);
635 any = 1;
636 }
637 free(chosen);
638 free(ent);
639 free(lines);
640 free(text);
641 return nkeys ? status : -1;
642}

644/* The kind of the help option arg: 2 for --morehelp (also -morehelp, --more-help, -more-help), 1 for --help (also
645 * -help, -h, -?), or 0; *topic is set to the text after '=' (e.g., --help=maxC), or to NULL. */
646static int help_option(const char * const arg, const char ** const topic) {
647 static const char * const opts[] = {"--morehelp", "-morehelp", "--more-help", "-more-help",
648 "--help", "-help", "-h", "-?", NULL};
649 *topic = NULL;
650 for (int k = 0; opts[k]; k++) {
651 const size_t len = strlen(opts[k]);
652 if ((strncmp(arg, opts[k], len) == 0) && ((arg[len] == '\0') || (arg[len] == '='))) {
653 if (arg[len] == '=')
654 *topic = arg + len + 1;
655 return k < 4 ? 2 : 1;
656 }
657 }
658 return 0;
659}

1397/* The `cnt` entries (row, column) of one DEM error line, all in the same column: sort them by row (insertion sort, few
1398 * entries), remove the pairs of equal rows (the sum mod 2), and return the number of entries left. The components of
1399 * a decomposed error (separated by `^`) are summed: a detector or an observable listed twice in the line cancels. */
1400static int dem_line_mod2(int_pair * const e, const int cnt) {
1401 for (int i = 1; i < cnt; i++) {
1402 const int_pair x = e[i];
1403 int j = i - 1;
1404 while (j >= 0 && e[j].a > x.a) {
1405 e[j + 1] = e[j];
1406 j--;
1407 }
1408 e[j + 1] = x;
1409 }
1410 int out = 0;
1411 for (int i = 0; i < cnt;) {
1412 if (i + 1 < cnt && e[i].a == e[i + 1].a) {
1413 i += 2; /* a repeated entry cancels */
1414 } else {
1415 e[out++] = e[i++];
1416 }
1417 }
1418 return out;
1419}

1421static void parse_instructions(dem_program_t *prog, int *p_line_idx,
1422 int *p_iD, int_pair **p_inH, int *p_maxH, int *p_r,
1423 int *p_iL, int_pair **p_inL, int *p_maxL, int *p_k,
1424 int *p_n, double pmin, int *p_detector_shift, int debug) {
1425 while (*p_line_idx < prog->size) {
1426 char *line = prog->lines[*p_line_idx];
1427 (*p_line_idx)++;
1428
1429 char *c = line;
1430 while (isspace(*c)) c++;
1431
1432 if (*c == '\0' || *c == '#') continue;
1433
1434 if (*c == '}') {
1435 return;
1436 }
1437
1438 int num = 0;
1439 int val = 0;
1440 double prob = 0.0;
1441
1442 // Parse repeat
1443 if (sscanf(c, "repeat %d { %n", &val, &num) == 1) {
1444 int start_idx = *p_line_idx;
1445 int temp_idx = start_idx;
1446 for (int r = 0; r < val; r++) {
1447 temp_idx = start_idx;
1448 parse_instructions(prog, &temp_idx,
1449 p_iD, p_inH, p_maxH, p_r,
1450 p_iL, p_inL, p_maxL, p_k,
1451 p_n, pmin, p_detector_shift, debug);
1452 }
1453 if (val > 0) {
1454 *p_line_idx = temp_idx;
1455 } else {
1456 int depth = 1;
1457 while (*p_line_idx < prog->size && depth > 0) {
1458 char *s = prog->lines[*p_line_idx];
1459 (*p_line_idx)++;
1460 while (isspace(*s)) s++;
1461 if (strncmp(s, "repeat", 6) == 0 && strchr(s, '{')) depth++;
1462 if (*s == '}') depth--;
1463 }
1464 }
1465 continue;
1466 }
1467
1468 // Parse shift_detectors
1469 int shift_val = 0;
1470 if (sscanf(c, "shift_detectors ( %*[^)] ) %d %n", &shift_val, &num) == 1) {
1471 *p_detector_shift += shift_val;
1472 continue;
1473 } else if (sscanf(c, "shift_detectors %d %n", &shift_val, &num) == 1) {
1474 *p_detector_shift += shift_val;
1475 continue;
1476 }
1477
1478 // Parse error
1479 if (sscanf(c, "error( %lg ) %n", &prob, &num) == 1) {
1480 if ((prob <= 0) || (prob >= 1))
1481 ERROR("probability should be in (0,1) exclusive p=%g\n"
1482 "line %d: '%s'\n", prob, *p_line_idx, line);
1483 c += num;
1484
1485 if (prob < pmin) {
1486 continue;
1487 }
1488 const int iD0 = *p_iD, iL0 = *p_iL; /* the first entries of this error line */
1489
1490 while (1) {
1491 while (isspace(c[0])) c++;
1492 if (c[0] == '\0' || c[0] == '#' || c[0] == '\n') break;
1493
1494 num = 0;
1495 if (sscanf(c, "D%d%n", &val, &num) == 1) {
1496 c += num;
1497 assert(val >= 0);
1498 int shifted_val = val + *p_detector_shift;
1499 if (shifted_val >= *p_r)
1500 *p_r = shifted_val + 1;
1501 if (*p_iD >= *p_maxH) {
1502 *p_maxH = 2 * (*p_maxH);
1503 *p_inH = realloc(*p_inH, (*p_maxH) * sizeof(**p_inH));
1504 }
1505 (*p_inH)[*p_iD].a = shifted_val;
1506 (*p_inH)[*p_iD].b = *p_n;
1507 (*p_iD)++;
1508 } else if (sscanf(c, "L%d%n", &val, &num) == 1) {
1509 c += num;
1510 assert(val >= 0);
1511 if (val >= *p_k)
1512 *p_k = val + 1;
1513 if (*p_iL >= *p_maxL) {
1514 *p_maxL = 2 * (*p_maxL);
1515 *p_inL = realloc(*p_inL, (*p_maxL) * sizeof(**p_inL));
1516 }
1517 (*p_inL)[*p_iL].a = val;
1518 (*p_inL)[*p_iL].b = *p_n;
1519 (*p_iL)++;
1520 } else if (c[0] == '^') {
1521 c++;
1522 } else {
1523 ERROR("unrecognized entry %s in error line %d: '%s'\n", c, *p_line_idx, line);
1524 }
1525 }
1526 /* the components separated by `^` are summed mod 2 */
1527 *p_iD = iD0 + dem_line_mod2(*p_inH + iD0, *p_iD - iD0);
1528 *p_iL = iL0 + dem_line_mod2(*p_inL + iL0, *p_iL - iL0);
1529 (*p_n)++;
1530 continue;
1531 }
1532
1533 if (strncmp(c, "detector", 8) == 0) {
1534 continue;
1535 }
1536
1537 if (strncmp(c, "logical_observable", 18) == 0) {
1538 continue;
1539 }
1540
1541 ERROR("unrecognized DEM entry in line %d: '%s'\n", *p_line_idx, line);
1542 }
1543}

1600/* Are the columns a and b equal in H and L (rows a and b of their transposes HT and LT, sorted)? */
1601static int dem_cols_equal(const csr_t * const HT, const csr_t * const LT, const int a, const int b) {
1602 const int wh = HT->p[a + 1] - HT->p[a], wl = LT->p[a + 1] - LT->p[a];
1603 if ((wh != HT->p[b + 1] - HT->p[b]) || (wl != LT->p[b + 1] - LT->p[b])) return 0;
1604 return (memcmp(HT->i + HT->p[a], HT->i + HT->p[b], (size_t)wh * sizeof(int)) == 0) &&
1605 (memcmp(LT->i + LT->p[a], LT->i + LT->p[b], (size_t)wl * sizeof(int)) == 0);
1606}

1608/* The rows of the CSR matrix M restricted to the columns j with keep[j] >= 0, renumbered to keep[j] (increasing in j,
1609 * so that the rows stay sorted) */
1610static csr_t * csr_keep_columns(const csr_t * const M, const int * const keep, const int ncols) {
1611 int nz = 0;
1612 for (int t = 0; t < M->p[M->rows]; t++) nz += (keep[M->i[t]] >= 0);
1613 csr_t * const R = csr_init(NULL, M->rows, ncols, nz > 0 ? nz : 1);
1614 int t2 = 0;
1615 for (int r = 0; r < M->rows; r++) {
1616 R->p[r] = t2;
1617 for (int t = M->p[r]; t < M->p[r + 1]; t++)
1618 if (keep[M->i[t]] >= 0) R->i[t2++] = keep[M->i[t]];
1619 }
1620 R->p[M->rows] = t2;
1621 R->nz = -1; /* compressed form */
1622 return R;
1623}

1625/* Merge the equal columns of the DEM matrices H and L (error mechanisms with the same detectors and observables, e.g.,
1626 * from stim's DEM of a circuit with a REPEAT block): a codeword with two equal columns contains a lighter one (without
1627 * both), so that the distance does not change, only the number of codewords. The first column of each class
1628 * represents it, and the classes keep the order of their first columns, so that a sorted support stays sorted when
1629 * mapped back to the original columns (p->col_orig; p->col_class: original column -> class). Nothing changes if all
1630 * columns are distinct. */
1631static void dem_merge_columns(params_t * const p) {
1632 const int n = p->spaH->cols;
1633 if (n <= 1) return;
1634 csr_t * const HT = csr_transpose(NULL, p->spaH);
1635 csr_t * const LT = csr_transpose(NULL, p->spaL);
1636 dem_col_hash_t * const ch = malloc((size_t)n * sizeof(dem_col_hash_t));
1637 int * const rep = malloc((size_t)n * sizeof(int)); /* the first column equal to column j */
1638 if (!ch || !rep) ERROR("memory allocation");
1639 for (int j = 0; j < n; j++) { /* (XOR of row hashes: equal columns have equal hashes) */
1640 uint64_t h = 0;
1641 for (int t = HT->p[j]; t < HT->p[j + 1]; t++) h ^= dem_hash64(2 * (uint64_t)HT->i[t] + 1);
1642 for (int t = LT->p[j]; t < LT->p[j + 1]; t++) h ^= dem_hash64(2 * (uint64_t)LT->i[t] + 2);
1643 ch[j].h = h;
1644 ch[j].j = j;
1645 }
1646 qsort(ch, n, sizeof(dem_col_hash_t), dem_col_hash_cmp);
1647 int nclass = 0;
1648 for (int a = 0; a < n;) { /* a run of equal hashes, in increasing column order */
1649 int b = a + 1;
1650 while (b < n && ch[b].h == ch[a].h) b++;
1651 for (int s = a; s < b; s++) {
1652 const int j = ch[s].j;
1653 rep[j] = j;
1654 for (int t = a; t < s; t++) { /* an earlier column of the run (a first one) equal to j */
1655 const int i = ch[t].j;
1656 if (rep[i] == i && dem_cols_equal(HT, LT, i, j)) {
1657 rep[j] = i;
1658 break;
1659 }
1660 }
1661 if (rep[j] == j) nclass++;
1662 }
1663 a = b;
1664 }
1665 free(ch);
1666 csr_free(HT);
1667 csr_free(LT);
1668 if (nclass == n) { /* all columns distinct */
1669 free(rep);
1670 return;
1671 }
1672 int * const col_class = malloc((size_t)n * sizeof(int));
1673 int * const col_orig = malloc((size_t)nclass * sizeof(int));
1674 int * const keep = malloc((size_t)n * sizeof(int));
1675 if (!col_class || !col_orig || !keep) ERROR("memory allocation");
1676 for (int j = 0, c = 0; j < n; j++) {
1677 if (rep[j] == j) {
1678 col_orig[c] = j;
1679 col_class[j] = keep[j] = c++;
1680 } else {
1681 col_class[j] = col_class[rep[j]]; /* (rep[j] < j) */
1682 keep[j] = -1;
1683 }
1684 }
1685 csr_t * const H2 = csr_keep_columns(p->spaH, keep, nclass);
1686 csr_t * const L2 = csr_keep_columns(p->spaL, keep, nclass);
1687 csr_free(p->spaH);
1688 csr_free(p->spaL);
1689 p->spaH = H2;
1690 p->spaL = L2;
1691 p->nvar_orig = n;
1692 p->nvar_merged = nclass;
1693 p->col_orig = col_orig;
1694 p->col_class = col_class;
1695 free(keep);
1696 free(rep);
1697}

1699/* Are rows a and b of the CSR matrix M (sorted) equal? */
1700static int csr_rows_equal(const csr_t * const M, const int a, const int b) {
1701 const int w = M->p[a + 1] - M->p[a];
1702 return (w == M->p[b + 1] - M->p[b]) && (memcmp(M->i + M->p[a], M->i + M->p[b], (size_t)w * sizeof(int)) == 0);
1703}

1705/* Are the kept detectors (inM[r] == 0) of column j the detectors of column t (rows of HT, sorted; t has no dropped
1706 * detector)? */
1707static int dem_kept_equal(const csr_t * const HT, const char * const inM, const int j, const int t) {
1708 int s = HT->p[t];
1709 for (int u = HT->p[j]; u < HT->p[j + 1]; u++) {
1710 if (inM[HT->i[u]]) continue;
1711 if (s >= HT->p[t + 1] || HT->i[s] != HT->i[u]) return 0;
1712 s++;
1713 }
1714 return s == HT->p[t + 1];
1715}

1717/* The rows of the CSR matrix M with keep[r] >= 0, renumbered to keep[r] (increasing in r) */
1718static csr_t * csr_keep_rows(const csr_t * const M, const int * const keep, const int nrows) {
1719 int nz = 0;
1720 for (int r = 0; r < M->rows; r++)
1721 if (keep[r] >= 0) nz += M->p[r + 1] - M->p[r];
1722 csr_t * const R = csr_init(NULL, nrows, M->cols, nz > 0 ? nz : 1);
1723 int t2 = 0, r2 = 0;
1724 for (int r = 0; r < M->rows; r++) {
1725 if (keep[r] < 0) continue;
1726 R->p[r2++] = t2;
1727 for (int t = M->p[r]; t < M->p[r + 1]; t++) R->i[t2++] = M->i[t];
1728 }
1729 R->p[nrows] = t2;
1730 R->nz = -1; /* compressed form */
1731 return R;
1732}

1734/* Read the detectors to drop from the file fnam (see dem_drop_detectors()): 0-based indices separated by white space or
1735 * commas, lines starting with '#' or '%' are comments. Sets inM[r] = 1 for the detectors r < m (a larger index is
1736 * ignored: no error mechanism has this detector) and returns their number. */
1737static int dem_read_drop_list(const char fnam[], const int m, char * const inM) {
1738 FILE * const f = fopen(fnam, "r");
1739 if (!f)
1740 ERROR("can't open the drop file '%s' for reading: %s", fnam, strerror(errno));
1741 char *line = NULL;
1742 size_t cap = 0;
1743 int num = 0, lineno = 0;
1744 while (getline(&line, &cap, f) != -1) {
1745 lineno++;
1746 char *s = line;
1747 while (isspace((unsigned char)*s)) s++;
1748 if (*s == '#' || *s == '%') continue;
1749 for (char *tok = strtok(s, " \t\r\n,"); tok; tok = strtok(NULL, " \t\r\n,")) {
1750 char *end;
1751 errno = 0;
1752 const long v = strtol(tok, &end, 10);
1753 if ((*end != '\0') || errno || (v < 0) || (v > INT_MAX))
1754 ERROR("drop file '%s', line %d: invalid entry '%s' (expected 0-based detector indices)", fnam, lineno, tok);
1755 if ((v < m) && !inM[v]) {
1756 inM[v] = 1;
1757 num++;
1758 }
1759 }
1760 }
1761 free(line);
1762 fclose(f);
1763 return num;
1764}

1779/* The columns of the DEM to (re)examine in the propagation of dem_drop_auto(): a queue without duplicates */
1780typedef struct {
1781 int *q; /* the columns, a circular buffer of size n */
1782 char *in; /* in[j]: column j is in the queue */
1783 int n, head, len;
1784} drop_queue_t;
1785
1786static void drop_queue_push(drop_queue_t * const Q, const int j) {
1787 if (Q->in[j]) return;
1788 Q->in[j] = 1;
1789 Q->q[(Q->head + Q->len++) % Q->n] = j;
1790}
1791
1792static int drop_queue_pop(drop_queue_t * const Q) {
1793 const int j = Q->q[Q->head];
1794 Q->head = (Q->head + 1) % Q->n;
1795 Q->len--;
1796 Q->in[j] = 0;
1797 return j;
1798}

1800static int drop_find(int * const parent, int x) { /* union-find with path halving */
1801 while (parent[x] != x) {
1802 parent[x] = parent[parent[x]];
1803 x = parent[x];
1804 }
1805 return x;
1806}

1808typedef struct { /* a group of detectors (dropped all or none), see dem_drop_auto() */

1810} drop_group_t;

1812static int drop_group_cmp(const void *a, const void *b) { /* the largest groups first, then by their root */
1813 const drop_group_t *x = a, *y = b;
1814 if (x->size != y->size) return (x->size < y->size) ? 1 : -1;
1815 return (x->root > y->root) - (x->root < y->root);
1816}

1818/* drop=auto: find a set M of detectors of the DEM which can be dropped exactly with A = 0 (every column j touching M
1819 * has a twin with the same other detectors and the same observables, or none if L(j) = 0 and j has no other detector,
1820 * see dem_drop_detectors()), from the DEM alone. Sets inM[r] = 1 for the detectors of M and returns |M|.
1821 * - The droppable sets are closed under union, so that there is a largest one, which gives the smallest problem.
1822 * - Sub(j): the columns t with the same observables whose detectors are a proper subset of those of j. If j touches
1823 * M, its detectors outside M are those of one t in Sub(j) (or none, if L(j) = 0). So a column with Sub(j) empty
1824 * (O-minimal) is dropped all or nothing (its detectors form a group), and not at all if L(j) != 0.
1825 * - Seeds: the groups not kept, the largest first. Once a detector of j is dropped, the candidates (the elements of
1826 * Sub(j), and none if L(j) = 0, consistent with the detectors of j already dropped or kept) force the others: a
1827 * detector in no candidate is dropped, one in all of them is kept (bit masks of the positions in j). Without any
1828 * candidate, no droppable set contains the seed group, which is kept. Otherwise, the detectors dropped are kept in M
1829 * if every column touching them has the mask of its other detectors in Sub(j).
1830 * A column with more than 64 detectors keeps its detectors (still exact). The result is verified again in
1831 * dem_drop_detectors(), which also allows A != 0 (drop=FILE). */
1832static int dem_drop_auto(const params_t * const p, char * const inM) {
1833 const csr_t * const H = p->spaH; /* row r: the columns with detector r */
1834 const int m = H->rows, n = H->cols;
1835 const double t0 = (p->debug & DBG_TIMING) ? drop_wall_time() : 0.0;
1836 csr_t * const HT = csr_transpose(NULL, p->spaH); /* row j: the detectors of column j (sorted) */
1837 csr_t * const LT = csr_transpose(NULL, p->spaL); /* row j: the observables of column j (sorted) */
1838 int * const fp = calloc((size_t)m + 2, sizeof(int)); /* the columns by their first (smallest) detector */
1839 int * const fi = malloc(((size_t)n + 1) * sizeof(int));
1840 int * const stamp = calloc((size_t)m + 1, sizeof(int)); /* stamp[r] = j + 1: detector r is in column j */
1841 char * const posv = malloc((size_t)m + 1); /* the position of detector r in column j */
1842 int * const sp = malloc(((size_t)n + 1) * sizeof(int)); /* Sub(j): entries sp[j] .. sp[j + 1] - 1 */
1843 char * const big = calloc((size_t)n + 1, 1); /* column j has more than 64 detectors */
1844 if (!fp || !fi || !stamp || !posv || !sp || !big) ERROR("memory allocation");
1845 for (int j = 0; j < n; j++)
1846 if (HT->p[j + 1] > HT->p[j]) fp[HT->i[HT->p[j]] + 1]++;
1847 for (int r = 0; r < m; r++) fp[r + 1] += fp[r];
1848 for (int j = 0, *cnt = fp; j < n; j++) /* (fp[r] counts up to the end of the list of r, restored below) */
1849 if (HT->p[j + 1] > HT->p[j]) fi[cnt[HT->i[HT->p[j]]]++] = j;
1850 for (int r = m; r > 0; r--) fp[r] = fp[r - 1];
1851 fp[0] = 0;
1852 size_t cap = (size_t)n + 16, ns = 0;
1853 int *st = malloc(cap * sizeof(int)); /* Sub(j): the column t */
1854 uint64_t *sm = malloc(cap * sizeof(uint64_t)); /* Sub(j): the mask of the positions of the detectors of t in j */
1855 if (!st || !sm) ERROR("memory allocation");
1856 for (int j = 0; j < n; j++) {
1857 sp[j] = (int)ns;
1858 const int b = HT->p[j], w = HT->p[j + 1] - b;
1859 if (w > 64) big[j] = 1;
1860 if ((w < 2) || (w > 64)) continue;
1861 for (int u = 0; u < w; u++) {
1862 stamp[HT->i[b + u]] = j + 1;
1863 posv[HT->i[b + u]] = (char)u;
1864 }
1865 for (int u = 0; u < w; u++) { /* the columns t whose first detector is in j */
1866 const int r = HT->i[b + u];
1867 for (int s = fp[r]; s < fp[r + 1]; s++) {
1868 const int t = fi[s], bt = HT->p[t], wt = HT->p[t + 1] - bt;
1869 if ((wt >= w) || (stamp[HT->i[bt + wt - 1]] != j + 1) || !csr_rows_equal(LT, j, t)) continue;
1870 uint64_t mask = 0;
1871 int v = 0;
1872 for (; (v < wt) && (stamp[HT->i[bt + v]] == j + 1); v++) mask |= 1ULL << posv[HT->i[bt + v]];
1873 if (v < wt) continue;
1874 if (ns == cap) {
1875 cap *= 2;
1876 st = realloc(st, cap * sizeof(int));
1877 sm = realloc(sm, cap * sizeof(uint64_t));
1878 if (!st || !sm) ERROR("memory allocation");
1879 }
1880 st[ns] = t;
1881 sm[ns++] = mask;
1882 }
1883 }
1884 }
1885 sp[n] = (int)ns;
1886 free(posv);
1887 free(stamp);
1888 free(fi);
1889 free(fp);
1890 /* the groups of the O-minimal columns (all or nothing); kept: those with observables, and big columns */
1891 int * const parent = malloc(((size_t)m + 1) * sizeof(int));
1892 char * const state = calloc((size_t)m + 1, 1);
1893 char * const keepg = calloc((size_t)m + 1, 1); /* keepg[root]: the group is kept */
1894 if (!parent || !state || !keepg) ERROR("memory allocation");
1895 for (int r = 0; r < m; r++) parent[r] = r;
1896 for (int j = 0; j < n; j++) {
1897 const int b = HT->p[j], w = HT->p[j + 1] - b;
1898 if ((w == 0) || big[j] || (sp[j + 1] > sp[j])) continue;
1899 const int r0 = drop_find(parent, HT->i[b]);
1900 for (int u = 1; u < w; u++) {
1901 const int r = drop_find(parent, HT->i[b + u]);
1902 if (r != r0) parent[r] = r0;
1903 }
1904 }
1905 for (int j = 0; j < n; j++) {
1906 const int b = HT->p[j], w = HT->p[j + 1] - b;
1907 if (w == 0) continue;
1908 if (big[j]) {
1909 for (int u = 0; u < w; u++) keepg[drop_find(parent, HT->i[b + u])] = 1;
1910 } else if ((sp[j + 1] == sp[j]) && (LT->p[j + 1] > LT->p[j])) {
1911 keepg[drop_find(parent, HT->i[b])] = 1;
1912 }
1913 }
1914 int * const gsize = calloc((size_t)m + 1, sizeof(int));
1915 if (!gsize) ERROR("memory allocation");
1916 for (int r = 0; r < m; r++) /* (then parent[r] is the root of r for all r) */
1917 parent[r] = drop_find(parent, r);
1918 for (int r = 0; r < m; r++)
1919 if (H->p[r + 1] == H->p[r]) keepg[r] = 1; /* (a detector in no column is kept: an empty row of H) */
1920 for (int r = 0; r < m; r++) {
1921 const int g = parent[r];
1922 gsize[g]++;
1923 if (keepg[g]) state[r] = DROP_NO;
1924 }
1925 int ng = 0;
1926 for (int r = 0; r < m; r++) ng += (parent[r] == r) && !keepg[r];
1927 drop_group_t * const grp = malloc(((size_t)ng + 1) * sizeof(drop_group_t));
1928 int * const gp = calloc((size_t)m + 2, sizeof(int)); /* the detectors by group */
1929 int * const gi = malloc(((size_t)m + 1) * sizeof(int));
1930 if (!grp || !gp || !gi) ERROR("memory allocation");
1931 for (int r = 0, g = 0; r < m; r++)
1932 if ((parent[r] == r) && !keepg[r]) grp[g++] = (drop_group_t){.root = r, .size = gsize[r]};
1933 qsort(grp, ng, sizeof(drop_group_t), drop_group_cmp);
1934 for (int r = 0; r < m; r++) gp[parent[r] + 1]++;
1935 for (int r = 0; r < m; r++) gp[r + 1] += gp[r];
1936 for (int r = 0, *cnt = gsize; r < m; r++) cnt[r] = gp[r];
1937 for (int r = 0; r < m; r++) gi[gsize[parent[r]]++] = r;
1938 /* the seeds: propagation, with an undo log */
1939 drop_queue_t Q = {.q = malloc(((size_t)n + 1) * sizeof(int)), .in = calloc((size_t)n + 1, 1), .n = n + 1};
1940 int * const ulog = malloc(((size_t)m + 1) * sizeof(int));
1941 if (!Q.q || !Q.in || !ulog) ERROR("memory allocation");
1942 int n_ok = 0, n_fail = 0, n_unver = 0;
1943 long long int work = 0;
1944 const long long int work_max = 64LL * (HT->p[n] + n) + 1000000LL; /* (seeds are skipped beyond this budget) */
1945 for (int g = 0; g < ng; g++) {
1946 const int g0 = gp[grp[g].root], g1 = gp[grp[g].root + 1];
1947 int unk = (g0 < g1);
1948 for (int s = g0; (s < g1) && unk; s++) unk = (state[gi[s]] == DROP_UNK);
1949 if (!unk) continue; /* (already dropped or kept) */
1950 if (work > work_max) break;
1951 int nlog = 0, fail = 0;
1952 for (int s = g0; s < g1; s++) {
1953 const int r = gi[s];
1954 state[r] = DROP_YES;
1955 ulog[nlog++] = r;
1956 for (int t = H->p[r]; t < H->p[r + 1]; t++) drop_queue_push(&Q, H->i[t]);
1957 }
1958 while (Q.len > 0) {
1959 const int j = drop_queue_pop(&Q);
1960 if (fail) continue; /* (empties the queue) */
1961 work++;
1962 const int b = HT->p[j], w = HT->p[j + 1] - b;
1963 if (big[j]) continue; /* (its detectors are kept) */
1964 uint64_t md = 0, mk = 0, mu = 0;
1965 for (int u = 0; u < w; u++) {
1966 const char s = state[HT->i[b + u]];
1967 if (s == DROP_YES) md |= 1ULL << u;
1968 else if (s == DROP_NO) mk |= 1ULL << u;
1969 else mu |= 1ULL << u;
1970 }
1971 if (!md) continue;
1972 uint64_t any = 0, all = ~0ULL;
1973 int nc = 0;
1974 for (int s = sp[j]; s < sp[j + 1]; s++) {
1975 if ((sm[s] & md) || ((sm[s] & mk) != mk)) continue;
1976 any |= sm[s];
1977 all &= sm[s];
1978 nc++;
1979 }
1980 if ((LT->p[j + 1] == LT->p[j]) && !mk) { /* none: all detectors of j dropped */
1981 all = 0;
1982 nc++;
1983 }
1984 if (nc == 0) {
1985 fail = 1;
1986 continue;
1987 }
1988 const uint64_t fd = mu & ~any, fk = mu & all;
1989 for (int u = 0; u < w; u++) {
1990 if (!((fd | fk) & (1ULL << u))) continue;
1991 const int r = HT->i[b + u];
1992 state[r] = (fd & (1ULL << u)) ? DROP_YES : DROP_NO;
1993 ulog[nlog++] = r;
1994 for (int t = H->p[r]; t < H->p[r + 1]; t++) drop_queue_push(&Q, H->i[t]);
1995 }
1996 }
1997 if (!fail) { /* the check: the other detectors of every column touching a detector dropped here */
1998 for (int l = 0; (l < nlog) && !fail; l++) {
1999 const int r = ulog[l];
2000 if (state[r] != DROP_YES) continue;
2001 for (int t = H->p[r]; (t < H->p[r + 1]) && !fail; t++) {
2002 const int j = H->i[t], b = HT->p[j], w = HT->p[j + 1] - b;
2003 uint64_t mo = 0; /* the positions of the detectors of j not dropped */
2004 for (int u = 0; u < w; u++)
2005 if (state[HT->i[b + u]] != DROP_YES) mo |= 1ULL << u;
2006 int ok = (mo == 0) && (LT->p[j + 1] == LT->p[j]);
2007 for (int s = sp[j]; (s < sp[j + 1]) && !ok; s++) ok = (sm[s] == mo);
2008 if (!ok) fail = 2;
2009 }
2010 }
2011 }
2012 if (fail) { /* undo; a contradiction (fail = 1): no droppable set contains the group, which is kept */
2013 for (int l = 0; l < nlog; l++) state[ulog[l]] = DROP_UNK;
2014 if (fail == 1) {
2015 for (int s = g0; s < g1; s++) state[gi[s]] = DROP_NO;
2016 n_fail++;
2017 } else {
2018 n_unver++;
2019 }
2020 } else {
2021 n_ok++;
2022 }
2023 }
2024 int num = 0;
2025 for (int r = 0; r < m; r++)
2026 if (state[r] == DROP_YES) {
2027 inM[r] = 1;
2028 num++;
2029 }
2030 if (p->debug & DBG_TIMING)
2031 fprintf(stderr, "# drop=auto: %d of %d detectors found (%d groups to try: %d dropped, %d kept, %d not verified%s; "
2032 "%zu sub-columns), %.3f s\n", num, m, ng, n_ok, n_fail, n_unver,
2033 (work > work_max) ? ", work limit reached" : "", ns, drop_wall_time() - t0);
2034 free(ulog);
2035 free(Q.in);
2036 free(Q.q);
2037 free(gi);
2038 free(gp);
2039 free(grp);
2040 free(gsize);
2041 free(keepg);

2069/* Drop the detectors (a set M) from a DEM if this is verified to be exact: those found by dem_drop_auto() (drop=auto,
2070 * the default), or those listed in the file p->fdrop (e.g., the minority-basis detectors of a CSS circuit). Every
2071 * column j touching M needs a twin: a column t touching no detector
2072 * of M with the same other (kept) detectors, or no column if j has no kept detector, such that L(t) = L(j) + A H_M(j)
2073 * for one binary matrix A (A = 0 for the usual CSS circuits; found by elimination otherwise). Replacing the columns
2074 * of a codeword c by their twins (two equal twins cancel) gives a codeword of weight at most |c| without the detectors

2076 * submatrix of the kept detectors and the columns touching no detector of M (with L), and its codewords are codewords
2077 * of the DEM. Otherwise, a note is printed and the DEM is not changed. The column maps p->col_orig and p->col_class

2378/* Remove the heaviest weight classes from the hash as long as the remaining codewords keep the collection window
2379 * and at least `cw_rep_size()` codewords (whole weight classes are kept) */
2380static void cw_trim(params_t * const p) {
2381 if (p->min_w == INT_MAX || p->cw_max_w <= p->min_w) return;
2382 const long long int rep = cw_rep_size(p);
2383 if (rep > 0 && p->num_cws <= rep) return;
2384 const long long int top = (long long int)p->min_w + cw_window_dw(p);
2385 if (top >= p->cw_max_w) return;
2386 int keep = (int)top;
2387 if (rep > 0) { /* the smallest weight such that the codewords up to this weight form the representative set */
2388 long long int cum = 0;
2389 int w_rep = p->cw_max_w;
2390 for (int w = p->min_w; w < p->cw_max_w; w++) {
2391 cum += p->cw_cnt_w[w];
2392 if (cum >= rep) {
2393 w_rep = w;
2394 break;
2395 }
2396 }
2397 if (w_rep > keep) keep = w_rep;
2398 }
2399 if (keep >= p->cw_max_w) return;
2400 cw_vec_t *cw, *tmp;
2401 HASH_ITER(hh, p->codewords, cw, tmp) {
2402 if (cw->weight > keep) {
2403 p->cw_cnt_w[cw->weight]--;
2404 p->cw_hits_w[cw->weight] -= cw->cnt;
2405 HASH_DEL(p->codewords, cw);
2406 free(cw);
2407 p->num_cws--;
2408 }
2409 }
2410 p->cw_max_w = keep;
2411 while (p->cw_max_w > p->min_w && p->cw_cnt_w[p->cw_max_w] == 0) p->cw_max_w--;
2412}

2605/* Upper tail probability P(X >= x) of the chi-square distribution with `df` degrees of freedom (exact for df <= 2,
2606 * Wilson-Hilferty approximation otherwise) */
2607static double chi2_upper_tail(const double x, const long long int df) {
2608 if (!(x > 0.0)) return 1.0;
2609 if (df == 1) return erfc(sqrt(0.5 * x));
2610 if (df == 2) return exp(-0.5 * x);
2611 const double s = 2.0 / (9.0 * (double)df);
2612 const double z = (cbrt(x / (double)df) - (1.0 - s)) / sqrt(s);
2613 return 0.5 * erfc(z / sqrt(2.0));
2614}

2616/* Mean mu of the Poisson distribution whose zero-truncated version has the mean m > 1, i.e., m = mu / (1 - e^-mu):
2617 * Newton iteration for the convex function g(mu) = mu - m (1 - e^-mu), starting to the right of the root */
2618static double ztp_mu(const double m) {
2619 double mu = m;
2620 for (int it = 0; it < 200; it++) {
2621 const double e = exp(-mu);
2622 const double dg = 1.0 - m * e;
2623 if (!(dg > 0.0)) break;
2624 const double step = (mu - m * (1.0 - e)) / dg;
2625 mu -= step;
2626 if (fabs(step) <= 1e-12 * (1.0 + mu)) break;
2627 }
2628 return mu;
2629}

2631int codeword_hit_check(FILE *stream, const params_t * const p) {
2632 if (!p || p->min_w == INT_MAX || !p->cw_cnt_w) return 0;
2633 cw_hit_stats_t st;
2634 codeword_hit_stats(p, &st);
2635 if (st.set_cws <= 0) return 0;
2636 cw_class_stats_t *cs = cw_class_stats(p, st.w_lo, st.w_hi);
2637 int num_bad = 0;
2638 for (int w = st.w_lo; w <= st.w_hi; w++) {
2639 const cw_class_stats_t * const c = &cs[w - st.w_lo];
2640 if (c->n < 2) continue;
2641 const double n = (double)c->n;
2642 const double m = (double)c->hits / n;
2643 if (m <= 1.0 + 1e-9) continue; /* every codeword found once: no information */
2644 double s2 = (c->sumsq - n * m * m) / (n - 1.0); /* sample variance of the hit counts */
2645 if (s2 < 0.0) s2 = 0.0;
2646 /* with equal hit rates, a hit count is Poisson(mu) conditioned on >= 1 (zero-truncated), with mean m and
2647 * variance v0; codewords tracked only after their first hit have a smaller variance (the test is conservative) */
2648 const double mu = ztp_mu(m);
2649 const double v0 = m * (1.0 + mu - m);
2650 if (!(mu > 0.0) || !(v0 > 0.0)) continue;
2651 const double pval = chi2_upper_tail((n - 1.0) * s2 / v0, c->n - 1); /* dispersion test */
2652 const double cv2 = (s2 - v0) / (mu * mu); /* squared relative spread of the hit rates */
2653 if (!(pval < 1e-3) || !(cv2 >= 1.0)) continue; /* significant and strong: spread of hit rates >= their mean */
2654 /* a lighter codeword with a gamma-distributed hit rate of the same relative spread is missed with the
2655 * probability E[exp(-lambda)] = (1 + cv2 mu)^(-1/cv2), instead of exp(-mu) with equal hit rates: warn if this
2656 * is at least 3 times larger */
2657 const double p_unif = exp(-mu);
2658 const double p_spread = pow(1.0 + cv2 * mu, -1.0 / cv2);
2659 if (p_spread >= 3.0 * p_unif) {
2660 num_bad++;
2661 if (stream) {
2662 fprintf(stream,
2663 "# Warning: non-uniform RW hit counts of the %lld codewords of weight %d: hits min=%d, max=%d, "
2664 "avg=%.2f, stdev=%.2f (expected %.2f; p=%.1e)\n"
2665 "# some codewords are found much more often than others of the same weight (relative spread of "
2666 "hit rates %.2f):\n"
2667 "# a lighter codeword may be missed with probability ~%.2g rather than exp(-%.2f)=%.2g; "
2668 "consider a larger min_hits\n",
2669 c->n, w, c->c_min, c->c_max, m, sqrt(s2), sqrt(v0), pval, sqrt(cv2), p_spread, mu, p_unif);
2670 }
2671 }
2672 }
2673 free(cs);
2674 return num_bad;
2675}

2682double rw_infoset_find_prob(const int n, const int rank, const int w) {
2683 const int k = n - rank; /* number of non-pivot columns */
2684 if (n <= 0 || rank < 0 || k <= 0 || w < 1 || w > n || w - 1 > rank) return 0.0;
2685 /* exactly one of the w positions among the k non-pivot columns */
2686 const double p1 = exp(log((double)w) + log_binom(n - w, k - 1) - log_binom(n, k));

2688}

2698void print_rw_infoset_estimate(FILE *stream, const int n, const int rank, const int w_lo, const int w_hi,
2699 const long steps, const long steps_total, const double t_step) {
2700 if (!stream || w_lo < 1 || w_hi < w_lo) return;
2701 int w_hard = w_hi; /* the weight hardest to find (smallest P1) */
2702 double p1 = rw_infoset_find_prob(n, rank, w_hi);
2703 for (int w = w_lo; w < w_hi; w++) {
2704 const double pw = rw_infoset_find_prob(n, rank, w);
2705 if (pw < p1) {
2706 p1 = pw;
2707 w_hard = w;
2708 }
2709 }
2710 if (!(p1 > 0.0)) {
2711 fprintf(stream, "# information-set estimate: a codeword of weight %d cannot be found by RW (n=%d, rank(H)=%d)\n",
2712 w_hard, n, rank);
2713 return;
2714 }
2715 const double lq = (p1 < 1.0) ? log1p(-p1) : -INFINITY; /* log of the miss probability per step */
2716 const double p_miss = (steps > 0) ? exp((double)steps * lq) : 1.0;
2717 /* RW steps for a 1% miss probability: uniform steps, scaled to all steps with the fraction of uniform steps */
2718 const double unif_steps_1pc = (p1 < 1.0) ? ceil(log(0.01) / lq) : 1.0;
2719 const double steps_1pc = (steps > 0 && steps_total > steps) ? ceil(unif_steps_1pc * steps_total / steps)
2720 : unif_steps_1pc;
2721 char steps_str[32], time_str[64] = "";
2722 snprintf(steps_str, sizeof(steps_str), (steps_1pc < 1e9) ? "%.0f" : "%.3g", steps_1pc);
2723 if (t_step > 0.0) {
2724 char dur[32];
2725 format_duration(dur, sizeof(dur), steps_1pc * t_step);
2726 snprintf(time_str, sizeof(time_str), ", ~%s", dur);
2727 }
2728 fprintf(stream,
2729 "# information-set estimate (uniform random information sets, n=%d, rank(H)=%d): a codeword of weight %d\n"
2730 "# is found with probability %.3g per RW step, missed in %ld uniform steps with probability %.2g "
2731 "(1%% after %s RW steps in total%s)\n",
2732 n, rank, w_hard, p1, steps, p_miss, steps_str, time_str);
2733}
