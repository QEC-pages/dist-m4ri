64def git_info(repo_dir: Path) -> Tuple[Optional[str], Optional[bool]]:
65 """Return (short_commit, dirty_bool) for a git repository, or (None, None) if unavailable."""
66 if not repo_dir.is_dir():
67 return None, None
68 try:
69 head = subprocess.run(
70 ["git", "-C", str(repo_dir), "rev-parse", "--short=7", "HEAD"],
71 capture_output=True,
72 text=True,
73 timeout=5,
74 )
75 if head.returncode != 0:
76 return None, None
77 commit = head.stdout.strip() or None
78 st = subprocess.run(
79 ["git", "-C", str(repo_dir), "status", "--porcelain"],
80 capture_output=True,
81 text=True,
82 timeout=5,
83 )
84 dirty = bool(st.stdout.strip()) if st.returncode == 0 else None
85 return commit, dirty
86 except Exception:
87 return None, None
