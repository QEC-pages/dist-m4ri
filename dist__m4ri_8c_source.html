92static inline uint64_t splitmix64(uint64_t *state) {
93 uint64_t z = (*state += 0x9e3779b97f4a7c15ULL);
94 z = (z ^ (z >> 30)) * 0xbf58476d1ce4e5b9ULL;
95 z = (z ^ (z >> 27)) * 0x94d049bb133111ebULL;
96 return z ^ (z >> 31);

104static inline mzp_t * mzp_rand_thread(mzp_t *q, rci_t length, uint64_t *state) {
105 if (q == NULL) return NULL;
106 for (int i = 0; i <= (int)length - 2; i++) {
107 q->values[i] = i + rand_uniform_thread(length - i, state);

161/* Recursive CC worker function (interruptible) */
162static int start_CC_recurs_mt(one_vec_t *err, one_vec_t *urr, one_vec_t * const syn[],
163 const int w_limit, const int max_col_wt,
164 const csr_t * const mH, const csr_t * const mHT,
165 worker_arg_t *warg) {
166 distfork_ctx_t *ctx = warg->ctx;
167 if (atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed)) {
168 return 0;
169 }
170 params_t * const p = ctx->p;
171 const int w = err->wei;
172 int row = syn[w]->vec[0];
173 const csr_t * const mL = p->spaL;
174 const int col_min = urr->vec[0];
175
176 for (int i1 = mH->p[row]; i1 < mH->p[row+1]; i1++) {
177 const int col = mH->i[i1];
178 if (col > col_min) {
179 int pos = one_ordered_search(err, col);
180 if (pos == -1) {
181 urr->vec[w] = col;
182 urr->wei++;
183 pos = one_ordered_ins(err, col);
184 syn[w+1]->wei = 0;
185 int swei = one_csr_row_combine(syn[w+1], syn[w], mHT, col);
186
187 if (p->smax && swei > 0 && swei <= p->smax && (w + 1 < MAX_W)) {
188 if (swei < warg->min_swei[w + 1]) {
189 warg->min_swei[w + 1] = swei;
190 }
191 }
192
193 int current_limit = w_limit;
194 int cur_dmax = atomic_load_explicit(&ctx->dmax, memory_order_relaxed);
195 if (cur_dmax > 0 && p->dW >= 0) {
196 current_limit = minint(w_limit, cur_dmax + p->dW);
197 }
198
199 if (err->wei < current_limit) {
200 if (swei) {
201 int result = start_CC_recurs_mt(err, urr, syn, w_limit, max_col_wt,
202 mH, mHT, warg);
203 if (result == 1) {
204 urr->wei--;
205 one_ordered_pos_del(err, col, pos);
206 return 1;
207 }
208 }
209 } else {
210 if (!swei) {
211 int nz = (!mL) || sparse_syndrome_non_zero(mL, err->wei, err->vec);
212 if (nz) {
213 bool stop = false;
214 pthread_mutex_lock(&ctx->cw_mutex);
215 p->codewords = codeword_add_maybe(p, err->vec, err->wei);
216 int cur_d = atomic_load(&ctx->dmax);
217 if (p->min_w < cur_d || cur_d == 0) {
218 atomic_store(&ctx->dmax, p->min_w);
219 }
220 int cur_cc_found = atomic_load(&ctx->cc_found_weight);
221 if (cur_cc_found == 0 || err->wei < cur_cc_found) {
222 atomic_store(&ctx->cc_found_weight, err->wei);
223 }
224 if (!p->outC && p->maxC == 0) {
225 atomic_store(&ctx->stop_flag, true);
226 stop = true;
227 }
228 if (p->maxC && p->num_cws >= p->maxC) {
229 atomic_store(&ctx->stop_flag, true);
230 stop = true;
231 }
232 pthread_mutex_unlock(&ctx->cw_mutex);
233
234 if (stop) {
235 urr->wei--;
236 one_ordered_pos_del(err, col, pos);
237 return 1;
238 }
239 }
240 }
241 }
242 urr->wei--;
243 one_ordered_pos_del(err, col, pos);
244 }
245 }
246 }
247 return 0;

250/* Run RW batch */
251static void run_rw_steps(distfork_ctx_t *ctx, int n_steps,
252 mzd_t *mH, mzd_t *mHT, rci_t *ee,
253 mzp_t *perm, mzp_t *pivs, mzp_t *pivs_srtd, mzp_t *skip_pivs,
254 uint64_t *rng_state, int tid) {
255 params_t * const p = ctx->p;
256 const csr_t * const spaL0 = p->spaL;
257 const int nvar = p->spaH->cols;
258 const int classical = p->classical;
259
260 for (int step = 0; step < n_steps; step++) {
261 if (atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed)) break;
262
263 pivs = mzp_rand_thread(pivs, nvar, rng_state);
264 mzp_set_ui(perm, 1);
265 perm = perm_p_trans(perm, pivs, 0);
266
267 int rank = 0;
268 for (int i = 0; i < nvar; i++) {
269 int col = perm->values[i];
270 int ret = gauss_one(mH, col, rank);
271 if (ret) {
272 pivs->values[rank++] = col;
273 }
274 }
275
276 pivs_srtd = mzp_copy(pivs_srtd, pivs);
277 qsort(pivs_srtd->values, rank, sizeof(pivs->values[0]), cmp_rci_t);
278 int end = -1, num = 0;
279 for (int i = 0; i < rank; i++) {
280 int beg = end + 1;
281 end = pivs_srtd->values[i];
282 for (int j = beg; j < end; j++) {
283 skip_pivs->values[num++] = j;
284 }
285 }
286 for (int j = end + 1; j < nvar; j++) {
287 skip_pivs->values[num++] = j;
288 }
289 skip_pivs->length = num;
290
291 mzd_transpose(mHT, mH);
292
293 int k = nvar - rank;
294 for (int ir = 0; ir < k; ir++) {
295 int cnt = 0;
296 const int col = ee[cnt++] = skip_pivs->values[ir];
297 int limit = nvar + 1;
298 if (p->wmax > 0) limit = p->wmax + 1;
299 int cur_dmax = atomic_load_explicit(&ctx->dmax, memory_order_relaxed);
300 if (cur_dmax > 0) {
301 if (p->dW >= 0) limit = minint(limit, cur_dmax + p->dW + 1);
302 else limit = minint(limit, cur_dmax);
303 }
304
305 word *rawrow = mzd_row(mHT, col);
306 rci_t j = -1;
307 while (cnt < limit) {
308 j = nextelement(rawrow, mHT->width, j);
309 if (j == -1 || j >= rank) break;
310 ee[cnt++] = pivs->values[j++];
311 }
312
313 if (cnt < limit) {
314 qsort(ee, cnt, sizeof(rci_t), cmp_rci_t);
315 int nz = classical ? 1 : sparse_syndrome_non_zero(spaL0, cnt, ee);
316 if (nz) {
317 pthread_mutex_lock(&ctx->cw_mutex);
318 p->codewords = codeword_add_maybe(p, ee, cnt);
319 if (cnt < p->min_w) p->min_w = cnt;
320 int best = p->min_w;
321 int old_dmax = atomic_load(&ctx->dmax);
322 if (old_dmax == 0 || best < old_dmax) {
323 atomic_store(&ctx->dmax, best);
324 if (p->debug & 16) {
325 int num_rw = (ctx->p->method == 1) ? ctx->num_threads : (ctx->num_threads - atomic_load(&ctx->cc_target_workers));
326 if (num_rw < 1) num_rw = 1;
327 fprintf(stderr, "# [thread %d] RW found new upper bound cw of weight %d (using %d RW threads)\n", tid, best, num_rw);
328 }
329 int cur_dmin = atomic_load(&ctx->dmin);
330 if (cur_dmin > 0 && best <= cur_dmin) {
331 atomic_store(&ctx->stop_flag, true);
332 }
333 }
334 if (p->wmin > 0 && best <= p->wmin) {
335 atomic_store(&ctx->stop_flag, true);
336 }
337 if (p->maxC && p->num_cws >= p->maxC) {
338 atomic_store(&ctx->stop_flag, true);
339 }
340 pthread_mutex_unlock(&ctx->cw_mutex);
341 }
342 }
343 }
344 atomic_fetch_add(&ctx->rw_steps_completed, 1);
345 }

348/* Worker thread main loop */
349static void *worker_thread_func(void *arg) {
350 worker_arg_t *warg = (worker_arg_t *)arg;
351 distfork_ctx_t *ctx = warg->ctx;
352 int tid = warg->tid;
353 const int nvar = ctx->p->spaH->cols;
354 const bool enable_rw = (ctx->p->method & 1) != 0;
355
356 /* Initialize min_swei for this thread */
357 for (int i = 0; i < MAX_W; i++) {
358 warg->min_swei[i] = ctx->p->spaH->rows + 1;
359 }
360
361 /* Thread-local RW matrices (allocated safely only if RW is enabled) */
362 mzd_t *mH = NULL;
363 mzd_t *mHT_rw = NULL;
364 rci_t *ee = NULL;
365 mzp_t *perm = NULL;
366 mzp_t *pivs = NULL;
367 mzp_t *pivs_srtd = NULL;
368 mzp_t *skip_pivs = NULL;
369 uint64_t rng_state = (uint64_t)ctx->p->seed + (uint64_t)tid * 0x9e3779b97f4a7c15ULL + 0x517cc1b727220a95ULL;
370
371 if (enable_rw) {
372 mH = safe_mzd_from_csr(NULL, ctx->p->spaH);
373 mHT_rw = safe_mzd_init(nvar, ctx->p->spaH->rows);
374 ee = malloc((nvar + 2) * sizeof(rci_t));
375 perm = safe_mzp_init(nvar);
376 pivs = safe_mzp_init(nvar);
377 pivs_srtd = safe_mzp_init(nvar);
378 skip_pivs = safe_mzp_init(nvar);
379 }
380
381 /* Thread-local CC memory */
382 const int wmax = ctx->p->wmax > 0 ? ctx->p->wmax : 100;
383 one_vec_t *err = calloc(1, sizeof(one_vec_t) + sizeof(int) * (wmax + 2));
384 one_vec_t *urr = calloc(1, sizeof(one_vec_t) + sizeof(int) * (wmax + 2));
385 one_vec_t **syn = calloc(wmax + 3, sizeof(one_vec_t *));
386 for (int i = 0; i <= wmax + 2; i++) {
387 syn[i] = calloc(1, sizeof(one_vec_t) + sizeof(int) * (ctx->p->spaH->rows + 1));
388 }
389
390 while (!atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed)) {
391 if (get_time_sec() - ctx->start_time >= ctx->timeout) {
392 atomic_store(&ctx->stop_flag, true);
393 break;
394 }
395
396 bool did_work = false;
397
398 /* 1. Try to take CC work if CC is active (method 2 or 3) */
399 if (ctx->p->method >= 2 && atomic_load(&ctx->cc_round_active)) {
400 int active = atomic_load(&ctx->cc_active_workers);
401 int target = atomic_load(&ctx->cc_target_workers);
402 if (active < target) {
403 int col = atomic_fetch_add(&ctx->cc_col_next, 1);
404 int end = ctx->cc_col_end;
405 if (col <= end) {
406 atomic_fetch_add(&ctx->cc_active_workers, 1);
407 int w = atomic_load(&ctx->cc_weight);
408
409 err->vec[0] = urr->vec[0] = col;
410 err->wei = urr->wei = 1;
411 syn[1]->wei = 0;
412 int swei = one_csr_row_combine(syn[1], syn[0], ctx->mHT_cc, col);
413
414 if (ctx->p->smax && swei > 0 && swei <= ctx->p->smax) {
415 if (swei < warg->min_swei[1]) {
416 warg->min_swei[1] = swei;
417 }
418 }
419
420 if (w > 1) {
421 if (swei) {
422 start_CC_recurs_mt(err, urr, syn, w, ctx->max_col_W, ctx->p->spaH, ctx->mHT_cc, warg);
423 }
424 } else {
425 if (!swei) {
426 int nz = (!ctx->p->spaL) || sparse_syndrome_non_zero(ctx->p->spaL, 1, err->vec);
427 if (nz) {
428 pthread_mutex_lock(&ctx->cw_mutex);
429 ctx->p->codewords = codeword_add_maybe(ctx->p, err->vec, 1);
430 atomic_store(&ctx->cc_found_weight, 1);
431 atomic_store(&ctx->dmin, 1);
432 atomic_store(&ctx->dmax, 1);
433 atomic_store(&ctx->stop_flag, true);
434 pthread_mutex_unlock(&ctx->cw_mutex);
435 }
436 }
437 }
438 err->wei = urr->wei = 0;
439 atomic_fetch_sub(&ctx->cc_active_workers, 1);
440 did_work = true;
441 continue;
442 }
443 }
444 }
445
446 /* 2. Try to take RW work if RW is active (method 1 or 3) */
447 if (enable_rw && !atomic_load(&ctx->stop_flag)) {
448 long cur_s = atomic_load(&ctx->rw_steps_started);
449 if (cur_s < ctx->total_rw_steps) {
450 long target_s = cur_s + 10;
451 if (target_s > ctx->total_rw_steps) target_s = ctx->total_rw_steps;
452 if (atomic_compare_exchange_weak(&ctx->rw_steps_started, &cur_s, target_s)) {
453 int n_steps = (int)(target_s - cur_s);
454 run_rw_steps(ctx, n_steps, mH, mHT_rw, ee, perm, pivs, pivs_srtd, skip_pivs, &rng_state, tid);
455 did_work = true;
456 continue;
457 }
458 }
459 }
460
461 if (!did_work) {
462 usleep(100);
463 }
464 }
465
466 if (enable_rw) {
467 safe_mzp_free(skip_pivs);
468 safe_mzp_free(pivs_srtd);
469 safe_mzp_free(perm);
470 safe_mzp_free(pivs);
471 free(ee);
472 safe_mzd_free(mHT_rw);
473 safe_mzd_free(mH);
474 }
475
476 for (int i = 0; i <= wmax + 2; i++) free(syn[i]);
477 free(syn);
478 free(err);
479 free(urr);
480
481 return NULL;
482}

503/* Method 2 coordinator */
504static void run_method2_coordinator(distfork_ctx_t *ctx) {
505 const int nvar = ctx->p->spaH->cols;
506 const int wmax = ctx->p->wmax;
507 const int w_start = ctx->p->noscan ? wmax : 1;
508
509 if (ctx->p->debug & 2) {
510 fprintf(stderr, "# running method=2 (multithreaded CC) with %d threads, w_start=%d wmax=%d\n",
511 ctx->num_threads, w_start, wmax);
512 }
513
514 int w_limit = wmax;
515
516 for (int w = w_start; w <= w_limit; w++) {
517 if (atomic_load(&ctx->stop_flag)) break;
518 if (get_time_sec() - ctx->start_time >= ctx->timeout) {
519 atomic_store(&ctx->stop_flag, true);
520 break;
521 }
522
523 int beg = (ctx->p->cbeg >= 0) ? ctx->p->cbeg : 0;
524 int end = (ctx->p->cend >= 0) ? minint(ctx->p->cend, nvar - w) : (nvar - w);
525
526 atomic_store(&ctx->cc_weight, w);
527 ctx->cc_col_beg = beg;
528 ctx->cc_col_end = end;
529 atomic_store(&ctx->cc_col_next, beg);
530 atomic_store(&ctx->cc_target_workers, ctx->num_threads);
531 atomic_store(&ctx->cc_round_active, 1);
532
533 double cc_start = get_time_sec();
534
535 if (ctx->p->debug & 2) {
536 fprintf(stderr, "# searching w=%d with %d CC threads, columns [%d, %d]\n",
537 w, ctx->num_threads, beg, end);
538 }
539
540 while (!atomic_load(&ctx->stop_flag)) {
541 if (get_time_sec() - ctx->start_time >= ctx->timeout) {
542 atomic_store(&ctx->stop_flag, true);
543 break;
544 }
545 if (atomic_load(&ctx->cc_col_next) > end && atomic_load(&ctx->cc_active_workers) == 0) {
546 break;
547 }
548 usleep(100);
549 }
550
551 atomic_store(&ctx->cc_round_active, 0);
552
553 double cc_dur = get_time_sec() - cc_start;
554
555 int cw_found = atomic_load(&ctx->cc_found_weight);
556 if (cw_found > 0) {
557 atomic_store(&ctx->dmin, cw_found);
558 atomic_store(&ctx->dmax, cw_found);
559
560 if (ctx->p->outC && ctx->p->dW > 0 && w < minint(wmax, cw_found + ctx->p->dW)) {
561 w_limit = minint(wmax, cw_found + ctx->p->dW);
562 if (ctx->p->debug & 1) {
563 if (w == cw_found) {
564 fprintf(stderr, "# CC round w=%d finished in %.3fs (%d CC threads): found min-weight codewords (dmin=%d, continuing up to w=%d for dW=%d, total %lld cws)\n",
565 w, cc_dur, ctx->num_threads, cw_found, w_limit, ctx->p->dW, ctx->p->num_cws);
566 } else {
567 fprintf(stderr, "# CC round w=%d finished in %.3fs (%d CC threads): extra dW round completed (dmin=%d, total %lld cws)\n",
568 w, cc_dur, ctx->num_threads, cw_found, ctx->p->num_cws);
569 }
570 }
571 } else {
572 if (ctx->p->debug & 1) {
573 if (w > cw_found) {
574 fprintf(stderr, "# CC round w=%d finished in %.3fs (%d CC threads): extra dW round completed (dmin=%d, total %lld cws)\n",
575 w, cc_dur, ctx->num_threads, cw_found, ctx->p->num_cws);
576 } else {
577 fprintf(stderr, "# CC found min-weight codeword: d=%d (using %d CC threads, total %lld cws)\n",
578 cw_found, ctx->num_threads, ctx->p->num_cws);
579 }
580 }
581 if (w >= w_limit) {
582 atomic_store(&ctx->stop_flag, true);
583 break;
584 }
585 }
586 } else {
587 /* Weight w analyzed without success */
588 atomic_store(&ctx->dmin, w + 1);
589 if (ctx->p->debug & 1) {
590 fprintf(stderr, "# CC w=%d completed in %.3fs (%d CC threads): no codewords found -> dmin=%d\n",
591 w, cc_dur, ctx->num_threads, w + 1);
592 }
593 }
594 }

597/* Method 3 coordinator */
598static void run_method3_coordinator(distfork_ctx_t *ctx) {
599 const int nvar = ctx->p->spaH->cols;
600 int w = ctx->p->noscan ? ctx->p->wmax : 1;
601
602 if (ctx->p->debug & 2) {
603 fprintf(stderr, "# running method=3 (bracketing mode) with %d threads, timeout=%.1fs, dexp=%d\n",
604 ctx->num_threads, ctx->timeout, ctx->dexp);
605 }
606
607 /* Initial RW probe to measure average step time */
608 double t_rw_start = get_time_sec();
609 usleep(2000);
610 double t_rw_dur = get_time_sec() - t_rw_start;
611 long initial_steps = atomic_load(&ctx->rw_steps_completed);
612 if (initial_steps > 0) {
613 ctx->avg_rw_step_time = (t_rw_dur * (double)ctx->num_threads) / (double)initial_steps;
614 } else {
615 ctx->avg_rw_step_time = 0.00005;
616 }
617
618 while (!atomic_load(&ctx->stop_flag)) {
619 double now = get_time_sec();
620 double remaining_time = ctx->timeout - (now - ctx->start_time);
621 if (remaining_time <= 0.0) {
622 atomic_store(&ctx->stop_flag, true);
623 break;
624 }
625
626 int cur_dmax = atomic_load(&ctx->dmax);
627 int cur_dmin = atomic_load(&ctx->dmin);
628
629 /* Target cluster size for CC */
630 int target_cc_w;
631 if (cur_dmax > 0) {
632 if (ctx->p->outC && ctx->p->dW > 0 && cur_dmin >= cur_dmax) {
633 target_cc_w = cur_dmax + ctx->p->dW;
634 } else {
635 target_cc_w = cur_dmax - 1;
636 }
637 } else if (ctx->dexp > 0) {
638 target_cc_w = ctx->dexp;
639 } else if (ctx->p->wmax > 0) {
640 target_cc_w = ctx->p->wmax;
641 } else {
642 target_cc_w = nvar;
643 }
644
645 if (ctx->p->wmax > 0 && target_cc_w > ctx->p->wmax) {
646 target_cc_w = ctx->p->wmax;
647 }
648
649 if (cur_dmax > 0 && cur_dmin >= cur_dmax && w > target_cc_w) {
650 /* Bracketing converged and all requested dW rounds completed */
651 atomic_store(&ctx->dmin, cur_dmax);
652 atomic_store(&ctx->stop_flag, true);
653 break;
654 }
655
656 if (w > target_cc_w) {
657 if (cur_dmax > 0 && cur_dmin < cur_dmax) {
658 atomic_store(&ctx->dmin, cur_dmax);
659 atomic_store(&ctx->stop_flag, true);
660 break;
661 }
662 /* Let remaining RW steps finish */
663 while (!atomic_load(&ctx->stop_flag)) {
664 if (get_time_sec() - ctx->start_time >= ctx->timeout) break;
665 if (atomic_load(&ctx->rw_steps_completed) >= ctx->total_rw_steps) break;
666 usleep(1000);
667 }
668 break;
669 }
670
671 /* Estimate CC time for weight w */
672 double t_cc_est;
673 if (w == 1) {
674 t_cc_est = 0.0001;
675 } else if (w == 2) {
676 t_cc_est = 0.001;
677 } else {
678 double prev = ctx->cc_time_per_weight[w - 1];
679 if (prev <= 0.0001) prev = 0.001;
680 double growth = 4.0;
681 if (w >= 3 && ctx->cc_time_per_weight[w - 2] > 0.0001) {
682 growth = ctx->cc_time_per_weight[w - 1] / ctx->cc_time_per_weight[w - 2];
683 if (growth < 2.0) growth = 2.0;
684 if (growth > 10.0) growth = 10.0;
685 }
686 t_cc_est = prev * growth;
687 }
688
689 /* Check if CC for weight w can finish within timeout */
690 if (t_cc_est / ctx->num_threads > remaining_time * 1.5) {
691 if (ctx->p->debug & 2) {
692 fprintf(stderr, "# CC for w=%d (est %.2fs) exceeds remaining timeout %.2fs, devoting %d threads to RW\n",
693 w, t_cc_est / ctx->num_threads, remaining_time, ctx->num_threads);
694 }
695 while (!atomic_load(&ctx->stop_flag)) {
696 if (get_time_sec() - ctx->start_time >= ctx->timeout) break;
697 if (atomic_load(&ctx->rw_steps_completed) >= ctx->total_rw_steps) break;
698 usleep(1000);
699 }
700 break;
701 }
702
703 /* Calculate thread balancing */
704 long steps_done = atomic_load(&ctx->rw_steps_completed);
705 long steps_rem = (ctx->total_rw_steps > steps_done) ? (ctx->total_rw_steps - steps_done) : 0;
706
707 int n_cc;
708 if (ctx->num_threads == 1) {
709 n_cc = 1;
710 } else if (steps_rem == 0) {
711 n_cc = ctx->num_threads;
712 } else if (t_cc_est < 0.005) {
713 n_cc = (ctx->num_threads >= 4) ? 2 : 1;
714 } else {
715 double t_rw_total_1t = (double)steps_rem * ctx->avg_rw_step_time;
716 double t_cc_total_1t = t_cc_est;
717 double est_accum = t_cc_est;
718 for (int k = w + 1; k <= target_cc_w && k <= w + 2; k++) {
719 est_accum *= 4.0;
720 t_cc_total_1t += est_accum;
721 }
722 double ratio = t_cc_total_1t / (t_cc_total_1t + t_rw_total_1t);
723 n_cc = (int)round((double)ctx->num_threads * ratio);
724 if (n_cc < 1) n_cc = 1;
725 if (n_cc >= ctx->num_threads && steps_rem > 0) n_cc = ctx->num_threads - 1;
726 }
727
728 int n_rw = ctx->num_threads - n_cc;
729
730 int beg = (ctx->p->cbeg >= 0) ? ctx->p->cbeg : 0;
731 int end = (ctx->p->cend >= 0) ? minint(ctx->p->cend, nvar - w) : (nvar - w);
732
733 atomic_store(&ctx->cc_weight, w);
734 ctx->cc_col_beg = beg;
735 ctx->cc_col_end = end;
736 atomic_store(&ctx->cc_col_next, beg);
737 atomic_store(&ctx->cc_target_workers, n_cc);
738 atomic_store(&ctx->cc_round_active, 1);
739
740 if (ctx->p->debug & 2) {
741 fprintf(stderr, "# CC round w=%d started: %d CC threads, %d RW threads (bounds [%d, %d], rem_rw=%ld, rem_time=%.2fs)\n",
742 w, n_cc, n_rw, cur_dmin, cur_dmax, steps_rem, remaining_time);
743 }
744
745 double cc_start = get_time_sec();
746
747 while (!atomic_load(&ctx->stop_flag)) {
748 if (get_time_sec() - ctx->start_time >= ctx->timeout) {
749 atomic_store(&ctx->stop_flag, true);
750 break;
751 }
752 if (atomic_load(&ctx->cc_col_next) > end && atomic_load(&ctx->cc_active_workers) == 0) {
753 break;
754 }
755 usleep(100);
756 }
757
758 atomic_store(&ctx->cc_round_active, 0);
759
760 double cc_dur = get_time_sec() - cc_start;
761 ctx->cc_time_per_weight[w] = cc_dur * (double)n_cc;
762
763 int cw_found = atomic_load(&ctx->cc_found_weight);
764 if (cw_found > 0) {
765 atomic_store(&ctx->dmin, cw_found);
766 atomic_store(&ctx->dmax, cw_found);
767
768 if (ctx->p->outC && ctx->p->dW > 0 && w < minint(ctx->p->wmax > 0 ? ctx->p->wmax : nvar, cw_found + ctx->p->dW)) {
769 if (ctx->p->debug & 1) {
770 fprintf(stderr, "# CC round w=%d finished in %.3fs (%d CC threads, %d RW threads): found codewords (dmin=%d, continuing up to w=%d for dW=%d, total %lld cws)\n",
771 w, cc_dur, n_cc, n_rw, cw_found, minint(ctx->p->wmax > 0 ? ctx->p->wmax : nvar, cw_found + ctx->p->dW), ctx->p->dW, ctx->p->num_cws);
772 }
773 } else {
774 if (ctx->p->debug & 1) {
775 fprintf(stderr, "# CC found min-weight codeword: d=%d (using %d CC threads, total %lld cws)\n",
776 cw_found, n_cc, ctx->p->num_cws);
777 }
778 atomic_store(&ctx->stop_flag, true);
779 break;
780 }
781 } else if (cur_dmin > 1 && cur_dmin >= cur_dmax) {
782 /* Extra dW round completed */
783 if (ctx->p->debug & 1) {
784 fprintf(stderr, "# CC round w=%d finished in %.3fs (%d CC threads, %d RW threads): extra dW round completed (dmin=%d, total %lld cws)\n",
785 w, cc_dur, n_cc, n_rw, cur_dmin, ctx->p->num_cws);
786 }
787 } else {
788 if (atomic_load(&ctx->stop_flag)) break;
789
790 /* Weight w analyzed without success */
791 int new_dmin = w + 1;
792 atomic_store(&ctx->dmin, new_dmin);
793 if (ctx->p->debug & 1) {
794 fprintf(stderr, "# CC round w=%d finished in %.3fs (%d CC threads, %d RW threads): no codewords -> dmin=%d\n",
795 w, cc_dur, n_cc, n_rw, new_dmin);
796 }
797
798 cur_dmax = atomic_load(&ctx->dmax);
799 if (cur_dmax > 0 && new_dmin >= cur_dmax) {
800 atomic_store(&ctx->dmin, cur_dmax);
801 if (ctx->p->outC && ctx->p->dW > 0 && cur_dmax + ctx->p->dW > cur_dmax) {
802 if (ctx->p->debug & 1) {
803 fprintf(stderr, "# bracketing bounds coincide: dmin = dmax = %d (continuing up to w=%d for dW=%d)\n",
804 cur_dmax, cur_dmax + ctx->p->dW, ctx->p->dW);
805 }
806 } else {
807 atomic_store(&ctx->stop_flag, true);
808 if (ctx->p->debug & 1) {
809 fprintf(stderr, "# bracketing bounds coincide: dmin = dmax = %d\n", cur_dmax);
810 }
811 break;
812 }
813 }
814 }
815
816 w++;
817 }
