296/* A CC round at weight w hands out the work units u = beg*R .. (end+1)*R-1 one at a time, where beg..end are the start
297 * indices of the round (see cc_round_beg(), cc_round_end()) and R = B^L (see cc_round_levels()). The unit u = i*R + k
298 * grows the clusters from the start index i = u / R, with the branches of the first L levels fixed by the digits
299 * k_0, k_1, ... of k = k_0 + k_1*B + ... in radix B, the maximum row weight of H: at level l = 1..L (a cluster of l
300 * columns), start_CC_recurs_mt() only adds the column at position k_{l-1} of the row it extends (none if the row is
301 * shorter: an empty unit). The units of a start index explore together the same clusters as the start index itself
302 * (L = 0), in pieces small enough to balance the load if there are few start indices per thread (the clusters from the
303 * first start column, which only grow to larger columns, are about w/n of the round). The weight, the number of fixed
304 * levels L, and the next unit are packed into the single atomic word `cc_next`, so that a worker always processes a
305 * claimed unit with the parameters of its round, even if it joins just as the coordinator publishes the next round. */
306#define CC_IDX_BITS 32
307#define CC_LEV_SHIFT 48
308static inline long long cc_pack(const int w, const int lev, const int idx) {
309 return ((long long)lev << CC_LEV_SHIFT) | ((long long)w << CC_IDX_BITS) | (long long)(unsigned int)idx;
310}

312static inline int cc_unpack_lev(const long long v) { return (int)((v >> CC_LEV_SHIFT) & 0xf); }
313static inline int cc_unpack_idx(const long long v) { return (int)(v & 0xffffffffLL); }

333#define CC_SPLIT_UNITS_PER_THREAD 4.0 /* aim at >= 4 work units of the size of the largest one per CC thread */
334#define CC_SPLIT_MIN_WORK 1e-3 /* rounds predicted to take < 1 ms on all threads: whole start indices */
335#define CC_SPLIT_CLAIMS_PER_WORK 1e4 /* at most 10^4 work units per thread-second of predicted work */
336#define CC_SPLIT_CLAIMS_UNKNOWN 65536 /* at most this many work units if the work of the round is not known */

338/* Number of fixed levels L of the work units of the CC round at weight w over the start indices beg..end, with the
339 * estimated work `est` (thread-seconds, 0 if not known, see cc_work_estimate()), see `cc_pack()`: the smallest L which
340 * gives CC_SPLIT_UNITS_PER_THREAD units per thread, where the largest unit is taken as the unit size: whole start
341 * indices count as n/w units (the clusters from the first start column are about w/n of the round; the columns of the
342 * expert `start` list are alike, with unlimited clusters), and each fixed level divides the units by the typical
343 * number of branches. L = 0 for short rounds, and L <= min(3, w-3), so that at least two levels of a unit are free.
344 * The number of units (each one claim of the shared counter, empty units included) is limited by the predicted work,
345 * and all units must have an index < INT_MAX - num_threads (over-claims at the end of the round). */
346static int cc_round_levels(const distfork_ctx_t * const ctx, const int w, const int beg, const int end,
347 const double est) {
348 const int nthr = ctx->num_threads;
349 const int n_start = end - beg + 1;
350 const int lmax = minint(CC_SPLIT_MAX_LEVELS, w - 3);
351 if (nthr <= 1 || n_start <= 0 || lmax <= 0 || ctx->cc_B <= 1) return 0;
352 if (est > 0.0 && est < CC_SPLIT_MIN_WORK * nthr) return 0;
353 const double max_units = (est > 0.0) ? CC_SPLIT_CLAIMS_PER_WORK * est : CC_SPLIT_CLAIMS_UNKNOWN;
354 double units = (ctx->p->start_num > 0) ? (double)n_start : (double)n_start / w;
355 int lev = 0;
356 while (lev < lmax && units < CC_SPLIT_UNITS_PER_THREAD * nthr) {
357 const long long R = ctx->cc_R[lev + 1];
358 if (R <= 0 || (double)n_start * (double)R > max_units || ((long long)end + 1) * R > (long long)INT_MAX - nthr - 1)
359 break;
360 lev++;
361 units *= ctx->cc_branch;
362 }
363 return lev;
364}

413static inline int bits_first(const uint64_t * const b, const int from, const int limit) {

446/* Record the codeword: the cluster with the columns c1 (unless negative) and c2, if its syndrome of L is nonzero;
447 * returns true if the run is to stop */
448static bool cc_walk_found(cc_walk_t * const t, const int c1, const int c2) {
449 one_vec_t * const cw = t->cw;
450 int n = 0;
451 for (int i = 0; i < t->w; i++) cw->vec[n++] = t->cols[i];
452 if (c1 >= 0) cw->vec[n++] = c1;
453 cw->vec[n++] = c2;
454 for (int i = 1; i < n; i++) { /* insertion sort */
455 const int x = cw->vec[i];
456 int j = i - 1;
457 while (j >= 0 && cw->vec[j] > x) {
458 cw->vec[j + 1] = cw->vec[j];
459 j--;
460 }
461 cw->vec[j + 1] = x;
462 }
463 cw->wei = n;
464 const colmask_t * const mL = t->ctx->p->maskL; /* NULL for a classical code */
465 if (mL && !colmask_syndrome_non_zero(mL, n, cw->vec)) return false;
466 return cc_record(t->ctx, t->warg, cw);
467}

469/* Is the syndrome S (bitset of weight sw) the sum h_{c1} + h_{c2} of the columns c1 and c2 of H (rows of mHT)? */
470static bool cc_sum2_equal(const uint64_t * const S, const int sw, const csr_t * const mHT, const int c1, const int c2) {
471 int a = mHT->p[c1], b = mHT->p[c2], k = 0;
472 const int a_end = mHT->p[c1 + 1], b_end = mHT->p[c2 + 1];
473 while (a < a_end || b < b_end) {
474 int r;
475 if (b >= b_end) r = mHT->i[a++];
476 else if (a >= a_end) r = mHT->i[b++];
477 else if (mHT->i[a] < mHT->i[b]) r = mHT->i[a++];
478 else if (mHT->i[b] < mHT->i[a]) r = mHT->i[b++];
479 else { /* a common row cancels */
480 a++;
481 b++;
482 continue;
483 }
484 if (!bit_get(S, r)) return false;
485 k++;
486 }
487 return k == sw;
488}

490/* The last two levels of cc_walk() at the level w = t->w with w + 2 = current_limit (the leaf level w + 1 is not a
491 * fixed level of the work unit; `i1_beg`..`i1_end`: the positions in mH of the columns of the row r0 = min(S) to add).
492 * There, the recursion adds each column c1 > col_min of r0 not in the cluster with the syndrome s = S + h_{c1} nonzero,
493 * and the leaf level adds each column c2 > col_min of the row min(s) not in the cluster with h_{c2} = s, which gives a
494 * codeword. These columns c2 are found by a lookup of the hash h ^ hc[c1] of s instead (cc_hash_t, verified exactly).
495 * The pairs (c1, c2) are the same and come in the same order (c2 in ascending order, as in the row min(s)). As at the
496 * entry of the leaf level, a codeword is not recorded once the stop flag is set or the weight limit has dropped (a
497 * smaller upper bound). Returns 1 if the run is to stop. */
498static __attribute__((noinline)) int cc_walk_last2(cc_walk_t * const t, const int sw, const uint64_t h,
499 const int i1_beg, const int i1_end) {
500 distfork_ctx_t * const ctx = t->ctx;
501 const cc_hash_t * const hs = t->hs;
502 const uint64_t * const hc = hs->hc;
503 const uint64_t * const bits = hs->bits;
504 const uint64_t bmask = hs->bmask;
505 const csr_t * const mH = t->mH;
506 const int col_min = t->col_min;
507 for (int i1 = i1_beg; i1 < i1_end; i1++) {
508 const int c1 = mH->i[i1];
509 if (c1 <= col_min) continue;
510 const uint64_t key = h ^ hc[c1]; /* the hash of s = S + h_{c1} */
511 const uint64_t fb = key & bmask;
512 if (!((bits[fb >> 6] >> (fb & 63)) & 1)) continue;
513 int j = (int)(key >> hs->tshift);
514 while (hs->tab[j].cnt > 0 && hs->tab[j].key != key) j = (j + 1) & hs->tmask;
515 if (hs->tab[j].cnt == 0) continue;
516 /* columns c2 with the hash of s (s is nonzero: no zero columns in the table) */
517 if (bit_get(t->E, c1)) continue;
518 const int k_end = hs->tab[j].beg + hs->tab[j].cnt;
519 for (int k = hs->tab[j].beg; k < k_end; k++) {
520 const int c2 = hs->cols[k];
521 if (c2 <= col_min || c2 == c1 || bit_get(t->E, c2)) continue;
522 if (!cc_sum2_equal(t->S, sw, t->mHT, c1, c2)) continue; /* a hash collision */
523 /* the checks at the entry of the leaf level */
524 if (atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed)) return 0;
525 const int cur_dmax = atomic_load_explicit(&ctx->dmax, memory_order_relaxed);
526 if (cur_dmax > 0 && ctx->p->dW >= 0 && minint(t->w_limit, cur_dmax + ctx->p->dW) < t->w + 2) return 0;
527 if (cc_walk_found(t, c1, c2)) return 1;
528 }
529 }
530 return 0;
531}

533/* The CC recursion of start_CC_recurs_mt() for smax = 0, with the syndrome of the cluster as the bitset t->S (weight
534 * sw, smallest row r0, hash h, see cc_hash_t), and its columns as the bitset t->E: a column is added by flipping its
535 * rows in S (and back after the recursion) instead of merging sorted lists, and the weight of the new syndrome is
536 * known before. The clusters are the same, in the same order, and so are the codewords found (and their CC hit
537 * counts). The last two levels use hash lookups (cc_walk_last2()). Returns 1 if the run is to stop. */
538static int cc_walk(cc_walk_t * const t, const int sw, const int r0, const uint64_t h) {
539 distfork_ctx_t * const ctx = t->ctx;
540 if (atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed)) return 0;
541 const params_t * const p = ctx->p;
542 const int w = t->w;
543 int current_limit = t->w_limit;
544 const int cur_dmax = atomic_load_explicit(&ctx->dmax, memory_order_relaxed);
545 if (cur_dmax > 0 && p->dW >= 0) current_limit = minint(t->w_limit, cur_dmax + p->dW);
546 if (w >= current_limit) return 0;
547 const csr_t * const mH = t->mH;
548 const csr_t * const mHT = t->mHT;
549 /* the positions in mH of the columns of the row r0 to add: all of them, or one at a fixed level of the work unit */
550 int i1_beg = mH->p[r0], i1_end = mH->p[r0 + 1];
551 if (w <= t->nfix) {
552 i1_beg += t->fix[w - 1];
553 if (i1_beg >= i1_end) return 0;
554 i1_end = i1_beg + 1;
555 }
556 const int col_min = t->col_min;
557 uint64_t * const S = t->S;
558 uint64_t * const E = t->E;
559
560 if (w + 1 == current_limit) { /* leaf level: the columns c with h_c = S */
561 for (int i1 = i1_beg; i1 < i1_end; i1++) {
562 const int c = mH->i[i1];
563 if (c <= col_min) continue;
564 const int pb = mHT->p[c];
565 if (mHT->p[c + 1] - pb != sw) continue;
566 int k = 0;
567 while (k < sw && bit_get(S, mHT->i[pb + k])) k++;
568 if (k < sw || bit_get(E, c)) continue;
569 if (cc_walk_found(t, -1, c)) return 1;
570 }
571 return 0;
572 }
573 if (w + 2 == current_limit && w + 1 > t->nfix) return cc_walk_last2(t, sw, h, i1_beg, i1_end);
574
575 /* internal level */
576 const int max_reach = (current_limit - (w + 1)) * t->max_col_wt;
577 const uint64_t * const hc = t->hs->hc;
578 for (int i1 = i1_beg; i1 < i1_end; i1++) {
579 const int c1 = mH->i[i1];
580 if (c1 <= col_min) continue;
581 const int pb = mHT->p[c1], pe = mHT->p[c1 + 1];
582 if (sw - (pe - pb) > max_reach || bit_get(E, c1)) continue;
583 int nset = 0; /* the rows of c1 in S */
584 for (int k = pb; k < pe; k++) nset += bit_get(S, mHT->i[k]);
585 const int sw1 = sw + (pe - pb) - 2 * nset;
586 if (sw1 <= 0 || sw1 > max_reach) continue;
587 /* add c1: flip its rows; the smallest row of the new syndrome (r0 is in c1, all the other rows of S are larger) */
588 int rmin = t->m;
589 for (int k = pb; k < pe; k++) {
590 const int r = mHT->i[k];
591 if (!bit_get(S, r) && r < rmin) rmin = r;
592 bit_flip(S, r);
593 }
594 const int r1 = (rmin < r0) ? rmin : bits_first(S, r0 + 1, rmin);
595 bit_flip(E, c1);
596 t->cols[t->w++] = c1;
597 const int result = cc_walk(t, sw1, r1, h ^ hc[c1]);
598 t->w--;
599 bit_flip(E, c1);
600 for (int k = pb; k < pe; k++) bit_flip(S, mHT->i[k]);
601 if (result == 1) return 1;
602 }
603 return 0;
604}

606/* The CC work unit from the start column `col` (w_limit > 1, see cc_walk()) */
607static void cc_walk_unit(cc_walk_t * const t, const int col, const int w_limit, const int * const fix, const int nfix) {
608 const csr_t * const mHT = t->mHT;
609 const int pb = mHT->p[col], pe = mHT->p[col + 1];
610 for (int k = pb; k < pe; k++) bit_flip(t->S, mHT->i[k]);
611 bit_flip(t->E, col);
612 t->cols[0] = col;
613 t->w = 1;
614 t->col_min = (t->ctx->p->start_num > 0) ? -1 : col;
615 t->w_limit = w_limit;
616 t->fix = fix;
617 t->nfix = nfix;
618 cc_walk(t, pe - pb, mHT->i[pb], t->hs->hc[col]);
619 t->w = 0;
620 bit_flip(t->E, col);
621 for (int k = pb; k < pe; k++) bit_flip(t->S, mHT->i[k]);
622}

624/* Recursive CC worker function (interruptible), used with smax > 0 (otherwise, cc_walk() explores the same clusters).
625 * At the fixed levels w = err->wei <= nfix of a CC work unit (see `cc_pack()`), only the column at position fix[w-1]
626 * of the row is added (none if the row is shorter). */
627static int start_CC_recurs_mt(one_vec_t *err, one_vec_t *urr, one_vec_t * const syn[],
628 const int w_limit, const int max_col_wt,
629 const csr_t * const mH, const csr_t * const mHT,
630 worker_arg_t *warg, const int * const fix, const int nfix) {
631 distfork_ctx_t *ctx = warg->ctx;
632 if (atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed)) {
633 return 0;
634 }
635 params_t * const p = ctx->p;
636 const int w = err->wei;
637 int current_limit = w_limit;
638 int cur_dmax = atomic_load_explicit(&ctx->dmax, memory_order_relaxed);
639 if (cur_dmax > 0 && p->dW >= 0) {
640 current_limit = minint(w_limit, cur_dmax + p->dW);
641 }
642 if (w >= current_limit) {
643 return 0;
644 }
645
646 const one_vec_t * const syn_w = syn[w];
647 const int syn_w_wei = syn_w->wei;
648 const int row = syn_w->vec[0];
649 /* the positions in mH of the columns of the row to add: all of them, or one at a fixed level of the work unit */
650 int i1_beg = mH->p[row], i1_end = mH->p[row + 1];
651 if (w <= nfix) {
652 i1_beg += fix[w - 1];
653 if (i1_beg >= i1_end) return 0;
654 i1_end = i1_beg + 1;
655 }
656 const colmask_t * const mL = p->maskL; /* NULL for a classical code */
657 /* Clusters are grown only to columns larger than the start column, except with the expert `start`
658 * list, where clusters are unlimited (columns already in `err` are skipped via `one_ordered_search`). */
659 const int col_min = (p->start_num > 0) ? -1 : urr->vec[0];
660
661 /* Leaf level: w + 1 == current_limit */
662 if (w + 1 == current_limit) {
663 int max_leaf_swei = 0;
664 if (p->smax > 0 && current_limit < MAX_W) {
665 int cur_min = warg->min_swei[current_limit] - 1;
666 max_leaf_swei = (cur_min < p->smax) ? (cur_min > 0 ? cur_min : 0) : p->smax;
667 }
668
669 for (int i1 = i1_beg; i1 < i1_end; i1++) {
670 const int col = mH->i[i1];
671 if (col <= col_min) continue;
672
673 const int p_beg = mHT->p[col];
674 const int col_wt = mHT->p[col + 1] - p_beg;
675 if (abs(syn_w_wei - col_wt) > max_leaf_swei) continue;
676
677 int swei;
678 if (max_leaf_swei == 0) {
679 /* Only swei == 0 is of interest: check if mHT[col] == syn[w] */
680 if (mHT->i[p_beg] != row ||
681 mHT->i[p_beg + col_wt - 1] != syn_w->vec[col_wt - 1]) {
682 continue;
683 }
684 if (memcmp(syn_w->vec, &mHT->i[p_beg], (size_t)col_wt * sizeof(int)) != 0) {
685 continue;
686 }
687 if (one_ordered_search(err, col) != -1) continue;
688 swei = 0;
689 } else {
690 if (one_ordered_search(err, col) != -1) continue;
691 syn[w + 1]->wei = 0;
692 swei = one_csr_row_combine(syn[w + 1], syn_w, mHT, col);
693 if (swei > 0) {
694 if (swei <= max_leaf_swei) {
695 warg->min_swei[w + 1] = swei;
696 int cur_min = swei - 1;
697 max_leaf_swei = (cur_min < p->smax) ? (cur_min > 0 ? cur_min : 0) : p->smax;
698 }
699 continue;
700 }
701 }
702
703 /* swei == 0: insert col into err to verify against mL and record codeword */
704 int pos = one_ordered_ins(err, col);
705 int nz = (!mL) || colmask_syndrome_non_zero(mL, err->wei, err->vec);
706 if (nz && cc_record(ctx, warg, err)) {
707 one_ordered_pos_del(err, col, pos);
708 return 1;
709 }
710 one_ordered_pos_del(err, col, pos);
711 }
712 return 0;
713 }
714
715 /* Internal level: w + 1 < current_limit */
716 const int rem = current_limit - (w + 1);
717 int max_s_needed = 0;
718 if (p->smax > 0 && current_limit < MAX_W) { /* smax > 0 only without noscan and with dmin <= 1 (var_init) */
719 int cur_min = warg->min_swei[current_limit] - 1;
720 max_s_needed = (cur_min < p->smax) ? (cur_min > 0 ? cur_min : 0) : p->smax;
721 }
722 const int max_reach = rem * max_col_wt + max_s_needed;
723
724 for (int i1 = i1_beg; i1 < i1_end; i1++) {
725 const int col = mH->i[i1];
726 if (col > col_min) {
727 const int col_wt = mHT->p[col + 1] - mHT->p[col];
728 if (syn_w_wei - col_wt > max_reach) continue;
729
730 int pos = one_ordered_search(err, col);
731 if (pos == -1) {
732 syn[w + 1]->wei = 0;
733 int swei = one_csr_row_combine(syn[w + 1], syn_w, mHT, col);
734
735 if (p->smax && swei > 0 && swei <= p->smax && (w + 1 < MAX_W)) {
736 if (swei < warg->min_swei[w + 1]) {
737 warg->min_swei[w + 1] = swei;
738 }
739 }
740
741 if (swei > 0 && swei <= max_reach) {
742 urr->vec[w] = col;
743 urr->wei++;
744 pos = one_ordered_ins(err, col);
745 int result = start_CC_recurs_mt(err, urr, syn, w_limit, max_col_wt,
746 mH, mHT, warg, fix, nfix);
747 urr->wei--;
748 one_ordered_pos_del(err, col, pos);
749 if (result == 1) {
750 return 1;
751 }
752 }
753 }
754 }
755 }
756 return 0;
757}

759/* Add the thread time `*acc_t` of `*acc_n` completed RW steps to the totals of the RW timing model, and reset them */

761 if (*acc_n > 0) {
762 atomic_fetch_add_explicit(&ctx->rw_busy_ns, (long long)(*acc_t * 1e9), memory_order_relaxed);
763 atomic_fetch_add_explicit(&ctx->rw_timed_steps, *acc_n, memory_order_relaxed);
764 }
765 *acc_t = 0.0;
766 *acc_n = 0;
767}

770 * and right away for the first step of a batch (`first`), so that the coordinator gets the step time early */
771static inline void rw_time_step(distfork_ctx_t *ctx, double *t_last, double *acc_t, int *acc_n, const bool first) {

775 (*acc_n)++;
776 if (first || *acc_t >= 0.002) rw_time_publish(ctx, acc_t, acc_n);
777}

785/* Record the RW codeword `ee` of weight `cnt` (sorted indices): add it to the hash (or count one more hit), update
786 * the upper bound dmax and the stop flags (`ksub` > 0 only for the message) */
787static void rw_record_codeword(distfork_ctx_t * const ctx, const rci_t * const ee, const int cnt, const int tid,
788 const int ksub) {
789 params_t * const p = ctx->p;
790 /* with outC or maxC, codewords are collected: RW does not end the run when the bounds coincide or on wmin (in
791 * method 3, the coordinator then runs the CC rounds which enumerate the codewords) */
792 const bool collecting = (p->outC != NULL) || (p->maxC > 0);
793 pthread_mutex_lock(&ctx->cw_mutex);
794 p->codewords = codeword_add_maybe(p, ee, cnt);
795 ctx_update_feed_limit(ctx);
796 const int best = p->min_w; /* a codeword lighter than min_w is always added */
797 const int old_dmax = atomic_load(&ctx->dmax);
798 if (old_dmax == 0 || best < old_dmax) {
799 atomic_store(&ctx->dmax, best);
800 if (p->debug & DBG_PROGRESS) {
801 int num_rw = (p->method == 1) ? ctx->num_threads
802 : minint(ctx->rw_threads, ctx->num_threads - atomic_load(&ctx->cc_target_workers));
803 if (num_rw < 1) num_rw = 1;
804 if (ksub > 0) {
805 fprintf(stderr, "# [thread %d] RW (ksub=%d) found new upper bound cw of weight %d (using %d RW threads)\n",
806 tid, ksub, best, num_rw);
807 } else {
808 fprintf(stderr, "# [thread %d] RW found new upper bound cw of weight %d (using %d RW threads)\n",
809 tid, best, num_rw);
810 }
811 }
812 if ((p->debug & DBG_CODEWORDS) && cnt == best) {
813 char prefix[48];
814 snprintf(prefix, sizeof(prefix), "# [thread %d] RW codeword", tid);
815 print_codeword_support(stderr, prefix, ee, cnt, p->debug);
816 }
817 const int cur_dmin = atomic_load(&ctx->dmin);
818 if (!collecting && cur_dmin > 0 && best <= cur_dmin) {
819 ctx_stop(ctx, STOP_BOUNDS);
820 }
821 }
822 if (!collecting && p->wmin > 0 && best <= p->wmin) {
823 ctx_stop(ctx, STOP_WMIN);
824 }
825 if (codeword_maxc_reached(p)) {
826 ctx_stop(ctx, STOP_MAXC);
827 }
828 if (check_min_hits_convergence(p)) {
829 if ((p->debug & DBG_PROGRESS) && !atomic_load(&ctx->stop_flag) && !atomic_load(&ctx->rw_stop_flag)) {
830 cw_hit_stats_t st;
831 codeword_hit_stats(p, &st);
832 fprintf(stderr,
833 "# RW convergence reached: <n>=%.2f >= min_hits=%d average hits per codeword (%lld cws of weight "
834 "%d..%d; w=%d: %lld cws, %lld hits), P_fail ~ exp(-<n>) = %.2g\n",
835 st.avg, p->min_hits, st.set_cws, st.w_lo, st.w_hi, st.w_lo, st.min_cws, st.min_hits, exp(-st.avg));
836 }
837 if (p->method == 1) {
838 ctx_stop(ctx, STOP_RW_CONVERGED);
839 } else {
840 atomic_store(&ctx->rw_stop_flag, true);
841 }
842 }
843 pthread_mutex_unlock(&ctx->cw_mutex);
844}

847static inline void rw_row_xor(word * __restrict dst, const word * __restrict src, const wi_t width) {
848 for (wi_t w = 0; w < width; w++) dst[w] ^= src[w];

855#define RW_K_MAX 8 /* at most 2^8 rows per table of rw_eliminate() */
856#define RW_NT_MAX 8 /* at most 8 tables, i.e., 64 pivots per pass over the rows */

858/* Parameters of rw_eliminate() for a matrix of `rows` x `width` words: the number of tables `*nt` and of pivots per
859 * table `*k` (column by column: *nt = 0, below 128 rows). For wide rows, the tables are kept at about 2 MB or less
860 * (cache): fewer rows per table, more tables. Measured (single thread): K = 6 with 4 tables is 1.7x to 2.2x faster
861 * than column by column for 648 to 2736 rows (circuit DEMs, a 1500 x 3000 LDPC code), also with 16 threads; between
862 * 128 and 512 rows, K = 5 with 4 tables is 0.75x to 1.4x as fast, depending on the matrix (see rw_elim_t). */
863static void rw_block_params(const int rows, const wi_t width, int * const k, int * const nt) {
864 *k = 0;
865 *nt = 0;
866 if (rows < 128) return;
867 *k = (rows < 512) ? 5 : 6;
868 *nt = 4;
869 while (*k > 3 && ((size_t)*nt << *k) * (size_t)width * sizeof(word) > ((size_t)2 << 20)) {
870 (*k)--;
871 if (*nt + 2 <= RW_NT_MAX) *nt += 2;
872 }
873}

875/* RW elimination of a thread: the tables and parameters of rw_eliminate(), and the choice between it and the column-by-
876 * column elimination for 128 to 512 rows, where the blocked one is not always faster: the steps 1..2*RW_TUNE_STEPS
877 * (after the first one, from the sparse H) alternate between the two eliminations, which give the same results, and
878 * the faster one (in total) is used for the following steps */
879#define RW_TUNE_STEPS 4
880typedef struct {
881 word *table; /* the tables of rw_eliminate(): nt << k rows of tstride words, row 0 of each table zero */
882 size_t tstride;
883 int k, nt; /* pivots per table, number of tables (0: column by column), see rw_block_params() */
884 int steps; /* RW steps completed while tuning */
885 double t[2]; /* time of the timed eliminations: column by column (0), blocked (1) */
886 int choice; /* -1: tuning, 0: column by column, 1: blocked */
887} rw_elim_t;

889/* RW elimination with the "Method of the Four Russians" (as in M4RI's echelonization, but for the scattered columns
890 * of a random permutation): Gauss-Jordan elimination of the columns order[0], order[1], ... in the rows
891 * 0..nrows_eff-1 of M, until nrows_eff pivots are found, in blocks of up to nt*K pivots.
892 * 1. The pivots of a block, in the order of the columns: a column is a pivot if a row r >= rank, reduced with the
893 * previous pivots, has a 1 in it. The block's pivot rows are kept reduced with each other, so that the reduced bit
894 * of a row is its bit plus the parity of its bits in the pivot columns of the block's pivot rows with a 1 in the
895 * column: the rows are searched without row operations, and only the new pivot row is reduced.
896 * 2. Tables of the 2^K combinations of K pivot rows each (built with one row XOR each, in Gray-code order; row 0 of
897 * each table is zero; table t at T + (t << K) * tstride, see rw_elim_t).
898 * 3. Every other row is reduced in one pass: for each table, the combination given by the row's bits in its pivot
899 * columns (the row stays in the L1 cache, while it is read and written only once from memory).
900 * The pivots, their rows, and the reduced matrix are the same as with gauss_one_rows() column by column, with about
901 * 1/(nt*K) of the passes over the rows of M. The pivot columns are stored in pivs[0..rank-1] and set in piv_mask.
902 * Returns the rank, or -1 if interrupted by a stop flag (checked every 16 columns, as before). (Not inlined, so that
903 * the column-by-column elimination in run_rw_steps() stays as compact as before for small matrices.) */
904__attribute__((noinline))
905static int rw_eliminate(distfork_ctx_t * const ctx, mzd_t * const M, const int nrows_eff, const int nvar,
906 const rci_t * const order, rci_t * const pivs, word * const piv_mask,
907 const rw_elim_t * const el) {
908 word * const T = el->table;
909 const size_t tstride = el->tstride;
910 const int K = el->k, nt = el->nt;
911 const wi_t width = M->width;
912 const int P = K * nt; /* pivots per block */
913 int bw[RW_K_MAX * RW_NT_MAX], bs[RW_K_MAX * RW_NT_MAX]; /* word and bit of the pivot columns of the block */
914 int rank = 0;
915 int i = 0;
916 while (i < nvar && rank < nrows_eff) {
917 const int r0 = rank; /* the first pivot row of the block */
918 int kk = 0;
919 /* 1. up to P pivots */
920 while (kk < P && i < nvar && rank < nrows_eff) {
921 if ((i & 15) == 15 && (atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed) ||
922 atomic_load_explicit(&ctx->rw_stop_flag, memory_order_relaxed))) {
923 return -1;
924 }
925 const int col = order[i++];
926 const int cw = col >> 6, cb = col & 63;
927 uint64_t mask = 0; /* the block's pivot rows with a 1 in the column */
928 for (int l = 0; l < kk; l++) mask |= (uint64_t)((mzd_row(M, r0 + l)[cw] >> cb) & 1) << l;
929 int found = -1;
930 for (int r = rank; r < nrows_eff; r++) {
931 const word * const row = mzd_row(M, r);
932 word b = (row[cw] >> cb) & 1;
933 for (uint64_t m2 = mask; m2; m2 &= m2 - 1) {
934 const int l = __builtin_ctzll(m2);
935 b ^= (row[bw[l]] >> bs[l]) & 1;
936 }
937 if (b) {
938 found = r;
939 break;
940 }
941 }
942 if (found < 0) continue; /* not a pivot: the column depends on the previous ones */
943 if (found != rank) mzd_row_swap(M, rank, found);
944 word * const prow = mzd_row(M, rank);
945 for (int l = 0; l < kk; l++) { /* reduce the new pivot row with the block's pivot rows */
946 if ((prow[bw[l]] >> bs[l]) & 1) rw_row_xor(prow, mzd_row(M, r0 + l), width);
947 }
948 for (uint64_t m2 = mask; m2; m2 &= m2 - 1) /* and clear the column in them */
949 rw_row_xor(mzd_row(M, r0 + __builtin_ctzll(m2)), prow, width);
950 bw[kk] = cw;
951 bs[kk] = cb;
952 kk++;
953 pivs[rank] = col;
954 piv_mask[cw] |= (word)1 << cb;
955 rank++;
956 }
957 if (kk == 0) break; /* no more columns */
958 /* 2. the tables: row g of table t is the sum of the pivot rows r0 + t*K + l of the bits l of g */
959 const int ntab = (kk + K - 1) / K;
960 for (int t = 0; t < ntab; t++) {
961 word * const Tt = T + ((size_t)t << K) * tstride;
962 const int kt = (kk - t * K < K) ? kk - t * K : K;
963 for (unsigned g = 1; g < (1u << kt); g++) {
964 const unsigned gc = g ^ (g >> 1), gp = (g - 1) ^ ((g - 1) >> 1); /* Gray codes of g and g-1 */
965 rw_row_xor3(Tt + gc * tstride, Tt + gp * tstride, mzd_row(M, r0 + t * K + __builtin_ctz(g)), width);
966 }
967 }
968 /* 3. all other rows, in one pass */
969 for (int r = 0; r < nrows_eff; r++) {
970 if (r == r0) {
971 r = rank - 1; /* (the pivot rows of the block) */
972 continue;
973 }
974 word * const row = mzd_row(M, r);
975 unsigned x[RW_NT_MAX];
976 for (int t = 0; t < ntab; t++) { /* (the bits first: the XORs change the row's bits in the pivot columns) */
977 const int lo = t * K, hi = (lo + K < kk) ? lo + K : kk;
978 unsigned xt = 0;
979 for (int l = lo; l < hi; l++) xt |= (unsigned)((row[bw[l]] >> bs[l]) & 1) << (l - lo);
980 x[t] = xt;
981 }
982 for (int t = 0; t < ntab; t++) {
983 if (x[t]) rw_row_xor(row, T + (((size_t)t << K) + x[t]) * tstride, width);
984 }
985 }
986 }
987 return rank;
988}

990/* Run RW batch of the global steps step0 .. step0+n_steps-1; returns the number of completed steps. `el`: the thread's
991 * RW elimination (blocked or column by column, see rw_elim_t) */
992static int run_rw_steps(distfork_ctx_t *ctx, long step0, int n_steps,
993 mzd_t *mH, mzd_t *mHT, rci_t *ee,
994 mzp_t *perm, mzp_t *pivs, word *piv_mask, rw_elim_t *el,
995 int *eff_nrows_ptr,
996 int *visited_cols, int *visited_checks, int *col_queue,
997 int *visit_marker, uint64_t *rng_state, int tid) {
998 params_t * const p = ctx->p;
999 const colmask_t * const maskL = p->maskL; /* NULL for a classical code */
1000 const int nvar = p->spaH->cols;
1001 const int classical = p->classical;
1002 const int kwin = p->kwin;
1003 const int win_mode = p->win_mode;
1004 int eff_nrows = *eff_nrows_ptr;
1005 int n_done = 0;
1006 long n_unif = 0; /* completed steps with uniform random permutations (for the information-set estimate) */
1007 double t_last = get_time_sec(), acc_t = 0.0;
1008 int acc_n = 0;
1009
1010 for (int step = 0; step < n_steps; step++) {
1011 if (atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed) ||
1012 atomic_load_explicit(&ctx->rw_stop_flag, memory_order_relaxed)) {
1013 break;
1014 }
1015
1016 int eff_kwin = kwin;
1017 int eff_win_mode = win_mode;
1018 if (eff_kwin == 0 && nvar >= 500 && ((step0 + step) & 1) == 1) { /* hybrid: every other global step */
1019 eff_kwin = minint(512, (nvar * 3) / 4);
1020 eff_win_mode = 1;
1021 }
1022 const bool uniform = !(eff_kwin > 0 && eff_kwin < nvar);
1023
1024 if (eff_kwin > 0 && eff_kwin < nvar) {
1025 int seed_col = rand_uniform_thread(nvar, rng_state);
1026 localized_window_perm(perm, nvar, seed_col, eff_kwin, eff_win_mode,
1027 p->spaH, ctx->mHT_cc,
1028 visited_cols, visited_checks, col_queue,
1029 visit_marker, rng_state);
1030 } else {
1031 pivs = mzp_rand_thread(pivs, nvar, rng_state);
1032 mzp_set_ui(perm, 1);
1033 perm = perm_p_trans(perm, pivs, 0);
1034 }
1035
1036 memset(piv_mask, 0, mH->width * sizeof(word));
1037 /* the elimination (see rw_elim_t): column by column, or blocked */
1038 int timed = -1; /* tuning: time this step's elimination, column by column (0) or blocked (1) */
1039 if (el->choice < 0 && el->steps > 0) timed = el->steps & 1;
1040 const double t_elim = (timed >= 0) ? get_time_sec() : 0.0;
1041 int rank = 0;
1042 if (el->choice == 0 || timed == 0) {
1043 for (int i = 0; i < nvar && rank < eff_nrows; i++) {
1044 /* a step on a large matrix can take many seconds: check the stop flags also every 16 columns */
1045 if ((i & 15) == 15 && (atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed) ||
1046 atomic_load_explicit(&ctx->rw_stop_flag, memory_order_relaxed))) {
1047 rank = -1;
1048 break;
1049 }
1050 int col = perm->values[i];
1051 if (gauss_one_rows(mH, col, rank, eff_nrows)) {
1052 pivs->values[rank++] = col;
1053 piv_mask[col >> 6] |= (word)1 << (col & 63);
1054 }
1055 }
1056 } else {
1057 rank = rw_eliminate(ctx, mH, eff_nrows, nvar, perm->values, pivs->values, piv_mask, el);
1058 }
1059 /* abandon the step: `eff_nrows` is kept, as rows rank..eff_nrows-1 are not processed (same row space) */
1060 if (rank < 0) break;
1061 if (el->choice < 0) {
1062 if (timed >= 0) el->t[timed] += get_time_sec() - t_elim;
1063 if (++el->steps > 2 * RW_TUNE_STEPS) {
1064 el->choice = (el->t[1] < el->t[0]) ? 1 : 0;
1065 if (tid == 0 && (p->debug & DBG_TIMING)) {
1066 fprintf(stderr, "# [thread 0] RW elimination: %s (blocked: %.2fx the time of column by column)\n",
1067 el->choice ? "blocked" : "column by column", el->t[1] / (el->t[0] > 0.0 ? el->t[0] : 1e-9));
1068 }
1069 }
1070 }
1071 eff_nrows = rank;
1072 *eff_nrows_ptr = eff_nrows;
1073 if (atomic_load_explicit(&ctx->rw_rank, memory_order_relaxed) != rank) { /* rank of H (the same in every step) */
1074 atomic_store_explicit(&ctx->rw_rank, rank, memory_order_relaxed);
1075 }
1076
1077 mzd_transpose(mHT, mH);
1078
1079 const int active_width = (rank + 63) >> 6;
1080 const word last_mask = (rank & 63) ? (((word)1 << (rank & 63)) - 1) : ~(word)0; /* bits of rows < rank */
1081 for (int col = 0; col < nvar; col++) {
1082 if ((piv_mask[col >> 6] >> (col & 63)) & 1) continue;
1083 const int limit = rw_cw_limit(ctx);
1084 word *rawrow = mzd_row(mHT, col);
1085 /* the weight 1 + (the number of pivot rows with a 1 in column col) first: word-wise popcount, which ends as soon
1086 * as the weight reaches the limit (most candidates are much heavier) */
1087 int wt = 1;
1088 for (int iw = 0; iw < active_width - 1 && wt < limit; iw++) wt += __builtin_popcountll(rawrow[iw]);
1089 if (wt < limit && active_width > 0) wt += __builtin_popcountll(rawrow[active_width - 1] & last_mask);
1090 if (wt >= limit) continue;
1091
1092 int cnt = 0;
1093 ee[cnt++] = col;
1094 rci_t j = -1;
1095 while (cnt < wt) {
1096 j = nextelement(rawrow, active_width, j);
1097 if (j == -1 || j >= rank) break;
1098 ee[cnt++] = pivs->values[j++];
1099 }
1100
1101 /* the check L c != 0 needs no sorted support: only the codewords recorded are sorted (hash key) */
1102 if (classical || colmask_syndrome_non_zero(maskL, cnt, ee)) {
1103 rci_quick_sort(ee, cnt);
1104 rw_record_codeword(ctx, ee, cnt, tid, 0);
1105 }
1106 }
1107 atomic_fetch_add(&ctx->rw_steps_completed, 1);
1108 n_done++;
1109 if (uniform) n_unif++;
1110 rw_time_step(ctx, &t_last, &acc_t, &acc_n, n_done == 1);
1111 }
1112 rw_time_publish(ctx, &acc_t, &acc_n);
1113 atomic_fetch_add_explicit(&ctx->rw_unif_steps, n_unif, memory_order_relaxed);
1114 return n_done;
1115}

1117/* After a completed RW step, only the first `rank` = rank(H) rows of the reduced thread-local matrix `*mH` are non-zero
1118 * (they span the rows of H), and the later steps eliminate only these rows: keep just these rows, so that the
1119 * transposition in each RW step skips the zero rows (for H with redundant rows) */
1120static void rw_shrink_matrices(mzd_t **mH, mzd_t **mHT, const int rank) {
1121 pthread_mutex_lock(&m4ri_mem_mutex);
1122 mzd_t * const top = mzd_init_window(*mH, 0, 0, rank, (*mH)->ncols);
1123 mzd_t * const small = mzd_copy(NULL, top);
1124 mzd_free_window(top);
1125 mzd_free(*mH);
1126 *mH = small;
1127 mzd_free(*mHT);
1128 *mHT = mzd_init(small->ncols, rank);
1129 pthread_mutex_unlock(&m4ri_mem_mutex);
1130}

1132/* Whether row `r` of N has a non-zero entry in a column of the current RW window (visited_cols[j] == marker) */
1133static inline bool ksub_row_in_window(const mzd_t * const N, const int r, const int nvar,
1134 const int * const visited_cols, const int marker) {
1135 const word * const raw_n = mzd_row_cons(N, r);
1136 for (int j = nextelement(raw_n, N->width, -1); j >= 0 && j < nvar; j = nextelement(raw_n, N->width, j + 1)) {
1137 if (visited_cols[j] == marker) return true;
1138 }
1139 return false;
1140}

1142/* Run compact subspace RW batch (ksub > 0) of the global steps step0 .. step0+n_steps-1; returns the number of
1143 * completed steps. `nidx` is the thread's permutation of the row indices 0..nu-1 of N, used to sample ksub distinct
1144 * rows (partial Fisher-Yates shuffle) */
1145static int run_rw_steps_ksub(distfork_ctx_t *ctx, long step0, int n_steps,
1146 mzd_t *M_sub, rci_t *ee, mzp_t *perm, mzp_t *pivs, int *nidx,
1147 int *visited_cols, int *visited_checks, int *col_queue,
1148 int *visit_marker, uint64_t *rng_state, int tid) {
1149 params_t * const p = ctx->p;
1150 const colmask_t * const maskL = p->maskL; /* NULL for a classical code */
1151 const int nvar = p->spaH->cols;
1152 const int classical = p->classical;
1153 const int nu = ctx->nu;
1154 const int ksub = M_sub->nrows;
1155 const int kwin = p->kwin;
1156 const int win_mode = p->win_mode;
1157
1158 if (nu <= 0 || ksub <= 0 || !ctx->N_global) { /* not reached: no subspace to sample, count the steps as done */
1159 atomic_fetch_add(&ctx->rw_steps_completed, n_steps);
1160 return 0;
1161 }
1162
1163 int n_done = 0;
1164 double t_last = get_time_sec(), acc_t = 0.0;
1165 int acc_n = 0;
1166 pthread_rwlock_rdlock(&ctx->basis_rwlock);
1167 const mzd_t * const N = ctx->N_global;
1168
1169 for (int step = 0; step < n_steps; step++) {
1170 if (atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed) ||
1171 atomic_load_explicit(&ctx->rw_stop_flag, memory_order_relaxed)) {
1172 break;
1173 }
1174
1175 /* 1. Generate column permutation (localized window or uniform) */
1176 int eff_kwin = kwin;
1177 int eff_win_mode = win_mode;
1178 if (eff_kwin == 0 && nvar >= 500 && ((step0 + step) & 1) == 1) { /* hybrid: every other global step */
1179 eff_kwin = minint(512, (nvar * 3) / 4);
1180 eff_win_mode = 1;
1181 }
1182
1183 if (eff_kwin > 0 && eff_kwin < nvar) {
1184 int seed_col = rand_uniform_thread(nvar, rng_state);
1185 localized_window_perm(perm, nvar, seed_col, eff_kwin, eff_win_mode,
1186 p->spaH, ctx->mHT_cc,
1187 visited_cols, visited_checks, col_queue,
1188 visit_marker, rng_state);
1189 } else {
1190 pivs = mzp_rand_thread(pivs, nvar, rng_state);
1191 mzp_set_ui(perm, 1);
1192 perm = perm_p_trans(perm, pivs, 0);
1193 }
1194
1195 /* 2. Sample ksub distinct rows of N into M_sub: partial Fisher-Yates shuffle of the row indices (rows drawn
1196 * with replacement would repeat and reduce the rank), with up to 4 redraws to prefer rows overlapping the window */
1197 const int marker = *visit_marker;
1198 const bool in_window = (eff_kwin > 0 && eff_kwin < nvar && visited_cols);
1199 for (int i = 0; i < ksub; i++) {
1200 int j = i + rand_uniform_thread(nu - i, rng_state);
1201 for (int attempt = 0; in_window && attempt < 4 && !ksub_row_in_window(N, nidx[j], nvar, visited_cols, marker);
1202 attempt++) {
1203 j = i + rand_uniform_thread(nu - i, rng_state);
1204 }
1205 SWAPINT(nidx[i], nidx[j]);
1206 mzd_copy_row(M_sub, i, N, nidx[i]);
1207 }
1208
1209 /* 3. Echelonize M_sub directly in L1/L2 cache using permuted column order */
1210 int rank = 0;
1211 for (int i = 0; i < nvar && rank < ksub; i++) {
1212 int col = perm->values[i];
1213 if (gauss_one(M_sub, col, rank)) {
1214 rank++;
1215 }
1216 }
1217
1218 /* 4. Extract candidate codewords directly from reduced rows (already in sorted order) */
1219 for (int ir = 0; ir < rank; ir++) {
1220 int cnt = 0;
1221 const int limit = rw_cw_limit(ctx);
1222
1223 word *rawrow = mzd_row(M_sub, ir);
1224 rci_t j = -1;
1225 while (cnt < limit) {
1226 j = nextelement(rawrow, M_sub->width, j);
1227 if (j == -1 || j >= nvar) break;
1228 ee[cnt++] = j++;
1229 }
1230
1231 if (cnt > 0 && cnt < limit && (classical || colmask_syndrome_non_zero(maskL, cnt, ee))) {
1232 rw_record_codeword(ctx, ee, cnt, tid, ksub);
1233 }
1234 }
1235 atomic_fetch_add(&ctx->rw_steps_completed, 1);
1236 n_done++;
1237 rw_time_step(ctx, &t_last, &acc_t, &acc_n, n_done == 1);
1238 }
1239 rw_time_publish(ctx, &acc_t, &acc_n);
1240
1241 pthread_rwlock_unlock(&ctx->basis_rwlock);
1242
1243 /* 5. Periodic adaptive basis refresh if enabled */
1244 if (p->refresh > 0 && !atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed) &&
1245 !atomic_load_explicit(&ctx->rw_stop_flag, memory_order_relaxed)) {
1246 long done = atomic_load_explicit(&ctx->rw_steps_completed, memory_order_relaxed);
1247 long next_ref = atomic_load_explicit(&ctx->next_refresh_step, memory_order_relaxed);
1248 if (done >= next_ref && next_ref > 0) {
1249 if (atomic_compare_exchange_strong(&ctx->next_refresh_step, &next_ref,
1250 done + p->refresh)) {
1251 int cw_wt = 0;
1252 pthread_mutex_lock(&ctx->cw_mutex);
1253 if (p->codewords && p->min_w < nvar) {
1254 cw_vec_t *cw, *tmp;
1255 HASH_ITER(hh, p->codewords, cw, tmp) {
1256 if (cw->weight == p->min_w) {
1257 cw_wt = cw->weight;
1258 for (int k = 0; k < cw_wt; k++) ee[k] = cw->arr[k];
1259 break;
1260 }
1261 }
1262 }
1263 pthread_mutex_unlock(&ctx->cw_mutex);
1264
1265 pthread_rwlock_wrlock(&ctx->basis_rwlock);
1266 pthread_mutex_lock(&m4ri_mem_mutex);
1267 refresh_nullspace_basis(ctx->N_global, cw_wt > 0 ? ee : NULL, cw_wt, rng_state);
1268 pthread_mutex_unlock(&m4ri_mem_mutex);
1269 pthread_rwlock_unlock(&ctx->basis_rwlock);
1270 }
1271 }
1272 }
1273 return n_done;
1274}

1276/* Worker thread main loop */
1277static void *worker_thread_func(void *arg) {

1279 distfork_ctx_t *ctx = warg->ctx;

1283 const bool enable_rw = ((ctx->p->method & 1) != 0) && (tid < ctx->rw_threads);
1284 const bool use_ksub = enable_rw && (ctx->p->ksub > 0);
1285 const int ksub_eff = (use_ksub && ctx->nu > 0) ? minint(ctx->p->ksub, ctx->nu) : 0;
1286 /* (warg->min_swei is initialized by the main thread before the workers are launched) */
1287

1293 rci_t *ee = NULL;
1294 mzp_t *perm = NULL;
1295 mzp_t *pivs = NULL;
1296 word *piv_mask = NULL;
1297 rw_elim_t rw_el = {NULL, 0, 0, 0, 0, {0.0, 0.0}, 0}; /* the RW elimination of this thread */
1298 int eff_nrows = ctx->p->spaH->rows;
1299 int *visited_cols = NULL;
1300 int *visited_checks = NULL;
1301 int *col_queue = NULL;
1302 int visit_marker = 0;
1303 /* RNG stream of this thread: a hashed starting point of splitmix64 (consecutive starting points would give
1304 * shifted copies of one stream) */
1305 uint64_t rng_state = mix64((uint64_t)ctx->p->seed ^ mix64((uint64_t)tid + 0x517cc1b727220a95ULL));
1306
1307 if (enable_rw) {
1308 ee = malloc((nvar + 2) * sizeof(rci_t));
1309 perm = safe_mzp_init(nvar);
1310 pivs = safe_mzp_init(nvar);
1311 if (ctx->p->kwin > 0 || nvar >= 500) {
1312 visited_cols = calloc(nvar, sizeof(int));
1313 visited_checks = calloc(ctx->p->spaH->rows, sizeof(int));
1314 col_queue = calloc(nvar, sizeof(int));
1315 }

1317 if (ksub_eff > 0) {
1318 M_sub = safe_mzd_init(ksub_eff, nvar);

1320 if (!nidx) ERROR("memory allocation");
1321 for (int i = 0; i < ctx->nu; i++) nidx[i] = i;
1322 }
1323 } else {
1324 mH = safe_mzd_from_csr(NULL, ctx->p->spaH);

1326 piv_mask = calloc(mH->width, sizeof(word));

1330 rw_el.choice = (rw_el.nt == 0) ? 0 : ((mH->nrows >= 512) ? 1 : -1); /* (tuning between 128 and 512 rows) */
1331 const size_t tbytes = (rw_el.nt > 0 ? ((size_t)rw_el.nt << rw_el.k) : 1) * rw_el.tstride * sizeof(word);
1332 rw_el.table = aligned_alloc(64, tbytes);

1334 memset(rw_el.table, 0, tbytes);
1335 }
1336 }

1338 /* Thread-local CC memory (only for CC, method 2 or 3) */
1339 const int wmax_alloc = MAX_W - 1;
1340 one_vec_t *err = NULL, *urr = NULL, **syn = NULL;
1341 cc_walk_t walk; /* the CC walk with bitset syndromes (smax = 0: ctx->cc_hash, see cc_walk()) */
1342 memset(&walk, 0, sizeof(walk));
1343 if (ctx->p->method >= 2) {
1344 err = calloc(1, sizeof(one_vec_t) + sizeof(int) * (wmax_alloc + 2));

1350 if (!syn[i]) ERROR("memory allocation");
1351 }

1353 walk.ctx = ctx;
1354 walk.warg = warg;
1355 walk.mH = ctx->p->spaH;
1356 walk.mHT = ctx->mHT_cc;
1357 walk.hs = ctx->cc_hash;
1358 walk.m = ctx->p->spaH->rows;
1359 walk.max_col_wt = ctx->max_col_W;
1360 walk.S = calloc((size_t)(walk.m / 64 + 2), sizeof(uint64_t));
1361 walk.E = calloc((size_t)(nvar / 64 + 2), sizeof(uint64_t));
1362 walk.cols = calloc(wmax_alloc + 3, sizeof(int));
1363 walk.cw = calloc(1, sizeof(one_vec_t) + sizeof(int) * (wmax_alloc + 3));
1364 if (!walk.S || !walk.E || !walk.cols || !walk.cw) ERROR("memory allocation");
1365 }

1369 if (ctx->timeout > 0.0 && (get_time_sec() - ctx->start_time >= ctx->timeout)) {
1370 ctx_stop(ctx, STOP_TIMEOUT);
1371 break;
1372 }

1375
1376 /* 1. Try to take CC work if CC is active (method 2 or 3) */
1377 if (ctx->p->method >= 2 && atomic_load_explicit(&ctx->cc_round_active, memory_order_acquire)) {
1378 int active = atomic_load_explicit(&ctx->cc_active_workers, memory_order_relaxed);
1379 int target = atomic_load_explicit(&ctx->cc_target_workers, memory_order_relaxed);
1380 /* A worker without RW work joins any CC round: method 2, a CC-only worker (tid >= rw_threads), RW stopped by
1381 * `min_hits`, or all RW steps already claimed by other workers (it would otherwise idle until the round ends) */
1382 if (!enable_rw || atomic_load_explicit(&ctx->rw_stop_flag, memory_order_relaxed) ||
1383 atomic_load_explicit(&ctx->rw_steps_started, memory_order_relaxed) >= ctx->total_rw_steps) {
1384 target = ctx->num_threads;
1385 }
1386 const long long next = atomic_load_explicit(&ctx->cc_next, memory_order_relaxed);
1387 if (active < target && cc_unpack_idx(next) <= cc_unit_last(ctx, cc_unpack_w(next), cc_unpack_lev(next))) {
1388 if (atomic_compare_exchange_weak(&ctx->cc_active_workers, &active, active + 1)) {
1389 const double t_join = get_time_sec();
1390 while (!atomic_load_explicit(&ctx->stop_flag, memory_order_relaxed)) {
1391 if (ctx->timeout > 0.0 && (get_time_sec() - ctx->start_time >= ctx->timeout)) {
1392 ctx_stop(ctx, STOP_TIMEOUT);
1393 break;
1394 }
1395 /* claim the next work unit together with the weight and the fixed levels of its round (see cc_pack()) */
1396 const long long claim = atomic_fetch_add_explicit(&ctx->cc_next, 1, memory_order_relaxed);
1397 const int w = cc_unpack_w(claim);
1398 const int lev = cc_unpack_lev(claim);
1399 const int unit = cc_unpack_idx(claim);
1400 if (unit > cc_unit_last(ctx, w, lev)) break;
1401 /* the start index of the unit, and its branches at the fixed levels: the digits of the unit in radix B */
1402 const int idx = unit / ctx->cc_R[lev];
1403 int fix[CC_SPLIT_MAX_LEVELS];
1404 for (int l = 0, k = unit - idx * ctx->cc_R[lev]; l < lev; l++, k /= ctx->cc_B) fix[l] = k % ctx->cc_B;
1405 /* with the expert `start` list, `idx` enumerates the listed start columns */
1406 const int col = (ctx->p->start_num > 0) ? ctx->p->start_list[idx] : idx;
1407
1408 err->vec[0] = urr->vec[0] = col;
1409 err->wei = urr->wei = 1;
1410 syn[1]->wei = 0;
1411 int swei = one_csr_row_combine(syn[1], syn[0], ctx->mHT_cc, col);
1412
1413 if (ctx->p->smax && swei > 0 && swei <= ctx->p->smax) {
1414 if (swei < warg->min_swei[1]) {
1415 warg->min_swei[1] = swei;
1416 }
1417 }
1418
1419 if (w > 1) {
1420 if (swei > 0 && swei <= (w - 1) * ctx->max_col_W + ctx->p->smax) {
1421 if (walk.S) {
1422 cc_walk_unit(&walk, col, w, fix, lev);
1423 } else {
1424 start_CC_recurs_mt(err, urr, syn, w, ctx->max_col_W,
1425 ctx->p->spaH, ctx->mHT_cc, warg, fix, lev);
1426 }
1427 }
1428 } else {
1429 if (!swei) {
1430 int nz = (!ctx->p->maskL) || colmask_syndrome_non_zero(ctx->p->maskL, 1, err->vec);
1431 if (nz) {
1432 params_t * const p = ctx->p;
1433 pthread_mutex_lock(&ctx->cw_mutex);
1434 p->codewords = codeword_add_maybe(p, err->vec, 1);
1435 ctx_update_feed_limit(ctx);
1436 if ((p->debug & DBG_CODEWORDS) && atomic_load(&ctx->dmax) != 1) {
1437 char prefix[48];
1438 snprintf(prefix, sizeof(prefix), "# [thread %d] CC codeword", tid);
1439 print_codeword_support(stderr, prefix, err->vec, 1, p->debug);
1440 }
1441 atomic_store(&ctx->cc_found_weight, 1);
1442 atomic_store(&ctx->cc_exact, true); /* weight 1 is always the exact distance */
1443 atomic_store(&ctx->dmin, 1);
1444 atomic_store(&ctx->dmax, 1);
1445 /* d=1 is known; with outC or maxC, the round goes on to collect all codewords of weight 1 */
1446 if (codeword_maxc_reached(p)) {
1447 ctx_stop(ctx, STOP_MAXC);
1448 } else if (!p->outC && p->maxC == 0) {
1449 ctx_stop(ctx, STOP_CC_FOUND);
1450 }
1451 pthread_mutex_unlock(&ctx->cw_mutex);
1452 }
1453 }
1454 }
1455 err->wei = urr->wei = 0;
1456 }
1457 /* CC thread time (added before leaving the round, so the coordinator sees it once the round is over) */
1458 atomic_fetch_add_explicit(&ctx->cc_busy_ns, (long long)((get_time_sec() - t_join) * 1e9),
1459 memory_order_relaxed);
1460 atomic_fetch_sub_explicit(&ctx->cc_active_workers, 1, memory_order_release);
1461 did_work = true;
1462 continue;
1463 }
1464 }
1465 }
1466
1467 /* 2. Try to take RW work if RW is active (method 1 or 3) */
1468 if (enable_rw && !atomic_load(&ctx->stop_flag) && !atomic_load(&ctx->rw_stop_flag)) {
1469 long cur_s = atomic_load(&ctx->rw_steps_started);
1470 if (cur_s < ctx->total_rw_steps) {
1471 long batch_size;
1472 if (ctx->p->chunk_size > 0) {
1473 batch_size = ctx->p->chunk_size;
1474 } else {
1475 int nvar = ctx->p->nvar;
1476 if (nvar < 500) {
1477 if (ctx->total_rw_steps >= 50000) batch_size = 500;
1478 else if (ctx->total_rw_steps >= 1000) batch_size = 250;
1479 else batch_size = 50;
1480 } else if (nvar < 5000) {
1481 if (ctx->total_rw_steps >= 10000) batch_size = 100;
1482 else batch_size = 50;
1483 } else {
1484 /* Large matrices (e.g. n >= 5000): keep chunk bounded to ~2-3s */
1485 if (ctx->total_rw_steps >= 10000) batch_size = 50;
1486 else batch_size = 25;
1487 }
1488 }
1489 /* Guided scheduling: a batch takes at most a quarter of the remaining steps per RW thread, so that no thread
1490 * monopolizes the steps and the RW threads finish together (no idle threads at the end of the steps) */
1491 if (ctx->rw_threads > 1) {
1492 const long guided = (ctx->total_rw_steps - cur_s + 4L * ctx->rw_threads - 1) / (4L * ctx->rw_threads);
1493 if (batch_size > guided) batch_size = (guided > 0) ? guided : 1;
1494 }
1495 long target_s = cur_s + batch_size;
1496 if (target_s > ctx->total_rw_steps) target_s = ctx->total_rw_steps;
1497 if (atomic_compare_exchange_weak(&ctx->rw_steps_started, &cur_s, target_s)) {
1498 int n_steps = (int)(target_s - cur_s);
1499 if (use_ksub) {
1500 if (ksub_eff > 0) {
1501 run_rw_steps_ksub(ctx, cur_s, n_steps, M_sub, ee, perm, pivs, nidx,
1502 visited_cols, visited_checks, col_queue,
1503 &visit_marker, &rng_state, tid);
1504 } else {
1505 atomic_fetch_add(&ctx->rw_steps_completed, n_steps);
1506 }
1507 } else {
1508 run_rw_steps(ctx, cur_s, n_steps, mH, mHT_rw, ee, perm, pivs,
1509 piv_mask, &rw_el, &eff_nrows, visited_cols, visited_checks,
1510 col_queue, &visit_marker, &rng_state, tid);
1511 if (eff_nrows > 0 && eff_nrows < mH->nrows) rw_shrink_matrices(&mH, &mHT_rw, eff_nrows);
1512 }
1513 did_work = true;
1514 continue;
1515 }
1516 }
1517 }
1518
1519 if (!did_work) {
1520 usleep(100);
1521 }
1522 }
1523
1524 if (enable_rw) {
1525 free(piv_mask);
1526 free(rw_el.table);
1527 safe_mzp_free(perm);
1528 safe_mzp_free(pivs);
1529 free(ee);
1530 free(visited_cols);
1531 free(visited_checks);
1532 free(col_queue);
1533 free(nidx);
1534 safe_mzd_free(M_sub);
1535 safe_mzd_free(mHT_rw);
1536 safe_mzd_free(mH);
1537 }
1538
1539 if (syn) {
1540 for (int i = 0; i <= wmax_alloc + 2; i++) free(syn[i]);
1541 free(syn);
1542 }
1543 free(err);
1544 free(urr);
1545 free(walk.S);
1546 free(walk.E);
1547 free(walk.cols);
1548 free(walk.cw);
1549
1550 return NULL;
1551}

1553/* Percentage of the start indices of the current (or last) CC round claimed by the workers (in work units) */
1554static double cc_round_claimed(distfork_ctx_t * const ctx) {
1555 const long long beg = (long long)ctx->round_beg * ctx->round_R;
1556 const long long end = ((long long)ctx->round_end + 1) * ctx->round_R; /* one past the last work unit */
1557 if (end <= beg) return 100.0;
1558 long long idx = cc_unpack_idx(atomic_load(&ctx->cc_next));
1559 if (idx > end) idx = end;
1560 if (idx < beg) idx = beg;
1561 return 100.0 * (double)(idx - beg) / (double)(end - beg);
1562}

1566 if (atomic_load(&ctx->rw_stop_flag)) return "converged";
1567 if (atomic_load(&ctx->rw_steps_completed) >= ctx->total_rw_steps) return "all steps done";
1568 return "stopped";
1569}

1571/* DBG_STATUS: periodic status line, after 1, 2, 4, ... s, then every 60 s (called by the coordinator only) */
1572static void ctx_status(distfork_ctx_t * const ctx) {
1573 params_t * const p = ctx->p;
1574 if (!(p->debug & DBG_STATUS)) return;
1575 const double t = get_time_sec() - ctx->start_time;
1576 if (t < ctx->status_next) return;
1577 ctx->status_next += ctx->status_dt;
1578 if (ctx->status_next <= t) ctx->status_next = t + ctx->status_dt;
1579 ctx->status_dt = (2.0 * ctx->status_dt < 60.0) ? 2.0 * ctx->status_dt : 60.0;
1580
1581 char cc_str[160] = "", rw_str[224] = "";
1582 if (p->method >= 2) {
1583 if (ctx->round_open) {
1584 int len = snprintf(cc_str, sizeof(cc_str), "; CC round w=%d: %.1f%% of start columns claimed, %.3gs",
1585 ctx->round_w, cc_round_claimed(ctx), get_time_sec() - ctx->round_t0);
1586 if (ctx->round_est > 0.0 && ctx->round_threads > 0 && len > 0 && len < (int)sizeof(cc_str))
1587 snprintf(cc_str + len, sizeof(cc_str) - len, " (predicted %.3gs on %d threads)",
1588 ctx->round_est / ctx->round_threads, ctx->round_threads);
1589 } else {
1590 snprintf(cc_str, sizeof(cc_str), "; CC %s", ctx->cc_idle ? ctx->cc_idle : "between rounds");
1591 }
1592 }
1593 if ((p->method & 1) && ctx->rw_threads > 0) {
1594 const long steps = atomic_load(&ctx->rw_steps_completed);
1595 const double rate = (t > ctx->status_t) ? (double)(steps - ctx->status_steps) / (t - ctx->status_t) : 0.0;
1596 ctx->status_steps = steps;
1597 ctx->status_t = t;
1598 const bool rw_on = !atomic_load(&ctx->rw_stop_flag) && steps < ctx->total_rw_steps;
1599 int len = snprintf(rw_str, sizeof(rw_str), "; RW: %ld of %ld steps (%.3g/s)%s%s", steps, ctx->total_rw_steps,
1600 rate, rw_on ? "" : ", ", rw_on ? "" : rw_end_state(ctx));
1601 if (p->min_hits > 0 && len > 0 && len < (int)sizeof(rw_str)) {
1602 cw_hit_stats_t st;
1603 pthread_mutex_lock(&ctx->cw_mutex);
1604 codeword_hit_stats(p, &st);
1605 pthread_mutex_unlock(&ctx->cw_mutex);
1606 if (st.set_cws > 0)
1607 snprintf(rw_str + len, sizeof(rw_str) - len, ", <n>=%.2f over %lld cws of weight %d..%d", st.avg,
1608 st.set_cws, st.w_lo, st.w_hi);
1609 }
1610 }
1611 fprintf(stderr, "# status %.1fs: bounds [%d, %d]%s%s\n", t, atomic_load(&ctx->dmin), atomic_load(&ctx->dmax),
1612 cc_str, rw_str);
1613}

1615/* Start the workers ctx->spawned .. n-1 (main thread only). If a thread cannot be created, the run goes on with the
1616 * workers already started, and no more are started. */
1617static void ctx_spawn(distfork_ctx_t * const ctx, const int n) {
1618 for (int i = ctx->spawned; i < n && !ctx->spawn_failed; i++) {
1619 const int err = pthread_create(&ctx->threads[i], NULL, worker_thread_func, &ctx->wargs[i]);
1620 if (err != 0) {
1621 if (i == 0) ERROR("cannot create a worker thread (error %d)", err);
1622 fprintf(stderr, "# Warning: cannot create worker thread %d (error %d), running with %d threads\n", i, err, i);
1623 ctx->spawn_failed = true;
1624 break;
1625 }
1626 ctx->spawned = i + 1;
1627 }
1628}

1630/* Method 2 (unless nothrottle=1) starts only CC_SPAWN_INITIAL workers, and the others for a CC round predicted to take
1631 * more than CC_SPAWN_WORK thread-seconds per started worker (or of unknown work), or once a round has run for
1632 * CC_SPAWN_TIME seconds: tiny runs avoid the start-up of many threads, longer runs use all of them (any worker can
1633 * join a CC round at any time) */
1634#define CC_SPAWN_INITIAL 4
1635#define CC_SPAWN_WORK 2e-3
1636#define CC_SPAWN_TIME 5e-3

1638/* Method 2: some workers not started yet */
1639static inline bool m2_can_spawn(const distfork_ctx_t * const ctx) {
1640 return ctx->spawned < ctx->num_threads && !ctx->spawn_failed;
1641}

1643/* Method 2: start the remaining workers during (or for) the CC round at weight w; `why`: for the message */

1649 fprintf(stderr, "# CC round w=%d: started %d more threads, %d in total (%s)\n", w, ctx->spawned - n0, ctx->spawned,
1650 why);
1651 }
1652}

1670/* Estimated work of the CC round at weight w in thread-seconds (the unit of `cc_time_per_weight`, i.e., wall time
1671 * times the number of CC threads): the measured work W(w-1) of the previous round times the growth factor
1672 * W(w-1)/W(w-2), clamped to [2, 10] (4 if W(w-2) is not known), also returned in `*growth`. Returns 0 if W(w-1)
1673 * has not been measured (the first round at a supplied dmin > 1). */
1674static double cc_work_estimate(const distfork_ctx_t * const ctx, const int w, double * const growth) {
1675 *growth = 4.0;
1676 if (w <= 1) return 1e-4; /* one syndrome per column */
1677 if (w > MAX_W) return 0.0;
1678 const double prev = ctx->cc_time_per_weight[w - 1];
1679 if (!(prev > 0.0)) return 0.0;
1680 if (w >= 3 && ctx->cc_time_per_weight[w - 2] > 1e-4) {
1681 const double g = prev / ctx->cc_time_per_weight[w - 2];
1682 *growth = (g < 2.0) ? 2.0 : ((g > 10.0) ? 10.0 : g);
1683 }
1684 return ((prev > 1e-4) ? prev : 1e-4) * (*growth);
1685}

1695/* Method 2 coordinator */
1696static void run_method2_coordinator(distfork_ctx_t *ctx) {
1697 const int wmax = ctx->p->wmax;
1698 const int w_start = ctx->p->noscan ? wmax : (ctx->p->dmin > 1 ? ctx->p->dmin : 1);
1699 /* with outC or maxC, CC does not stop at the first codeword but collects all codewords of a round */
1700 const bool collecting = (ctx->p->outC != NULL) || (ctx->p->maxC > 0);
1701 /* with outC and dW > 0, extra CC rounds export the codewords of weight up to d + dW */
1702 const int extra_w = (ctx->p->outC && ctx->p->dW > 0) ? ctx->p->dW : 0;
1703
1704 int w_limit = wmax;
1705 for (int w = w_start; ; w++) {
1706 if (atomic_load(&ctx->stop_flag)) break;
1707 /* The current upper bound dmax (given, read from finC, or found by CC) limits the rounds: since the CC kernel
1708 * caps the cluster weight at dmax + dW, a round w > dmax + extra_w would only enumerate the same clusters
1709 * again. Unless codewords are collected, the round w = dmax is not needed once dmin = dmax. */
1710 const int cur_dmax = atomic_load(&ctx->dmax);
1711 if (cur_dmax > 0) {
1712 if (!collecting && atomic_load(&ctx->dmin) >= cur_dmax) {
1713 if (ctx->p->debug & DBG_PROGRESS) {
1714 fprintf(stderr, "# bounds coincide: dmin = dmax = %d (CC round w=%d not needed)\n", cur_dmax, w);
1715 }
1716 ctx_stop(ctx, STOP_BOUNDS);
1717 break;
1718 }
1719 w_limit = minint(w_limit, cur_dmax + extra_w);
1720 }
1721 /* The stop target dstop (if below dmax): no rounds w >= dstop are needed, except the rounds w = dstop..dstop+dW
1722 * which collect codewords */
1723 const int dstop = ctx->p->dstop;
1724 if (dstop > 0 && (cur_dmax == 0 || dstop < cur_dmax)) {
1725 if (!collecting && atomic_load(&ctx->dmin) >= dstop) {
1726 if (ctx->p->debug & DBG_PROGRESS) {
1727 fprintf(stderr, "# lower bound dmin=%d reached dstop=%d (CC round w=%d not needed)\n",
1728 atomic_load(&ctx->dmin), dstop, w);
1729 }
1730 ctx_stop(ctx, STOP_DSTOP);
1731 break;
1732 }
1733 w_limit = minint(w_limit, collecting ? dstop + extra_w : dstop - 1);
1734 }
1735 if (w > w_limit) {
1736 ctx->stop_w = w - 1;
1737 ctx_stop(ctx, (dstop > 0 && atomic_load(&ctx->dmin) >= dstop) ? STOP_DSTOP : STOP_CC_DONE);
1738 break;
1739 }
1740
1741 double now = get_time_sec();
1742 double remaining_time = ctx->timeout - (now - ctx->start_time);
1743 if (ctx->timeout > 0.0 && remaining_time <= 0.0) {
1744 ctx_stop(ctx, STOP_TIMEOUT);
1745 break;
1746 }
1747
1748 /* With a timeout, a round predicted not to finish in time is started anyway: it can still end early with a
1749 * codeword of weight w, which gives the exact distance (all lower weights have been analyzed) */
1750 double growth;
1751 const double t_cc_work = cc_work_estimate(ctx, w, &growth); /* thread-seconds (0: unknown) */
1752 if (ctx->timeout > 0.0 && (ctx->p->debug & DBG_PROGRESS)) {
1753 const double t_cc_est = t_cc_work / ctx->num_threads; /* wall time */
1754 if (t_cc_est > remaining_time) {
1755 fprintf(stderr, "# CC for w=%d (est %.2fs) may not finish in the remaining time %.2fs, searching anyway "
1756 "(dmin=%d)\n", w, t_cc_est, remaining_time, atomic_load(&ctx->dmin));
1757 }
1758 }
1759
1760 const int beg = cc_round_beg(ctx->p);
1761 const int end = cc_round_end(ctx->p, w);
1762 /* all weights below w analyzed (or a supplied dmin): a codeword of weight w found is exact */
1763 const bool certified = (w == atomic_load(&ctx->dmin));
1764 /* the workers not started yet, for a longer round (or one of unknown work) */
1765 if (m2_can_spawn(ctx) && !(t_cc_work > 0.0 && t_cc_work <= CC_SPAWN_WORK * ctx->spawned)) {
1766 char why[64];
1767 if (t_cc_work > 0.0) snprintf(why, sizeof(why), "predicted %.3g thread-s", t_cc_work);
1768 else snprintf(why, sizeof(why), "work not known");
1769 m2_spawn_rest(ctx, w, why);
1770 }
1771 const int lev = cc_round_levels(ctx, w, beg, end, t_cc_work); /* fixed levels of the CC work units */
1772 const int last = cc_unit_last(ctx, w, lev);
1773
1774 atomic_store(&ctx->cc_target_workers, ctx->num_threads);
1775 const long long busy_ns0 = atomic_load(&ctx->cc_busy_ns);
1776 atomic_store(&ctx->cc_next, cc_pack(w, lev, beg * ctx->cc_R[lev]));
1777 atomic_store(&ctx->cc_round_active, 1);
1778
1779 double cc_start = get_time_sec();
1780 ctx->round_open = 1;
1781 ctx->round_w = w;
1782 ctx->round_beg = beg;
1783 ctx->round_end = end;
1784 ctx->round_R = ctx->cc_R[lev];
1785 ctx->round_threads = ctx->spawned;
1786 ctx->round_t0 = cc_start;
1787 ctx->round_est = t_cc_work;
1788
1789 if (ctx->p->debug & DBG_TIMING) {
1790 const char *note = (certified || atomic_load(&ctx->cc_exact)) ? "" : " (lower weights not scanned)";
1791 char units[64] = "";
1792 if (lev > 0) snprintf(units, sizeof(units), ", %d work units per start column", ctx->cc_R[lev]);
1793 if (ctx->p->start_num > 0) {
1794 fprintf(stderr, "# searching w=%d with %d CC threads, start list (%d columns)%s%s\n",
1795 w, ctx->spawned, ctx->p->start_num, units, note);
1796 } else {
1797 fprintf(stderr, "# searching w=%d with %d CC threads, columns [%d, %d]%s%s\n",
1798 w, ctx->spawned, beg, end, units, note);
1799 }
1800 }
1801
1802 bool round_completed = false;
1803 while (!atomic_load(&ctx->stop_flag)) {
1804 const double t = get_time_sec();
1805 if (ctx->timeout > 0.0 && (t - ctx->start_time >= ctx->timeout)) {
1806 ctx_stop(ctx, STOP_TIMEOUT);
1807 break;
1808 }
1809 if (cc_unpack_idx(atomic_load(&ctx->cc_next)) > last && atomic_load(&ctx->cc_active_workers) == 0) {
1810 round_completed = true;
1811 break;
1812 }
1813 if (m2_can_spawn(ctx) && t - cc_start > CC_SPAWN_TIME) {
1814 char why[64];
1815 snprintf(why, sizeof(why), "running for %.3fs", t - cc_start);
1816 m2_spawn_rest(ctx, w, why);
1817 }
1818 ctx_status(ctx);
1819 usleep(100);
1820 }
1821
1822 atomic_store(&ctx->cc_round_active, 0);
1823 if (round_completed) ctx->round_open = 0;
1824
1825 double cc_dur = get_time_sec() - cc_start;
1826 if (w < MAX_W) { /* CC work of the round in thread-seconds, as measured by the workers */
1827 ctx->cc_time_per_weight[w] = 1e-9 * (double)(atomic_load(&ctx->cc_busy_ns) - busy_ns0);
1828 if ((ctx->p->debug & DBG_TIMING) && round_completed) {
1829 char est_str[48] = "unknown";
1830 if (t_cc_work > 0.0) snprintf(est_str, sizeof(est_str), "%.3g thread-s", t_cc_work);
1831 fprintf(stderr, "# CC round w=%d work: %.3g thread-s measured, %s predicted (%.3fs on %d threads)\n", w,
1832 ctx->cc_time_per_weight[w], est_str, cc_dur, ctx->spawned);
1833 }
1834 }
1835
1836 int cw_found = atomic_load(&ctx->cc_found_weight);
1837 if (cw_found > 0 && certified && cw_found == w && !atomic_load(&ctx->cc_exact)) {
1838 atomic_store(&ctx->cc_exact, true);
1839 }
1840 const bool exact = (cw_found > 0) && atomic_load(&ctx->cc_exact);
1841 if (exact) {
1842 atomic_store(&ctx->dmin, cw_found);
1843 atomic_store(&ctx->dmax, cw_found);
1844 /* the distance is known: no rounds beyond w = cw_found + extra_w */
1845 w_limit = minint(w_limit, cw_found + extra_w);
1846 const bool more_rounds = round_completed && (w < w_limit);
1847 if (ctx->p->debug & DBG_PROGRESS) {
1848 if (w == cw_found && more_rounds) {
1849 fprintf(stderr,
1850 "# CC round w=%d finished in %.3fs (%d CC threads): found min-weight codewords "
1851 "(dmin=%d, continuing up to w=%d for dW=%d, total %lld cws)\n",
1852 w, cc_dur, ctx->spawned, cw_found, w_limit, ctx->p->dW, ctx_num_cws(ctx));
1853 } else if (w == cw_found) {
1854 fprintf(stderr, "# CC found min-weight codeword: d=%d (using %d CC threads, total %lld cws)\n",
1855 cw_found, ctx->spawned, ctx_num_cws(ctx));
1856 } else if (round_completed) {
1857 fprintf(stderr,
1858 "# CC round w=%d finished in %.3fs (%d CC threads): extra dW round completed "
1859 "(dmin=%d, total %lld cws)\n",
1860 w, cc_dur, ctx->spawned, cw_found, ctx_num_cws(ctx));
1861 }
1862 }
1863 if (!more_rounds) {
1864 /* (a stop reason recorded during the round, e.g., the timeout, is kept) */
1865 ctx_stop(ctx, collecting ? STOP_COLLECTED : STOP_CC_FOUND);
1866 break;
1867 }
1868 } else if (cw_found > 0) {
1869 /* noscan=1 without dmin=wmax: lower weights not scanned, the codeword only gives an upper bound */
1870 if (ctx->p->debug & DBG_PROGRESS) {
1871 fprintf(stderr,
1872 "# CC round w=%d finished in %.3fs (%d CC threads): found codeword of weight %d -> dmax=%d "
1873 "(dmin=%d not certified: lower weights not scanned)\n",
1874 w, cc_dur, ctx->spawned, cw_found, atomic_load(&ctx->dmax), atomic_load(&ctx->dmin));
1875 }
1876 ctx_stop(ctx, STOP_CC_FOUND);
1877 break;
1878 } else {
1879 if (!round_completed) {
1880 break;
1881 }
1882 if (certified) {
1883 /* Weight w analyzed without success */
1884 atomic_store(&ctx->dmin, w + 1);
1885 if (ctx->p->debug & DBG_PROGRESS) {
1886 fprintf(stderr, "# CC w=%d completed in %.3fs (%d CC threads): no codewords found -> dmin=%d\n",
1887 w, cc_dur, ctx->spawned, w + 1);
1888 }
1889 } else if (ctx->p->debug & DBG_PROGRESS) {
1890 fprintf(stderr, "# CC w=%d completed in %.3fs (%d CC threads): no codewords found "
1891 "(dmin=%d not raised: lower weights not scanned)\n",
1892 w, cc_dur, ctx->spawned, atomic_load(&ctx->dmin));
1893 }
1894 }
1895 }
1896}

1898/* Method 3: RW still runs (RW threads, RW steps not all completed, RW not stopped by min_hits) */
1899static inline bool m3_rw_running(distfork_ctx_t * const ctx) {
1900 return ctx->rw_threads > 0 && !atomic_load(&ctx->rw_stop_flag) &&
1901 atomic_load(&ctx->rw_steps_completed) < ctx->total_rw_steps;
1902}

1904/* Method 3: RW thread time per step, measured continuously by the RW workers (before the first step is completed, a
1905 * step takes at least the time elapsed since the start) */
1906static double m3_rw_step_time(distfork_ctx_t * const ctx) {
1907 const long n = atomic_load(&ctx->rw_timed_steps);
1908 if (n > 0) return 1e-9 * (double)atomic_load(&ctx->rw_busy_ns) / (double)n;
1909 const double elapsed = get_time_sec() - ctx->start_time;
1910 return (elapsed > 5e-5) ? elapsed : 5e-5;
1911}

1913/* Method 3: the largest CC weight needed: t-1 once the target t is known, i.e., the upper bound dmax or the stop
1914 * target dstop if smaller (with outC, t+dW: the rounds w = t..t+dW enumerate the codewords to export), otherwise wmax
1915 * (or n); at most MAX_W-2. (`dexp` only affects the thread split, see m3_plan_cc_threads().) */
1916static int m3_cc_target_w(const distfork_ctx_t * const ctx, const int cur_dmin, const int cur_dmax) {
1917 const params_t * const p = ctx->p;
1918 const int t = ctx_target(ctx, cur_dmax);
1919 int target;
1920 if (t > 0) {
1921 if (p->outC && (p->dW > 0 || cur_dmin >= t)) {
1922 target = t + (p->dW > 0 ? p->dW : 0);
1923 } else {
1924 target = t - 1;
1925 }
1926 } else {
1927 target = (p->wmax > 0) ? p->wmax : p->spaH->cols;
1928 }
1929 const int max_allowed_w = (p->wmax > 0) ? minint(p->wmax, MAX_W - 2) : (MAX_W - 2);
1930 return minint(target, max_allowed_w);
1931}

1933#define M3_DEADLINE_FRACTION 0.75 /* with a timeout, a round gets enough CC threads to finish in 3/4 of the time left */
1934#define M3_OVERRUN_FACTOR 1.5 /* a round taking 1.5 times (+20 ms) longer than predicted gets all threads */
1935
1936/* Method 3: number of CC threads for the round at weight w with the estimated work `est` (thread-seconds, 0 if not
1937 * known) and the growth factor `growth` (see cc_work_estimate()); the other threads run RW. Returns 0 if CC is
1938 * paused (`dexp`, below). */
1939static int m3_plan_cc_threads(distfork_ctx_t * const ctx, const int w, const double est, const double growth,
1940 const double remaining_time) {
1941 const int nthr = ctx->num_threads;
1942 if (nthr <= 1) return 1;
1943 /* RW finished, or all RW steps claimed: the RW workers join the round anyway */
1944 if (!m3_rw_running(ctx) || atomic_load(&ctx->rw_steps_started) >= ctx->total_rw_steps) return nthr;
1945 const int cur_dmin = atomic_load(&ctx->dmin);
1946 const int cur_dmax = atomic_load(&ctx->dmax);
1947 /* The round w = t-1 certifies dmin = t (t: dmax, or the stop target dstop if smaller; later rounds only collect
1948 * codewords). RW cannot improve this result, it could only find a codeword of weight w before CC does: all threads
1949 * run CC. */
1950 const int t = ctx_target(ctx, cur_dmax);
1951 if (t > 0 && (w >= t - 1 || cur_dmin >= t)) return nthr;
1952 /* `dexp` hint: as long as RW has not found any codeword, CC rounds w > dexp run only on the threads which cannot
1953 * run RW (none: CC pauses) */
1954 if (cur_dmax == 0 && ctx->dexp > 0 && w > ctx->dexp) return nthr - ctx->rw_threads;
1955
1956 int n_cc;
1957 if (!(est > 0.0)) {
1958 n_cc = nthr / 2; /* the first round at a supplied dmin: its work is not known */
1959 } else if (est < 0.005) {
1960 n_cc = (nthr >= 4) ? 2 : 1;
1961 } else {
1962 /* Split by the remaining work in thread-seconds: the CC rounds w, w+1, w+2 up to the horizon (dmax-1, or dexp
1963 * before RW has found a codeword), and the remaining RW steps (at most 2000 once dmax is known), but not more RW
1964 * work than the RW threads can do before the timeout */
1965 int horizon = m3_cc_target_w(ctx, cur_dmin, cur_dmax);
1966 if (cur_dmax == 0 && ctx->dexp > 0 && ctx->dexp < horizon) horizon = ctx->dexp;
1967 double t_cc = est, t_k = est;
1968 for (int k = w + 1; k <= horizon && k <= w + 2; k++) {
1969 t_k *= growth;
1970 t_cc += t_k;
1971 }
1972 const long steps_rem = ctx->total_rw_steps - atomic_load(&ctx->rw_steps_completed);
1973 const long eff_steps = (cur_dmax > 0 && steps_rem > 2000) ? 2000 : steps_rem;
1974 double t_rw = (double)eff_steps * m3_rw_step_time(ctx);
1975 if (ctx->timeout > 0.0 && t_rw > remaining_time * ctx->rw_threads) t_rw = remaining_time * ctx->rw_threads;
1976 n_cc = (int)round((double)nthr * t_cc / (t_cc + t_rw));
1977 if (n_cc < 1) n_cc = 1;
1978 if (n_cc > nthr - 1) n_cc = nthr - 1; /* keep at least one RW thread */
1979 }
1980 /* timeout: enough CC threads to finish the round in time */
1981 if (ctx->timeout > 0.0 && est > 0.0 && remaining_time > 0.0) {
1982 const double n_min = ceil(est / (M3_DEADLINE_FRACTION * remaining_time));
1983 if (n_min > n_cc) n_cc = (n_min < nthr) ? (int)n_min : nthr;
1984 }
1985 /* only `rw_threads` workers can run RW (memory / steps limits): the other workers always run CC */
1986 if (nthr - n_cc > ctx->rw_threads) n_cc = nthr - ctx->rw_threads;
1987 return n_cc;
1988}

1990/* Method 3: wait while RW runs, until the upper bound dmax differs from `dmax0`, RW ends, or the run stops; `why`:
1991 * the reason why no CC round runs, for the status line */
1992static void m3_wait_rw(distfork_ctx_t * const ctx, const int dmax0, const char * const why) {
1993 ctx->cc_idle = why;
1994 while (!atomic_load(&ctx->stop_flag) && m3_rw_running(ctx) && atomic_load(&ctx->dmax) == dmax0) {
1995 if (ctx->timeout > 0.0 && (get_time_sec() - ctx->start_time >= ctx->timeout)) {
1996 ctx_stop(ctx, STOP_TIMEOUT);
1997 break;
1998 }
1999 ctx_status(ctx);
2000 usleep(1000);
2001 }
2002 ctx->cc_idle = NULL;
2003}

2005/* Method 3 coordinator */
2006static void run_method3_coordinator(distfork_ctx_t *ctx) {
2007 const int nvar = ctx->p->spaH->cols;
2008 const int nthr = ctx->num_threads;
2009 int w = ctx->p->noscan ? ctx->p->wmax : (ctx->p->dmin > 1 ? ctx->p->dmin : 1);
2010
2011 int init_dmax = atomic_load(&ctx->dmax);
2012 int init_dmin = atomic_load(&ctx->dmin);
2013 if (init_dmax > 0 && init_dmin >= init_dmax && !ctx->p->outC) {
2014 ctx_stop(ctx, STOP_BOUNDS);
2015 return;
2016 }
2017
2018 /* No RW probe: the RW step time is measured continuously by the RW workers (m3_rw_step_time()). CC is never ended
2019 * while RW still runs: if no CC round can be started (CC done up to wmax, a round predicted to exceed the timeout,
2020 * or CC paused by `dexp`), the coordinator waits for RW and re-plans when dmax changes or RW ends. */
2021 int wait_msg_w = 0; /* weight w for which a waiting message was printed (once per round) */
2022 while (!atomic_load(&ctx->stop_flag)) {
2023 double now = get_time_sec();
2024 double remaining_time = (ctx->timeout > 0.0) ? (ctx->timeout - (now - ctx->start_time)) : 1e9;
2025 if (ctx->timeout > 0.0 && remaining_time <= 0.0) {
2026 ctx_stop(ctx, STOP_TIMEOUT);
2027 break;
2028 }
2029
2030 int cur_dmax = atomic_load(&ctx->dmax);
2031 int cur_dmin = atomic_load(&ctx->dmin);
2032 const bool rw_on = m3_rw_running(ctx);
2033
2034 /* Target cluster size for CC */
2035 const int target_cc_w = m3_cc_target_w(ctx, cur_dmin, cur_dmax);
2036
2037 if (cur_dmax > 0 && cur_dmin >= cur_dmax && w > target_cc_w) {
2038 /* Bracketing converged and all requested dW rounds completed */
2039 atomic_store(&ctx->dmin, cur_dmax);
2040 ctx_stop(ctx, (ctx->p->outC && w > cur_dmax) ? STOP_COLLECTED : STOP_BOUNDS);
2041 break;
2042 }
2043 if (ctx->p->dstop > 0 && cur_dmin >= ctx->p->dstop && w > target_cc_w) {
2044 /* The stop target dstop reached (with outC, after the rounds w = dstop..dstop+dW which collect codewords) */
2045 ctx_stop(ctx, STOP_DSTOP);
2046 break;
2047 }
2048
2049 if (w > target_cc_w) {
2050 /* CC done up to wmax: only RW can still lower dmax */
2051 if (!rw_on) {
2052 ctx->stop_w = target_cc_w;
2053 ctx_stop(ctx, STOP_CC_DONE);
2054 break;
2055 }
2056 if ((ctx->p->debug & DBG_TIMING) && wait_msg_w != w) {
2057 wait_msg_w = w;
2058 fprintf(stderr, "# CC done up to w=%d, waiting for RW (bounds [%d, %d])\n", target_cc_w, cur_dmin, cur_dmax);
2059 }
2060 m3_wait_rw(ctx, cur_dmax, "done up to wmax, waiting for RW");
2061 continue;
2062 }
2063
2064 /* Estimated CC work of the round (thread-seconds). A round predicted not to finish in time on all threads is not
2065 * started while RW runs; once RW has ended, the run ends. Without RW (steps=0), as in method 2, the round is
2066 * started anyway: it can still end early with a codeword of weight w (exact if w = dmin). */
2067 double growth;
2068 const double t_cc_est = cc_work_estimate(ctx, w, &growth);
2069 if (ctx->timeout > 0.0 && t_cc_est / nthr > remaining_time) {
2070 if (rw_on) {
2071 if ((ctx->p->debug & DBG_TIMING) && wait_msg_w != w) {
2072 wait_msg_w = w;
2073 fprintf(stderr, "# CC for w=%d (est %.2fs) exceeds remaining timeout %.2fs, devoting %d threads to RW\n",
2074 w, t_cc_est / nthr, remaining_time, ctx->rw_threads);
2075 }
2076 m3_wait_rw(ctx, cur_dmax, "round predicted to exceed the timeout, waiting for RW");
2077 continue;
2078 }
2079 if (ctx->rw_threads > 0) {
2080 if (ctx->p->debug & DBG_PROGRESS) {
2081 fprintf(stderr, "# CC for w=%d (est %.2fs) exceeds remaining timeout %.2fs, RW ended: terminating early "
2082 "(dmin=%d)\n", w, t_cc_est / nthr, remaining_time, cur_dmin);
2083 }
2084 ctx->stop_w = w;
2085 ctx_stop(ctx, STOP_CC_SLOW);
2086 break;
2087 }
2088 if ((ctx->p->debug & DBG_PROGRESS) && wait_msg_w != w) {
2089 wait_msg_w = w;
2090 fprintf(stderr, "# CC for w=%d (est %.2fs) may not finish in the remaining time %.2fs, searching anyway "
2091 "(dmin=%d)\n", w, t_cc_est / nthr, remaining_time, cur_dmin);
2092 }
2093 }
2094
2095 /* Calculate thread balancing */
2096 int n_cc = m3_plan_cc_threads(ctx, w, t_cc_est, growth, remaining_time);
2097 if (n_cc <= 0) {
2098 if ((ctx->p->debug & DBG_TIMING) && wait_msg_w != w) {
2099 wait_msg_w = w;
2100 fprintf(stderr, "# CC paused before w=%d > dexp=%d until RW finds a codeword (or ends)\n", w, ctx->dexp);
2101 }
2102 m3_wait_rw(ctx, cur_dmax, "paused (w > dexp) until RW finds a codeword");
2103 continue;
2104 }
2105 const int n_cc0 = n_cc;
2106 int n_rw = nthr - n_cc;
2107 const long steps_rem = rw_on ? ctx->total_rw_steps - atomic_load(&ctx->rw_steps_completed) : 0;
2108
2109 const int beg = cc_round_beg(ctx->p);
2110 const int end = cc_round_end(ctx->p, w);
2111 /* all weights below w analyzed (or a supplied dmin): a codeword of weight w found is exact */
2112 const bool certified = (w == atomic_load(&ctx->dmin));
2113 const int lev = cc_round_levels(ctx, w, beg, end, t_cc_est); /* fixed levels of the CC work units */
2114 const int last = cc_unit_last(ctx, w, lev);
2115
2116 atomic_store(&ctx->cc_target_workers, n_cc);
2117 const long long busy_ns0 = atomic_load(&ctx->cc_busy_ns);
2118 atomic_store(&ctx->cc_next, cc_pack(w, lev, beg * ctx->cc_R[lev]));
2119 atomic_store(&ctx->cc_round_active, 1);
2120
2121 if (ctx->p->debug & DBG_TIMING) {
2122 char est_str[64] = "unknown";
2123 char rw_str[64] = "";
2124 char units[64] = "";
2125 if (t_cc_est > 0.0) snprintf(est_str, sizeof(est_str), "%.3g thread-s", t_cc_est);
2126 if (rw_on) snprintf(rw_str, sizeof(rw_str), ", RW step %.3g s", m3_rw_step_time(ctx));
2127 if (lev > 0) snprintf(units, sizeof(units), ", %d work units per start column", ctx->cc_R[lev]);
2128 fprintf(stderr,
2129 "# CC round w=%d started: %d CC threads, %d RW threads "
2130 "(bounds [%d, %d], rem_rw=%ld, rem_time=%.2fs, est %s%s%s)\n",
2131 w, n_cc, n_rw, cur_dmin, cur_dmax, steps_rem, remaining_time, est_str, rw_str, units);
2132 }
2133
2134 double cc_start = get_time_sec();
2135 bool round_completed = false;
2136 int dmax_seen = cur_dmax;
2137 unsigned int poll = 0;
2138 ctx->round_open = 1;
2139 ctx->round_w = w;
2140 ctx->round_beg = beg;
2141 ctx->round_end = end;
2142 ctx->round_R = ctx->cc_R[lev];
2143 ctx->round_threads = n_cc;
2144 ctx->round_t0 = cc_start;
2145 ctx->round_est = t_cc_est;
2146
2147 while (!atomic_load(&ctx->stop_flag)) {
2148 const double t = get_time_sec();
2149 if (ctx->timeout > 0.0 && (t - ctx->start_time >= ctx->timeout)) {
2150 ctx_stop(ctx, STOP_TIMEOUT);
2151 break;
2152 }
2153 if (cc_unpack_idx(atomic_load(&ctx->cc_next)) > last && atomic_load(&ctx->cc_active_workers) == 0) {
2154 round_completed = true;
2155 break;
2156 }
2157 /* Re-plan during the round (only adding CC threads, workers do not leave a round): when RW finds a new upper
2158 * bound (e.g., this round now certifies dmin = dmax), and when the round takes much longer than predicted */
2159 if ((++poll & 15) == 0 && n_cc < nthr) {
2160 int n_new = n_cc;
2161 const char *why = "";
2162 const int d_now = atomic_load(&ctx->dmax);
2163 if (d_now != dmax_seen) {
2164 dmax_seen = d_now;
2165 const double rem = (ctx->timeout > 0.0) ? ctx->timeout - (t - ctx->start_time) : 1e9;
2166 n_new = m3_plan_cc_threads(ctx, w, t_cc_est, growth, rem);
2167 why = "new upper bound";
2168 }
2169 const bool paused = (d_now == 0 && ctx->dexp > 0 && w > ctx->dexp && m3_rw_running(ctx));
2170 if (!paused && t_cc_est > 0.0 && t - cc_start > M3_OVERRUN_FACTOR * t_cc_est / n_cc0 + 0.02) {
2171 n_new = nthr;
2172 why = "longer than predicted";
2173 }
2174 if (n_new > n_cc) {
2175 if (ctx->p->debug & DBG_TIMING) {
2176 fprintf(stderr, "# CC round w=%d: %d -> %d CC threads (%s, bounds [%d, %d], %.3fs into the round)\n",
2177 w, n_cc, n_new, why, atomic_load(&ctx->dmin), d_now, t - cc_start);
2178 }
2179 n_cc = n_new;
2180 n_rw = nthr - n_cc;
2181 ctx->round_threads = n_cc;
2182 atomic_store(&ctx->cc_target_workers, n_cc);
2183 }
2184 }
2185 ctx_status(ctx);
2186 usleep(100);
2187 }
2188
2189 atomic_store(&ctx->cc_round_active, 0);
2190 if (round_completed) ctx->round_open = 0;
2191
2192 double cc_dur = get_time_sec() - cc_start;
2193 if (w < MAX_W) {
2194 /* CC work of the round in thread-seconds, as measured by the workers: the number of CC threads may exceed
2195 * n_cc (CC-only workers, and RW workers that ran out of RW steps join the round) */
2196 ctx->cc_time_per_weight[w] = 1e-9 * (double)(atomic_load(&ctx->cc_busy_ns) - busy_ns0);
2197 if ((ctx->p->debug & DBG_TIMING) && round_completed) {
2198 char est_str[48] = "unknown";
2199 if (t_cc_est > 0.0) snprintf(est_str, sizeof(est_str), "%.3g thread-s", t_cc_est);
2200 fprintf(stderr, "# CC round w=%d work: %.3g thread-s measured, %s predicted (%.3fs, %d CC threads at the "
2201 "end)\n", w, ctx->cc_time_per_weight[w], est_str, cc_dur, n_cc);
2202 }
2203 }
2204
2205 int cw_found = atomic_load(&ctx->cc_found_weight);
2206 if (cw_found > 0 && certified && cw_found == w && !atomic_load(&ctx->cc_exact)) {
2207 atomic_store(&ctx->cc_exact, true);
2208 }
2209 const bool exact = (cw_found > 0) && atomic_load(&ctx->cc_exact);
2210 if (exact) {
2211 atomic_store(&ctx->dmin, cw_found);
2212 atomic_store(&ctx->dmax, cw_found);
2213
2214 int max_w_lim = minint(ctx->p->wmax > 0 ? ctx->p->wmax : nvar, cw_found + ctx->p->dW);
2215 if (ctx->p->dstop > 0 && ctx->p->dstop < cw_found) {
2216 /* a stop target dstop below the distance: the codewords of weight up to dstop + dW only (as in method 2) */
2217 max_w_lim = minint(max_w_lim, ctx->p->dstop + (ctx->p->dW > 0 ? ctx->p->dW : 0));
2218 }
2219 if (ctx->p->outC && ctx->p->dW > 0 && w < max_w_lim) {
2220 if (ctx->p->debug & DBG_PROGRESS) {
2221 if (w == cw_found) {
2222 fprintf(stderr,
2223 "# CC round w=%d finished in %.3fs (%d CC threads, %d RW threads): found codewords "
2224 "(dmin=%d, continuing up to w=%d for dW=%d, total %lld cws)\n",
2225 w, cc_dur, n_cc, n_rw, cw_found, max_w_lim, ctx->p->dW, ctx_num_cws(ctx));
2226 } else if (round_completed) {
2227 fprintf(stderr,
2228 "# CC round w=%d finished in %.3fs (%d CC threads, %d RW threads): "
2229 "extra dW round completed (dmin=%d, total %lld cws)\n",
2230 w, cc_dur, n_cc, n_rw, cw_found, ctx_num_cws(ctx));
2231 }
2232 }
2233 } else {
2234 if (ctx->p->debug & DBG_PROGRESS) {
2235 if (w > cw_found) {
2236 if (round_completed) {
2237 fprintf(stderr,
2238 "# CC round w=%d finished in %.3fs (%d CC threads, %d RW threads): "
2239 "extra dW round completed (dmin=%d, total %lld cws)\n",
2240 w, cc_dur, n_cc, n_rw, cw_found, ctx_num_cws(ctx));
2241 }
2242 } else {
2243 fprintf(stderr, "# CC found min-weight codeword: d=%d (using %d CC threads, total %lld cws)\n",
2244 cw_found, n_cc, ctx_num_cws(ctx));
2245 }
2246 }
2247 /* (a stop reason recorded during the round, e.g., the timeout, is kept) */
2248 ctx_stop(ctx, (ctx->p->outC || ctx->p->maxC > 0) ? STOP_COLLECTED : STOP_CC_FOUND);
2249 break;
2250 }
2251 } else if (cur_dmax > 0 && cur_dmin >= cur_dmax) {
2252 /* Extra dW round completed */
2253 if (round_completed && (ctx->p->debug & DBG_PROGRESS)) {
2254 fprintf(stderr,
2255 "# CC round w=%d finished in %.3fs (%d CC threads, %d RW threads): "
2256 "extra dW round completed (dmin=%d, total %lld cws)\n",
2257 w, cc_dur, n_cc, n_rw, cur_dmin, ctx_num_cws(ctx));
2258 }
2259 if (!round_completed) {
2260 break;
2261 }
2262 } else {
2263 if (!round_completed) {
2264 break;
2265 }
2266
2267 /* Weight w analyzed without success */
2268 int new_dmin = w + 1;
2269 atomic_store(&ctx->dmin, new_dmin);
2270 if (ctx->p->debug & DBG_PROGRESS) {
2271 fprintf(stderr, "# CC round w=%d finished in %.3fs (%d CC threads, %d RW threads): no codewords -> dmin=%d\n",
2272 w, cc_dur, n_cc, n_rw, new_dmin);
2273 }
2274
2275 cur_dmax = atomic_load(&ctx->dmax);
2276 if (cur_dmax > 0 && new_dmin >= cur_dmax) {
2277 atomic_store(&ctx->dmin, cur_dmax);
2278 /* with outC, the CC rounds w = dmax..dmax+dW enumerate all codewords to export (RW may have missed some) */
2279 const int w_last = ctx->p->outC ? m3_cc_target_w(ctx, cur_dmax, cur_dmax) : 0;
2280 if (w_last > w) {
2281 if (ctx->p->debug & DBG_PROGRESS) {
2282 fprintf(stderr, "# bracketing bounds coincide: dmin = dmax = %d (continuing up to w=%d for dW=%d to export "
2283 "codewords)\n", cur_dmax, w_last, ctx->p->dW > 0 ? ctx->p->dW : 0);
2284 }
2285 } else {

2330/* Maximum number of RW workers for thread throttling (all workers in method 1, the RW share in method 3):
2331 * - dense memory: a full-matrix RW worker holds dense copies of H (m x n) and of its transpose (n x m), a worker
2332 * with ksub > 0 only the sampled subspace matrix (ksub x n). Above 15 MB per worker, the total is kept at
2333 * ~1.5 GB (2 to 32 workers) to prevent DRAM bus and L3 cache thrashing;
2334 * - small step counts: at most ceil(steps/10) RW workers. */
2335static int rw_thread_cap(const params_t * const p) {
2336 int cap = INT_MAX;
2337 const int nvar = p->nvar;
2338 const int nrows = p->spaH ? p->spaH->rows : 0;
2339 if (nrows > 0 && nvar > 0) {
2340 const size_t words_n = ((size_t)nvar + 63) / 64;
2341 size_t dense_bytes_per_thread;
2342 if (p->ksub > 0) {
2343 dense_bytes_per_thread = (size_t)minint(p->ksub, nrows) * words_n * 8;
2344 } else {
2345 dense_bytes_per_thread = (size_t)nrows * words_n * 8 + (size_t)nvar * (((size_t)nrows + 63) / 64) * 8;
2346 }
2347 if (dense_bytes_per_thread > 15ULL * 1024 * 1024) {
2348 int max_mem_threads = (int)((1536ULL * 1024 * 1024) / dense_bytes_per_thread);
2349 if (max_mem_threads < 2) max_mem_threads = 2;
2350 if (max_mem_threads > 32) max_mem_threads = 32;
2351 cap = max_mem_threads;
2352 }
2353 }
2354 if (p->steps > 0) {
2355 const int max_step_threads = (int)(((long)p->steps + 9) / 10);
2356 cap = minint(cap, max_step_threads);
2357 }
2358 return cap;
2359}

2382/* DBG_SUMMARY: why the run ended (see ctx_stop()) */
2383static void print_stop_reason(distfork_ctx_t * const ctx, const int final_dmax, const bool cc_exact) {
2384 params_t * const p = ctx->p;
2385 const int cc_found = atomic_load(&ctx->cc_found_weight);
2386 const bool rw_ran = (p->method & 1) && ctx->rw_threads > 0;
2387 fprintf(stderr, "# stopped after %.3fs: ", get_time_sec() - ctx->start_time);
2388 switch (atomic_load(&ctx->stop_reason)) {
2389 case STOP_BOUNDS:
2390 fprintf(stderr, "bounds coincide: dmin = dmax = %d", final_dmax);
2391 break;
2392 case STOP_CC_FOUND:
2393 if (cc_exact)
2394 fprintf(stderr, "CC found min-weight codeword: d=%d", cc_found);
2395 else
2396 fprintf(stderr, "CC found a codeword of weight %d (upper bound only: lower weights not scanned)", cc_found);
2397 break;
2398 case STOP_RW_CONVERGED: {
2399 cw_hit_stats_t st;
2400 codeword_hit_stats(p, &st);
2401 fprintf(stderr, "RW convergence reached (<n>=%.2f >= min_hits=%d after %ld RW steps)", st.avg, p->min_hits,
2402 atomic_load(&ctx->rw_steps_completed));
2403 break;
2404 }
2405 case STOP_RW_STEPS:
2406 fprintf(stderr, "all %ld RW steps done", ctx->total_rw_steps);
2407 break;
2408 case STOP_TIMEOUT:
2409 fprintf(stderr, "timeout=%gs reached", ctx->timeout);
2410 if (ctx->round_open)
2411 fprintf(stderr, " during CC round w=%d (%.1f%% of start columns claimed)", ctx->round_w, cc_round_claimed(ctx));
2412 if (rw_ran)
2413 fprintf(stderr, ", %ld of %ld RW steps done", atomic_load(&ctx->rw_steps_completed), ctx->total_rw_steps);
2414 break;
2415 case STOP_CC_DONE:
2416 fprintf(stderr, "CC done up to w=%d", ctx->stop_w);
2417 if (p->method == 3 && rw_ran) fprintf(stderr, ", RW ended (%s)", rw_end_state(ctx));
2418 break;
2419 case STOP_CC_SLOW:
2420 fprintf(stderr, "CC round w=%d predicted not to finish before the timeout, RW ended (%s)", ctx->stop_w,
2421 rw_end_state(ctx));
2422 break;
2423 case STOP_WMIN:
2424 fprintf(stderr, "found a codeword of weight %d <= wmin=%d", final_dmax, p->wmin);
2425 break;
2426 case STOP_MAXC:
2427 fprintf(stderr, "maxC=%lld codewords collected", p->maxC);
2428 break;
2429 case STOP_COLLECTED: {
2430 /* the collection rounds w = d..d+dW (with outC), but at most up to dstop+dW (a stop target dstop < d) and wmax */

2432 if (p->outC && p->dW > 0) {
2433 w_hi = final_dmax + p->dW;
2434 if (p->dstop > 0 && p->dstop < final_dmax) w_hi = minint(w_hi, p->dstop + p->dW);
2435 if (p->wmax > 0) w_hi = minint(w_hi, p->wmax);
2436 }
2437 fprintf(stderr, "CC enumerated all codewords of weight %d", final_dmax);
2438 if (w_hi > final_dmax) fprintf(stderr, "..%d", w_hi);
2439 if (p->outC) fprintf(stderr, " (for outC)");
2440 break;
2441 }
2442 case STOP_DSTOP:
2443 fprintf(stderr, "lower bound dmin=%d reached dstop=%d", atomic_load(&ctx->dmin), p->dstop);
2444 break;
2445 default:
2446 fprintf(stderr, "search ended");
2447 }
2448 fprintf(stderr, "\n");
2449}

2451typedef struct {
2452 uint64_t h;
2453 int c;
2454} cc_hash_col_t;

2456static int cc_hash_col_cmp(const void *a, const void *b) {
2457 const cc_hash_col_t *x = a, *y = b;
2458 if (x->h != y->h) return (x->h < y->h) ? -1 : 1;
2459 return (x->c > y->c) - (x->c < y->c);
2460}

2462/* The CC column-syndrome hash of H (mHT: its transpose with m rows), see cc_hash_t; NULL if H has no nonzero column */
2463static cc_hash_t *cc_hash_new(const csr_t * const mHT, const int m) {
2464 const int n = mHT->rows;
2465 cc_hash_t * const hs = calloc(1, sizeof(cc_hash_t));
2466 cc_hash_col_t * const tmp = malloc((size_t)(n > 0 ? n : 1) * sizeof(cc_hash_col_t));
2467 if (!hs || !tmp) ERROR("memory allocation");
2468 hs->zr = malloc((size_t)(m > 0 ? m : 1) * sizeof(uint64_t));
2469 hs->hc = calloc((size_t)(n > 0 ? n : 1), sizeof(uint64_t));
2470 if (!hs->zr || !hs->hc) ERROR("memory allocation");
2471 uint64_t state = 0x243f6a8885a308d3ULL; /* fixed: the results do not depend on the hash */
2472 for (int r = 0; r < m; r++) hs->zr[r] = splitmix64(&state);
2473 int nz = 0;
2474 for (int c = 0; c < n; c++) {
2475 if (mHT->p[c + 1] == mHT->p[c]) continue;
2476 uint64_t h = 0;
2477 for (int k = mHT->p[c]; k < mHT->p[c + 1]; k++) h ^= hs->zr[mHT->i[k]];
2478 hs->hc[c] = h;
2479 tmp[nz].h = h;
2480 tmp[nz++].c = c;
2481 }
2482 if (nz == 0) {
2483 free(tmp);
2484 free(hs->zr);
2485 free(hs->hc);
2486 free(hs);
2487 return NULL;
2488 }
2489 qsort(tmp, nz, sizeof(cc_hash_col_t), cc_hash_col_cmp);
2490 int nkeys = 0;
2491 for (int k = 0; k < nz; k++) nkeys += (k == 0 || tmp[k].h != tmp[k - 1].h);
2492 int lg = 0; /* ceil(log2(nkeys)) */
2493 while ((1LL << lg) < nkeys) lg++;
2494 const int tbits = lg + 1; /* table: at most half full */
2495 const int fbits = (lg + 4 > 6) ? lg + 4 : 6; /* filter: 16 bits per key */
2496 hs->tshift = 64 - tbits;
2497 hs->tmask = (int)((1ULL << tbits) - 1);
2498 hs->tab = calloc((size_t)1 << tbits, sizeof(cc_hash_slot_t));
2499 hs->bmask = ((uint64_t)1 << fbits) - 1;
2500 hs->bits = calloc((size_t)1 << (fbits - 6), sizeof(uint64_t));
2501 hs->cols = malloc((size_t)nz * sizeof(int));
2502 if (!hs->tab || !hs->bits || !hs->cols) ERROR("memory allocation");
2503 for (int k = 0; k < nz; k++) hs->cols[k] = tmp[k].c;
2504 for (int k = 0; k < nz;) {
2505 const uint64_t key = tmp[k].h;
2506 int k2 = k + 1;
2507 while (k2 < nz && tmp[k2].h == key) k2++;
2508 int j = (int)(key >> hs->tshift);
2509 while (hs->tab[j].cnt > 0) j = (j + 1) & hs->tmask;
2510 hs->tab[j].key = key;
2511 hs->tab[j].beg = k;
2512 hs->tab[j].cnt = k2 - k;
2513 const uint64_t fb = key & hs->bmask;
2514 hs->bits[fb >> 6] |= (uint64_t)1 << (fb & 63);
2515 k = k2;
2516 }
2517 hs->ncols = nz;
2518 hs->nkeys = nkeys;
2519 free(tmp);
2520 return hs;
2521}

2523static void cc_hash_free(cc_hash_t * const hs) {
2524 if (!hs) return;
2525 free(hs->zr);
2526 free(hs->hc);
2527 free(hs->bits);
2528 free(hs->tab);
2529 free(hs->cols);
2530 free(hs);
2531}
