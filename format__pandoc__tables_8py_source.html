38def cell_metrics(cell: str) -> tuple[float, float]:
39 """Return (full_width_chars, max_unbreakable_chars) in approximate LaTeX width units."""
40 # Strip markdown links [text](url) -> text
41 s = re.sub(r"\‍[([^\‍]]+)\‍]\‍([^)]+\‍)", r"\1", cell)
42 # Tokenize into backtick code `...`, math $...$, and plain text
43 pattern = re.compile(r"(`[^`]+`|\$[^$]+\$)")
44 parts = pattern.split(s)

46 total_w = 0.0
47 max_unbreak = 0.0
48 cur_unbreak = 0.0

50 def flush_space() -> None:
51 nonlocal max_unbreak, cur_unbreak
52 if cur_unbreak > max_unbreak:
53 max_unbreak = cur_unbreak
54 cur_unbreak = 0.0
55
56 for part in parts:
57 if not part:
58 continue
59 if part.startswith("`") and part.endswith("`"):
60 code = part[1:-1]
61 # Monospace is slightly wider than roman; breaks only at space or '_' (via \allowbreak)
62 chunks = re.split(r"( +|_)", code)
63 for ch in chunks:
64 if not ch:
65 continue
66 if ch.startswith(" "):
67 total_w += len(ch)
68 flush_space()
69 elif ch == "_":
70 total_w += 1.12
71 cur_unbreak += 1.12
72 flush_space()
73 else:
74 w = len(ch) * 1.12
75 total_w += w
76 cur_unbreak += w
77 elif part.startswith("$") and part.endswith("$"):
78 mt = approx_math_text(part[1:-1])
79 # Math can break around binary/rel ops or spaces, approximate unbreakable chunks
80 chunks = re.split(r"( +|=|\+|-|,)", mt)
81 for ch in chunks:
82 if not ch:
83 continue
84 if ch.startswith(" "):
85 total_w += 0.8
86 flush_space()
87 elif ch in ("=", "+", "-", ","):
88 total_w += 1.0
89 cur_unbreak += 1.0
90 flush_space()
91 else:
92 w = len(ch) * 0.95
93 total_w += w
94 cur_unbreak += w
95 else:
96 plain = part.replace("**", "").replace("*", "")
97 chunks = re.split(r"( +|-)", plain)
98 for ch in chunks:
99 if not ch:
100 continue
101 if ch.startswith(" "):
102 total_w += len(ch) * 0.85
103 flush_space()
104 elif ch == "-":
105 total_w += 0.9
106 cur_unbreak += 0.9
107 flush_space()
108 else:
109 w = len(ch) * 1.00
110 total_w += w
111 cur_unbreak += w
112
113 flush_space()

126def make_sep_token(left: bool, right: bool, length: int) -> str:
127 dashes = length - (1 if left else 0) - (1 if right else 0)

132def allocate_tokens(rows: list[list[str]], aligns: list[tuple[bool, bool, int]], total_units: int = 80) -> list[int]:
133 ncols = len(aligns)
134 # Effective chars per unit out of 80:
135 # Usable width = 576 - 12*(ncols - 1) pt; 1 monospace-equiv char ~ 5.0 pt => chars_per_unit = usable / (5.0 * 80)
136 usable_pt = 576.0 - 12.0 * (ncols - 1)
137 chars_per_unit = usable_pt / (5.05 * total_units)
138
139 max_full = [0.0] * ncols
140 max_unb = [0.0] * ncols
141 for r in rows:
142 for j in range(min(ncols, len(r))):
143 fw, uw = cell_metrics(r[j])
144 if fw > max_full[j]:
145 max_full[j] = fw
146 if uw > max_unb[j]:
147 max_unb[j] = uw
148
149 floors = []
150 desires = []
151 for j in range(ncols):
152 _, _, min_tok = aligns[j]
153 # Floor guarantees the longest unbreakable chunk fits in p{...} without Overfull \hbox
154 unb_units = math.ceil(max_unb[j] / chars_per_unit + 0.65)
155 fl = max(min_tok, unb_units)
156 floors.append(fl)
157 # Concave desire: full width up to 22 chars, then 0.50 slope above 22
158 m = max_full[j]
159 des_chars = min(m, 22.0) + 0.50 * max(0.0, m - 22.0)
160 des_units = max(float(fl), des_chars / chars_per_unit)
161 desires.append(des_units)
162
163 sum_floors = sum(floors)
164 if sum_floors >= total_units:
165 # Scale down proportionally while keeping min_tok
166 min_toks = [a[2] for a in aligns]
167 rem = total_units - sum(min_toks)
168 extra = [max(0, floors[j] - min_toks[j]) for j in range(ncols)]
169 s_ex = sum(extra) or 1
170 alloc = [min_toks[j] + int(math.floor(rem * extra[j] / s_ex)) for j in range(ncols)]
171 while sum(alloc) < total_units:
172 best = max(range(ncols), key=lambda j: (rem * extra[j] / s_ex) - (alloc[j] - min_toks[j]))
173 alloc[best] += 1
174 return alloc
175
176 rem = total_units - sum_floors
177 weights = [max(0.05, desires[j] - floors[j]) for j in range(ncols)]
178 w_sum = sum(weights)
179 raw_add = [rem * w / w_sum for w in weights]
180 add_int = [int(math.floor(x)) for x in raw_add]
181 alloc = [floors[j] + add_int[j] for j in range(ncols)]
182 leftover = total_units - sum(alloc)
183 fracs = sorted(range(ncols), key=lambda j: raw_add[j] - add_int[j], reverse=True)
184 for j in fracs[:leftover]:
185 alloc[j] += 1
186 return alloc
