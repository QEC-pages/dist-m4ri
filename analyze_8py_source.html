222def load_session(session_dir: str | Path) -> tuple[list[dict[str, Any]], dict[str, Any] | None, dict[str, Any]]:
223 """
224 Load one benchmark session directory.
225
226 Reads `results.jsonl` (skipping and counting malformed JSON lines) or falls back to
227 `runs/*/run.json` when `results.jsonl` is absent. Optionally loads `env.json`, `plan.json`,
228 and `session.json`.
229
230 Args:
231 session_dir: Path to a session directory written by `harness.py`.
232
233 Returns:
234 Tuple `(records, env, meta)` where `records` is the list of parsed run dicts in file order,
235 `env` is the parsed `env.json` dict (or None), and `meta` contains loader metadata.
236 """
237 sdir = Path(session_dir)
238 records: list[dict[str, Any]] = []
239 bad_lines = 0
240 bad_details: list[str] = []
241 source = "empty"
242
243 jsonl_path = sdir / "results.jsonl"
244 if jsonl_path.is_file():
245 source = "results.jsonl"
246 with jsonl_path.open("r", encoding="utf-8", errors="replace") as fh:
247 for lineno, raw_line in enumerate(fh, start=1):
248 line = raw_line.strip()
249 if not line:
250 continue
251 try:
252 obj = json.loads(line)
253 except ValueError as exc:
254 bad_lines += 1
255 bad_details.append(f"{jsonl_path.name}:{lineno}: {exc}")
256 continue
257 if isinstance(obj, dict):
258 records.append(obj)
259 else:
260 bad_lines += 1
261 bad_details.append(f"{jsonl_path.name}:{lineno}: JSON value is not an object")
262 else:
263 runs_dir = sdir / "runs"
264 if runs_dir.is_dir():
265 source = "runs"
266 for rpath in sorted(runs_dir.glob("*/run.json")):
267 try:
268 obj = json.loads(rpath.read_text(encoding="utf-8", errors="replace"))
269 except (OSError, ValueError) as exc:
270 bad_lines += 1
271 bad_details.append(f"{rpath.parent.name}/run.json: {exc}")
272 continue
273 if isinstance(obj, dict):
274 records.append(obj)
275 else:
276 bad_lines += 1
277 bad_details.append(f"{rpath.parent.name}/run.json: JSON value is not an object")

406 manifest: Mapping[str, Any],
407 records: Sequence[dict[str, Any]],
408) -> tuple[list[dict[str, Any]], list[dict[str, Any]], int]:
409 """

419 continue

427 m_ver = manifest.get("version")

438 mismatch_reasons.append(f"manifest.version={rm_ver!r} != {m_ver!r}")

487def _aggregate_cells(
488 main_recs: Sequence[dict[str, Any]],
489 inst_meta: dict[str, dict[str, Any]],
490 insts_by_tier: dict[str, list[str]],
491 tiers_order: list[str],
492 tier_budgets: Mapping[str, float],
493 baseline: str,
494) -> tuple[dict[tuple[str, str], dict[str, Any]], dict[str, list[str]], list[str]]:
495 """Aggregate main-set records per `(entry, instance_id)`."""
496 grouped: dict[tuple[str, str], list[dict[str, Any]]] = {}
497 entries_per_tier_set: dict[str, set[str]] = {t: set() for t in tiers_order}
498 all_entries_set: set[str] = set()
499
500 for rec in main_recs:
501 iid, tier = _resolve_instance(rec, inst_meta, insts_by_tier, tiers_order)
502 entry = _entry_name(rec)
503 grouped.setdefault((entry, iid), []).append(rec)
504 entries_per_tier_set.setdefault(tier, set()).add(entry)
505 all_entries_set.add(entry)
506
507 cells: dict[tuple[str, str], dict[str, Any]] = {}
508 for (entry, iid), runs in grouped.items():
509 tier = str(inst_meta[iid]["tier"])
510 n_runs = len(runs)
511 par2_vals = [_record_par2(r, tier_budgets, tier) for r in runs]
512 par10_vals = [_record_par10(r, tier_budgets, tier) for r in runs]
513 budgets = [_record_budget(r, tier_budgets, tier) for r in runs]
514 solved_walls = [
515 float((r.get("result") or {})["wall_sec"])
516 for r in runs
517 if _is_solved(r) and _as_float((r.get("result") or {}).get("wall_sec")) is not None
518 ]
519 n_solved = len(solved_walls)
520 solved_by_med = (2 * n_solved) > n_runs
521 med_par2 = float(np.median(par2_vals))
522 med_par10 = float(np.median(par10_vals))
523 min_par2 = float(min(par2_vals))
524 max_par2 = float(max(par2_vals))
525 t_min = float(min(solved_walls)) if solved_walls else None
526 t_max = float(max(solved_walls)) if solved_walls else None
527 t_med_solved = float(np.median(solved_walls)) if solved_walls else None
528
529 lbs: list[int] = []
530 ubs: list[int] = []
531 status_counts: dict[str, int] = {}
532 seeds: list[int] = []
533 for r in runs:
534 res = r.get("result") or {}
535 st = str(res.get("status") or "unknown")
536 status_counts[st] = status_counts.get(st, 0) + 1
537 sd = _as_int((r.get("options") or {}).get("seed"))
538 if sd is not None:
539 seeds.append(sd)
540 if st in VALID_BRACKET_STATUSES:
541 lb = _as_int(res.get("lb"))
542 ub = _as_int(res.get("ub"))
543 if lb is not None:
544 lbs.append(lb)
545 if ub is not None:
546 ubs.append(ub)
547
548 best_lb = max(lbs) if lbs else None
549 best_ub = min(ubs) if ubs else None
550 best_gap = (best_ub - best_lb) if (best_lb is not None and best_ub is not None) else None
551
552 cells[(entry, iid)] = {
553 "entry": entry,
554 "instance": iid,
555 "tier": tier,
556 "n_runs": n_runs,
557 "n_solved": n_solved,
558 "solved_by_median": solved_by_med,
559 "budget_s": float(max(budgets)) if budgets else float(tier_budgets.get(tier, 0.0)),
560 "med_par2": med_par2,
561 "med_par10": med_par10,
562 "min_par2": min_par2,
563 "max_par2": max_par2,
564 "t_med_solved": t_med_solved,
565 "t_min": t_min,
566 "t_max": t_max,
567 "best_lb": best_lb,
568 "best_ub": best_ub,
569 "best_gap": best_gap,
570 "status_counts": status_counts,
571 "seeds": sorted(set(seeds)),
572 }
573
574 entries_by_tier = {t: _sort_entries(list(entries_per_tier_set.get(t, set())), baseline) for t in tiers_order}
575 all_entries = _sort_entries(list(all_entries_set), baseline)
576 return cells, entries_by_tier, all_entries

579def _bootstrap_geomean_ci(
580 ratios: Sequence[float],
581 n_boot: int,
582 seed: int,
583) -> tuple[float | None, float | None, float | None, float | None, float | None]:
584 """Compute `(geomean, ci_low, ci_high, min_r, max_r)` via bootstrap resampling over instances."""
585 vals = [float(r) for r in ratios if r > 0.0 and math.isfinite(r)]
586 n = len(vals)
587 if n == 0:
588 return None, None, None, None, None
589 arr = np.asarray(vals, dtype=np.float64)
590 log_arr = np.log(arr)
591 gm = float(np.exp(np.mean(log_arr)))
592 r_min = float(np.min(arr))
593 r_max = float(np.max(arr))
594 if n == 1 or n_boot <= 0:
595 return gm, gm, gm, r_min, r_max
596 rng = np.random.default_rng(seed)
597 idx = rng.integers(0, n, size=(n_boot, n))
598 boot_gm = np.exp(np.mean(log_arr[idx], axis=1))
599 ci_lo, ci_hi = np.percentile(boot_gm, [2.5, 97.5])
600 return gm, float(ci_lo), float(ci_hi), r_min, r_max

603def _classify_wtl(
604 base_cell: Mapping[str, Any],
605 entry_cell: Mapping[str, Any],
606) -> str:
607 """
608 Classify an instance where both baseline and entry solved by median as `win`, `tie`, or `loss`.
609
610 A tie holds when `max(t_b, t_e) <= 1.10 * min(t_b, t_e)` (within +-10%) OR when the seed
611 min-max ranges `[min_par2, max_par2]` overlap (equal to solved `[t_min, t_max]` when all seeds
612 are solved, and conservative when some seeds timed out).
613 """
614 tb = float(base_cell["med_par2"])
615 te = float(entry_cell["med_par2"])
616 lo_b, hi_b = float(base_cell["min_par2"]), float(base_cell["max_par2"])
617 lo_e, hi_e = float(entry_cell["min_par2"]), float(entry_cell["max_par2"])
618
619 t_lo = min(tb, te)
620 t_hi = max(tb, te)
621 within_10 = (t_hi <= 1.10 * t_lo) if t_lo > 0.0 else (t_hi == 0.0)
622 overlap = (lo_e <= hi_b) and (lo_b <= hi_e)
623 if within_10 or overlap:
624 return "tie"
625 return "win" if te < tb else "loss"

628def _compute_vbs_for_instances(
629 inst_ids: Sequence[str],
630 entries: Sequence[str],
631 cells: Mapping[tuple[str, str], Mapping[str, Any]],
632 tier_budgets: Mapping[str, float],
633 inst_meta: Mapping[str, Mapping[str, Any]],
634) -> tuple[dict[str, dict[str, Any]], dict[str, float | None], float | None]:
635 """
636 Compute per-instance VBS, per-entry marginal VBS contribution, and tier VBS mean PAR-2.
637
638 For each instance `i` that has data from at least one entry:
639 - `VBS(i) = min_e med_par2(e, i)`
640 - `VBS_without(e, i) = min_{o != e} med_par2(o, i)` if another entry has data, else `2 * budget_s(i)`
641 - `marginal_vbs(e) = mean_i VBS_without(e, i) - mean_i VBS(i)`
642 """
643 per_inst: dict[str, dict[str, Any]] = {}
644 scored_insts: list[str] = []
645
646 for iid in inst_ids:
647 avail = [(e, cells[(e, iid)]) for e in entries if (e, iid) in cells]
648 if not avail:
649 per_inst[iid] = {
650 "has_data": False,
651 "solved_by_median": False,
652 "med_par2": None,
653 "best_entries": [],
654 "best_lb": None,
655 "best_ub": None,
656 "best_gap": None,
657 "marginal_by_entry": {},
658 }
659 continue
660 scored_insts.append(iid)
661 min_par2 = min(float(c["med_par2"]) for _, c in avail)
662 solved_any = any(bool(c["solved_by_median"]) for _, c in avail)
663 tol = max(1e-12, 1e-12 * abs(min_par2))
664 best_entries = [e for e, c in avail if abs(float(c["med_par2"]) - min_par2) <= tol]
665
666 lbs = [int(c["best_lb"]) for _, c in avail if c["best_lb"] is not None]
667 ubs = [int(c["best_ub"]) for _, c in avail if c["best_ub"] is not None]
668 vbs_lb = max(lbs) if lbs else None
669 vbs_ub = min(ubs) if ubs else None
670 vbs_gap = (vbs_ub - vbs_lb) if (vbs_lb is not None and vbs_ub is not None) else None
671
672 tier = str(inst_meta[iid]["tier"])
673 budget_i = max(float(c["budget_s"]) for _, c in avail)
674 if budget_i <= 0.0:
675 budget_i = float(tier_budgets.get(tier, 0.0))
676
677 marg_i: dict[str, float] = {}
678 for e in entries:
679 others = [float(c["med_par2"]) for oe, c in avail if oe != e]
680 without_e = min(others) if others else (2.0 * budget_i)
681 marg_i[e] = float(without_e - min_par2)
682
683 per_inst[iid] = {
684 "has_data": True,
685 "solved_by_median": solved_any,
686 "med_par2": float(min_par2),
687 "best_entries": best_entries,
688 "best_lb": vbs_lb,
689 "best_ub": vbs_ub,
690 "best_gap": vbs_gap,
691 "budget_s": budget_i,
692 "marginal_by_entry": marg_i,
693 }
694
695 if not scored_insts:
696 return per_inst, {e: None for e in entries}, None
697
698 vbs_par2 = float(np.mean([float(per_inst[i]["med_par2"]) for i in scored_insts]))
699 marg_by_entry: dict[str, float | None] = {}
700 for e in entries:
701 deltas = [float(per_inst[i]["marginal_by_entry"][e]) for i in scored_insts]
702 marg_by_entry[e] = float(np.mean(deltas))
703 return per_inst, marg_by_entry, vbs_par2

979def _pick_best_entries_for_instance(
980 entries: Sequence[str],
981 cells: Mapping[tuple[str, str], Mapping[str, Any]],
982 iid: str,
983) -> list[str]:
984 """Select the best entry (or tied entries) to bold in Table B for one instance."""
985 avail = [(e, cells[(e, iid)]) for e in entries if (e, iid) in cells]
986 if not avail:
987 return []
988 solved = [(e, c) for e, c in avail if c["solved_by_median"]]
989 if solved:
990 best_t = min(float(c["med_par2"]) for _, c in solved)
991 tol = max(1e-12, 1e-12 * abs(best_t))
992 return [e for e, c in solved if abs(float(c["med_par2"]) - best_t) <= tol]
993 # Nobody solved by median: smallest valid bracket gap (tie-break: highest lb, lowest ub).
994 with_gap = [(e, c) for e, c in avail if c["best_gap"] is not None]
995 if with_gap:
996 best_key = min((int(c["best_gap"]), -int(c["best_lb"]), int(c["best_ub"])) for _, c in with_gap)
997 return [
998 e
999 for e, c in with_gap
1000 if (int(c["best_gap"]), -int(c["best_lb"]), int(c["best_ub"])) == best_key
1001 ]
1002 with_lb = [(e, c) for e, c in avail if c["best_lb"] is not None]
1003 if with_lb:
1004 max_lb = max(int(c["best_lb"]) for _, c in with_lb)
1005 return [e for e, c in with_lb if int(c["best_lb"]) == max_lb]
1006 return []

1009def _format_table_b_cell(cell: Mapping[str, Any] | None) -> str:
1010 """Format one `(entry, instance)` cell for Table B."""
1011 if cell is None:
1012 return "no data"
1013 n_runs = int(cell["n_runs"])
1014 n_solved = int(cell["n_solved"])
1015 if cell["solved_by_median"]:

1017 if n_runs == 1:
1018 return med_s
1019 lo_s = _fmt_time(cell["t_min"])
1020 hi_s = _fmt_time(cell["t_max"])
1021 base = f"{med_s} [{lo_s}, {hi_s}]"

1025 if lb is not None or ub is not None:

1028 return f"[{lb_s}, {ub_s}] {n_solved}/{n_runs}"
1029 counts = cell.get("status_counts") or {}
1030 if counts.get("wrong"):
1031 return f"wrong {n_solved}/{n_runs}"
1032 if counts.get("error"):
1033 return f"err {n_solved}/{n_runs}"
1034 return f"timeout {n_solved}/{n_runs}"

1098def _compute_table_c(
1099 valid_recs: Sequence[dict[str, Any]],
1100 excluded_recs: Sequence[dict[str, Any]],
1101 exclude_contaminated: bool,
1102) -> dict[str, Any]:
1103 """
1104 Compute Table C integrity audit across all deduplicated records matching the manifest.
1105
1106 Checks:
1107 1. `wrong`: runs with `result.status == "wrong"` and their `verify.violations`.
1108 2. `lb_conflicts`: across all runs (all leagues and canaries), any run whose `result.lb` exceeds
1109 the minimum `verify.witness_weight` among runs of the same instance with
1110 `verify.witness_verified_numpy is True`.
1111 3. `unverified_certified`: runs with `result.status == "certified"` whose
1112 `verify.witness_verified_numpy` is not True.
1113 4. `contaminated`: runs with `env.contaminated` True.
1114 5. `errors`: runs with `result.status == "error"` (with `error.kind` and `error.msg`).
1115 6. `excluded`: records dropped before analysis (e.g. manifest id/version mismatch).
1116 """
1117 wrong_rows: list[dict[str, Any]] = []
1118 unverified_rows: list[dict[str, Any]] = []
1119 contaminated_rows: list[dict[str, Any]] = []
1120 error_rows: list[dict[str, Any]] = []
1121
1122 # Collect minimum verified witness weight per instance across ALL valid records.
1123 min_witness: dict[str, dict[str, Any]] = {}
1124 for rec in valid_recs:
1125 iid = str((rec.get("instance") or {}).get("id") or "unknown")
1126 ver = rec.get("verify") or {}
1127 if ver.get("witness_verified_numpy") is True:

1757 scaling_recs.append(rec)
1758 elif league == "default" and cfg == "default" and thr in (64, 128):
1759 default_p64_128.append(rec)

1783 for r in default_p64_128
1784 if (r.get("instance") or {}).get("id") == iid and _is_scaling_complete(r, "default")
1785 }
1786 if len(ps) >= 2:
1787 scaling_insts.append(iid)
1788 seen_si.add(iid)

1791 # both p=64 and p=128 complete.

1822 iid in seen_si

1824 and (iid, "default", thr) not in grouped_scaling

1826 ):
1827 grouped_default_supp.setdefault((iid, "default", thr), []).append(r)

1858 "efficiency": eff,

1868 "t0": t0,
1869 "points": pts,
1870 }
1871 )
