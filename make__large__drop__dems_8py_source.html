2"""Deterministic generator for large (> 2 MB) A != 0 detector-drop DEM benchmarks.
3
4Generates <prefix>_dropA.dem and <prefix>_dropA.drop from benchmark/stim_bb/ circuits by perturbing
5logical observables as L(j) ^= A * H_M(j) on the minority-basis detector set M with fixed seed=12345:
6 - dgross_288_dropA.{dem,drop}: from double_gross_288_12_18_si1000_r18_X.stim (~17 MB DEM, |M|=2448, k=12)
7 - bb_756_dropA.{dem,drop}: from bb_756_16_34_si1000_r34_Z.stim (> 50 MB DEM, |M|=12474, k=16)
8
9Usage:
10 python3 benchmark/dem/make_large_drop_dems.py --out-dir scratch/large_drop [--only dgross_288]
11"""

26def generate_one(repo_root: Path, name: str, stim_file: str, out_dir: Path, seed: int = 12345) -> None:
27 bin_path = repo_root / "src" / "dist_m4ri"
28 stim_path = repo_root / "benchmark" / "stim_bb" / stim_file
29 out_dir.mkdir(parents=True, exist_ok=True)
30 tmp_src = out_dir / f"_{name}_src.dem"
31 drop_path = out_dir / f"{name}_dropA.drop"
32 out_dem = out_dir / f"{name}_dropA.dem"
33
34 circ = stim.Circuit.from_file(str(stim_path))
35 dem = circ.detector_error_model(decompose_errors=False, flatten_loops=True)
36 dem.to_file(str(tmp_src))
37
38 subprocess.run(
39 [str(bin_path), f"fdem={tmp_src}", f"outD={drop_path}", "method=2", "wmax=1", "threads=4", "debug=1"],
40 check=True, capture_output=True, text=True,
41 )
42 dropped: list[int] = []
43 for ln in drop_path.read_text(encoding="utf-8").splitlines():
44 if ln.strip() and not ln.startswith("#"):
45 dropped.extend(int(x) for x in ln.split())
46 drop_idx = {d: q for q, d in enumerate(dropped)}
47 num_m = len(dropped)
48 k = dem.num_observables
49 rng = random.Random(seed)
50 a_mask = [rng.randrange(1, 1 << k) if q == 0 else rng.randrange(0, 1 << k) for q in range(num_m)]
51
52 out_lines = [
53 f"# Perturbed DEM from {stim_file}: L(j) ^= A * H_M(j) (|M|={num_m}, k={k}, seed={seed})",
54 ]
55 for ln in tmp_src.read_text(encoding="utf-8").splitlines():
56 if not ln.startswith("error"):
57 out_lines.append(ln)
58 continue
59 toks = ln.split()
60 head = toks[0]
61 dets = []
62 lmask = 0
63 hm_xor = 0
64 for tok in toks[1:]:
65 if tok == "^":
66 continue
67 if tok.startswith("D"):
68 d = int(tok[1:])
69 dets.append(tok)
70 q = drop_idx.get(d)
71 if q is not None:
72 hm_xor ^= a_mask[q]
73 elif tok.startswith("L"):
74 lmask ^= 1 << int(tok[1:])
75 new_lmask = lmask ^ hm_xor
76 ltoks = [f"L{i}" for i in range(k) if (new_lmask >> i) & 1]
77 out_lines.append(" ".join([head] + dets + ltoks))
78
79 out_dem.write_text("\n".join(out_lines) + "\n", encoding="utf-8")
80 tmp_src.unlink()
81 print(f"Wrote {out_dem} and {drop_path} (|M|={num_m}, k={k}, seed={seed})")
