64def _unscramble_stderr(stderr: str) -> List[str]:
65 """Recover stderr lines when two threads interleave `# ...` messages mid-line."""
66 raw_lines = stderr.splitlines()
67 out: List[str] = []
68 pending: Optional[str] = None
69 for line in raw_lines:
70 if pending is not None and not line.startswith("#"):
71 out.append(pending + line)
72 pending = None
73 continue
74 if pending is not None:
75 out.append(pending)
76 pending = None
77 if line.startswith("#"):
78 # Look for an embedded '# ' starting a concurrent log line after position 2
79 idx = line.find("# ", 2)
80 if idx > 0 and ("codeword of weight" in line[:idx] or "found new upper bound" in line[:idx]):
81 pending = line[:idx]
82 out.append(line[idx:])
83 continue
84 out.append(line)
85 if pending is not None:
86 out.append(pending)
87 return out

90def _map_stop_code(text: Optional[str]) -> Optional[str]:
91 if not text:
92 return None
93 low = text.lower()
94 if "bounds coincide" in low:
95 return "bounds"
96 if "cc found" in low:
97 return "cc_found"
98 if "timeout" in low:
99 return "timeout"
100 if "rw convergence" in low:
101 return "rw_converged"
102 if "all rw steps" in low or "rw steps done" in low:
103 return "rw_steps"
104 if "predicted" in low or "cc_slow" in low or "not to finish before" in low:
105 return "cc_slow"
106 if "wmax" in low or "largest weight" in low or "cc done" in low:
107 return "cc_done"
108 if "wmin" in low:
109 return "wmin"
110 if "maxc" in low:
111 return "maxc"
112 if "collected" in low:
113 return "collected"
114 if "dstop" in low:
115 return "dstop"
116 if "k = 0" in low or "k=0" in low or "no logical" in low:

128def parse_binary_stderr(
129 stdout: str,
130 stderr: str,
131 *,
132 rc: Optional[int] = 0,
133 killed: bool = False,
134 config: str = "default",
135 dem_mode: str = "canonical",
136 requested_threads: Optional[int] = None,
137) -> Dict[str, Any]:
138 """Parse stdout and stderr of `dist_m4ri` (using `# RESULT {json}` if present, else stderr text)."""
139 clean_err = _strip_ansi(stderr or "")
140 lines = _unscramble_stderr(clean_err)
141
142 # 1. Check for machine-readable `# RESULT {json}` from @binstamp
143 json_obj: Optional[Dict[str, Any]] = None
144 for line in reversed(lines):
145 if line.startswith("# RESULT "):
146 payload = line[len("# RESULT "):].strip()
147 try:
148 parsed = json.loads(payload)
149 if isinstance(parsed, dict) and parsed.get("schema") == "dm4ri-result/1":
150 json_obj = parsed
151 break
152 except ValueError:
153 pass
154
155 if json_obj is not None:
156 return _from_json_result(json_obj, rc=rc, killed=killed, config=config, dem_mode=dem_mode)
157
158 # 2. Fallback: parse stdout result line + stderr progress/summary/witness lines
159 lb: Optional[int] = None
160 ub: Optional[int] = None
161 rw_steps: Optional[int] = None
162 for line in reversed((stdout or "").splitlines()):
163 m = RE_RESULT_LINE.match(line)
164 if m:
165 dmin_v, dmax_v, steps_v = int(m.group(1)), int(m.group(2)), int(m.group(3))
166 lb = dmin_v
167 ub = dmax_v if dmax_v > 0 else None
168 rw_steps = steps_v
169 break
170
171 t_stop: Optional[float] = None
172 stop_reason: Optional[str] = None
173 stop_code: Optional[str] = None
174 t_ub: Optional[float] = None
175 t_lb: Optional[float] = None
176 events: List[List[Any]] = []
177 cc_rounds_by_w: Dict[int, Dict[str, Any]] = {}
178 searched_size: Optional[List[int]] = None
179 full_m: Optional[int] = None
180 full_n: Optional[int] = None
181 rw_opt: Optional[int] = None
182 rw_threads: Optional[int] = None
183 cc_threads: Optional[int] = None
184 k0 = False
185 prog_lb: Optional[int] = None
186 prog_ub: Optional[int] = None
187 witnesses: List[Dict[str, Any]] = []
188
189 for raw_line in lines:
190 line, stamp_t = _parse_stamp(raw_line)
191
192 if "k = 0" in line or "no logical operators (k=0)" in line:
193 k0 = True
194
195 m = RE_H_SIZE.match(line)
196 if m and searched_size is None:
197 searched_size = [int(m.group(1)), int(m.group(2))]
198
199 m = RE_MERGED.search(line)
200 if m:
201 full_n = int(m.group(2))
202
203 m = RE_DROPPED.search(line)
204 if m:
205 full_m = int(m.group(1))
206
207 m = RE_RW_OPT.search(line)
208 if m:
209 rw_opt = int(m.group(1), 16)
210
211 m = RE_RUN_PLAN_THREADS.match(line)
212 if m:
213 cc_threads = int(m.group(1))
214 if rw_threads is None:
215 rw_threads = int(m.group(2))
216
217 m = RE_RW_INFO.search(line)
218 if m:
219 rw_threads = int(m.group(4))
220 if rw_steps is None:
221 rw_steps = int(m.group(3))
222
223 m = RE_UB.search(line)
224 if m:
225 w = int(m.group(1))
226 if prog_ub is None or w < prog_ub:
227 prog_ub = w
228 if stamp_t is not None:
229 t_ub = stamp_t
230 events.append([stamp_t, prog_lb, prog_ub])
231
232 m = RE_CC_ROUND.match(line)
233 if m:
234 w = int(m.group(1))
235 wall_s = float(m.group(2))
236 if m.group(3):
237 cc_threads = max(cc_threads or 0, int(m.group(3)))
238 entry = cc_rounds_by_w.setdefault(
239 w, {"w": w, "thread_s": None, "wall_s": wall_s, "nodes": None, "complete": True}
240 )
241 entry["wall_s"] = wall_s
242 entry["complete"] = True
243 md = RE_DMIN_ARROW.search(line)
244 if md:
245 new_lb = int(md.group(1))
246 if prog_lb is None or new_lb > prog_lb:
247 prog_lb = new_lb
248 if stamp_t is not None:
249 t_lb = stamp_t
250 events.append([stamp_t, prog_lb, prog_ub])
251
252 m = RE_CC_WORK.match(line)
253 if m:
254 w = int(m.group(1))
255 entry = cc_rounds_by_w.setdefault(
256 w, {"w": w, "thread_s": None, "wall_s": None, "nodes": None, "complete": True}
257 )
258 entry["thread_s"] = float(m.group(2))
259 entry["nodes"] = int(m.group(3))
260
261 m = RE_CW.match(line)
262 if m:
263 src_raw = m.group(1).rstrip(":").lower()
264 wt = int(m.group(2))
265 rest = m.group(3)
266 truncated = "..." in rest
267 body = rest.split("...", 1)[0]
268 nums = [int(tok) - 1 for tok in re.findall(r"\d+", body)]
269 complete = (not truncated) and (len(nums) == wt)
270 witnesses.append({
271 "weight": wt,
272 "support": nums,
273 "base": 0,
274 "source": src_raw,
275 "complete": complete,
276 "dem_mode": dem_mode,
277 })
278
279 m = RE_STOP.match(line)
280 if m:
281 t_stop = float(m.group(1))
282 stop_reason = m.group(2).strip()
283 stop_code = _map_stop_code(stop_reason)
284 else:
285 m2 = RE_STOP_NOSEARCH.match(line)
286 if m2 and stop_reason is None:
287 stop_reason = m2.group(1).strip()
288 stop_code = _map_stop_code(stop_reason)
289
290 # For killed runs without stdout result line, use last progress bounds
291 if lb is None and prog_lb is not None:
292 lb = prog_lb
293 if ub is None and prog_ub is not None:
294 ub = prog_ub
295 if k0 and lb == 0 and ub is None:
296 ub = 0
297
298 # Pick the best witness matching ub (preferring complete support of weight == ub, else lightest)
299 chosen_witness: Optional[Dict[str, Any]] = None
300 if witnesses:
301 if ub is not None and ub > 0:
302 matching = [w for w in witnesses if w["weight"] == ub]
303 if matching:
304 complete_m = [w for w in matching if w["complete"]]
305 chosen_witness = complete_m[-1] if complete_m else matching[-1]
306 if chosen_witness is None:
307 witnesses.sort(key=lambda w: (w["weight"], not w["complete"]))
308 chosen_witness = witnesses[0]
309
310 # Full size defaults to searched size when no merge/drop occurred
311 full_size: Optional[List[int]] = None
312 if searched_size is not None:
313 full_size = [
314 full_m if full_m is not None else searched_size[0],
315 full_n if full_n is not None else searched_size[1],
316 ]
317
318 # Status classification
319 error_obj: Optional[Dict[str, str]] = None
320 if killed:
321 status = "timeout"
322 stop_code = "killed"
323 stop_reason = "killed by harness at kill_after_s"
324 exact = False
325 elif rc is not None and rc != 0:
326 status = "error"
327 exact = False
328 err_msg = _extract_error_msg(clean_err) or f"exit code {rc}"
329 error_obj = {"kind": "engine", "msg": err_msg}
330 elif lb is None and not k0:
331 status = "error"
332 exact = False
333 error_obj = {"kind": "engine", "msg": "missing result line on stdout"}
334 elif k0:
335 status = "certified"
336 exact = True
337 stop_code = stop_code or "k0"
338 elif config == "cc_fixed":
339 exact = bool(lb is not None and ub is not None and lb > 0 and lb == ub)
340 if stop_code == "timeout":
341 status = "timeout"
342 elif exact:
343 status = "certified"
344 else:
345 status = "bracket"
346 elif lb is not None and ub is not None and lb > 0 and lb == ub:
347 status = "certified"
348 exact = True
349 elif stop_code == "timeout":
350 status = "timeout"
351 exact = False
352 else:
353 status = "bracket"
354 exact = False
355
356 cc_rounds = [cc_rounds_by_w[w] for w in sorted(cc_rounds_by_w)]
357 return {
358 "status": status,
359 "status_engine": status,
360 "lb": lb,
361 "ub": ub,
362 "exact": exact,
363 "t_internal": {"t_stop": t_stop, "t_input": None, "t_import": None, "t_call": None},
364 "stop_reason": stop_reason,
365 "stop_code": stop_code,
366 "t_ub": t_ub,
367 "t_lb": t_lb,
368 "rw_steps": rw_steps,
369 "size": {"full": full_size, "searched": searched_size},
370 "cc_rounds": cc_rounds,
371 "threads_used": {"requested": requested_threads, "rw": rw_threads, "cc": cc_threads},
372 "rw_opt": rw_opt,
373 "k0": k0,
374 "witness": chosen_witness,
375 "events": events,
376 "parse": "stderr",
377 "engine_info": {},
378 "error": error_obj,
379 }

439 exact = True
440 elif config == "cc_fixed":
441 exact = bool(lb is not None and ub is not None and lb > 0 and lb == ub)
442 status = "timeout" if stop_code == "timeout" else ("certified" if exact else "bracket")
443 elif lb is not None and ub is not None and lb > 0 and lb == ub:
444 status = "certified"
445 exact = True
446 elif stop_code == "timeout":
447 status = "timeout"

491def _extract_error_msg(stderr: str) -> str:
492 lines = [ln.strip() for ln in stderr.splitlines() if ln.strip()]
493 for i, ln in enumerate(lines):
494 if "*** ERROR" in ln and i + 1 < len(lines):
495 return lines[i + 1][:240]
496 if "unrecognized parameter" in ln:
497 return ln[:240]
498 for ln in lines:
499 if not ln.startswith("#"):
500 return ln[:240]
501 return ""

514class DistM4riBinAdapter(Adapter):
515 """Adapter for the native `dist_m4ri` binary (`method=3` default or `method=2` fixed-work scaling)."""
516
517 name = "dist_m4ri"
518 interface = "binary"
519
520 def __init__(
521 self,
522 binary_path: Optional[Path] = None,
523 config: str = "default",
524 repo_root: Path = REPO_ROOT,
525 cache_dir: Optional[Path] = None,
526 allow_fallback: bool = True,
527 ) -> None:
528 if config not in ("default", "cc_fixed"):
529 raise ValueError(f"Unsupported dist_m4ri binary config {config!r} (expected 'default' or 'cc_fixed')")
530 self.config = config
531 self.repo_root = repo_root.resolve()
532 self.binary = (binary_path if binary_path is not None else (self.repo_root / "src" / "dist_m4ri")).resolve()
533 self.cache_dir = (cache_dir if cache_dir is not None else mi.default_cache_dir(self.repo_root)).resolve()
534 self.allow_fallback = allow_fallback
535 self._has_json: Optional[bool] = None
536 self._desc_cache: Optional[Dict[str, Any]] = None
537 self._last_threads: Optional[int] = None
538
539 def probe_json_support(self) -> bool:
540 """Probe `<binary> --help json` once; rc == 0 means opt-in json=/stamp= are supported."""
541 if self._has_json is not None:
542 return self._has_json
543 if not self.binary.is_file():
544 self._has_json = False
545 return False
546 try:
547 p = subprocess.run(
548 [str(self.binary), "--help", "json"],
549 capture_output=True,
550 text=True,
551 timeout=5,
552 )
553 out = (p.stdout or "") + (p.stderr or "")
554 self._has_json = (p.returncode == 0 and "json=" in out)
555 except Exception:
556 self._has_json = False
557 return self._has_json
558
559 def describe(self) -> Dict[str, Any]:
560 if self._desc_cache is not None:
561 return dict(self._desc_cache)
562 ver = None
563 if self.binary.is_file():
564 try:
565 p = subprocess.run([str(self.binary), "--version"], capture_output=True, text=True, timeout=5)
566 txt = (p.stdout + p.stderr).strip()
567 m = re.search(r"([0-9]+\.[0-9]+\.[0-9]+[^\s]*)", txt)
568 ver = m.group(1) if m else txt
569 except Exception:
570 ver = None
571 commit, dirty = git_info(self.repo_root)
572 has_json = self.probe_json_support()
573 self._desc_cache = {
574 "name": self.name,
575 "interface": self.interface,
576 "config": self.config,
577 "version": ver,
578 "commit": commit,
579 "dirty": dirty,
580 "binary": str(self.binary),
581 "sha256": mi.sha256_file(self.binary) if self.binary.is_file() else None,
582 "deps": {"m4ri": _m4ri_version()},
583 "features": {"json": has_json, "stamp": has_json},
584 }
585 return dict(self._desc_cache)
586
587 def supports(self, inst: Dict[str, Any]) -> bool:
588 if not self.binary.is_file():
589 return False
590 if inst["kind"] not in ("css_sector", "classical", "hl", "dem", "circuit"):
591 return False
592 if self.config == "cc_fixed":
593 d_ref = inst.get("d_ref")
594 return d_ref is not None and int(d_ref) >= 2
595 return True
596
597 def prepare(self, inst: Dict[str, Any]) -> Dict[str, Any]:
598 files = mi.resolve_instance_files(
599 inst, repo_root=self.repo_root, cache_dir=self.cache_dir, verify_sha=False
600 )
601 if inst["kind"] != "circuit":
602 return {"prep_s": 0.0, "files": {k: str(v) for k, v in files.items()}, "dem_mode": "canonical"}
603
604 # Native binary cannot read .stim directly; convert to canonical DEM in cache and record prep_s
605 recipe = inst.get("recipe")
606 if not recipe:
607 raise ValueError(f"Circuit instance {inst['id']} lacks a recipe to generate canonical DEM")
608 src_sha = recipe["source_stim"]["sha256"]
609 prep_dir = self.cache_dir / "prep_dem" / src_sha[:2]
610 dem_path = prep_dir / f"{src_sha}_canonical.dem"
611 meta_path = prep_dir / f"{src_sha}_canonical.meta.json"
612 if dem_path.is_file() and meta_path.is_file():
613 meta = json.loads(meta_path.read_text(encoding="utf-8"))
614 if mi.sha256_file(dem_path) == meta.get("sha256"):
615 return {
616 "prep_s": float(meta.get("gen_s", 0.0)),
617 "files": {"dem": str(dem_path)},
618 "dem_mode": "canonical",
619 "dem_path": str(dem_path),
620 }
621 prep_dir.mkdir(parents=True, exist_ok=True)
622 t0 = time.monotonic()
623 data, meta = mi.generate_canonical_dem(recipe, repo_root=self.repo_root)
624 gen_s = round(time.monotonic() - t0, 4)
