42static inline int one_csr_row_combine(one_vec_t * const v1, const one_vec_t * const v0,
43 const csr_t * const mat, const int row){
44#ifndef NDEBUG
45 if ((!v1) || (!v0) || (!mat))
46 ERROR("all arguments must be allocated: v1=%p v0=%p mat=%p\n",v1,v0,mat);
47 if(v1 == v0)
48 ERROR("the two vectors should not be the same !");
49 if((row<0) || (row >= mat->rows))
50 // (v1->max < mat->cols) || (v0->max < mat->cols)
51 ERROR("this should not happen\n");
52#endif
53 int iM, i0=0, i1=0;
54 for (iM = mat->p[row]; iM < mat->p[row+1]; iM++){
55 const int ic = mat->i[iM];
56 while((i0 < v0->wei) && (v0->vec[i0] < ic))
57 v1->vec[i1++] = v0->vec[i0++];
58 if(i0 >= v0->wei)
59 break;
60 if(v0->vec[i0]==ic)
61 i0++;
62 else
63 v1->vec[i1++] = ic;
64 }
65 if(i0 >= v0->wei)
66 for ( ; iM < mat->p[row+1]; iM++){
67 const int ic = mat->i[iM];
68 v1->vec[i1++] = ic;
69 }
70 else
71 while(i0 < v0->wei)
72 v1->vec[i1++] = v0->vec[i0++];
73
74 v1->wei = i1;
75 return i1;
76}

79static inline int one_ordered_ins(one_vec_t * const err, const int j){
80 int pos=err->wei-1;
81 while(j < err->vec[pos]){
82 err->vec[pos+1] = err->vec[pos];
83 pos--;
84 }
85#ifndef NDEBUG
86 if (j == err->vec[pos])
87 ERROR("Unexpected! vec[%d]=%d is already present!",pos,j);
88#endif
89 err->vec[pos+1]=j;
90#ifndef NDEBUG
91 // if(err->wei >= err->max) ERROR("unexpected!"); /** before increment */
92 for(int i=0; i < err->wei; i++)
93 if(err->vec[i] >= err->vec[i+1]){
94 printf("check ordering at i=%d! ",i);
95 one_vec_print(err);
96 ERROR("unexpected");
97 }
98#endif
99 err->wei ++;
100 return pos+1;
101}

107static inline int one_ordered_search(one_vec_t * const err, const int val){
109 int bot=0, top=err->wei , mid=0;
110#ifndef NDEBUG
111 if (!top)
112 return -1;
113#endif
114 while(top - bot > 1){
115 mid = (top+bot) >> 1;
116#ifndef NDEBUG
117 if (mid>=err->wei)
118 ERROR("this should not happen");
119#endif
120 if (err->vec[mid] <= val)
121 bot = mid;
122 else
123 top = mid;
124 // printf("bot=%d mid=%d top=%d err->vec[mid]=%d val=%d\n",bot,mid,top,err->vec[mid],val);
125 }
126 if ( err->vec[bot] == val)
127 return bot;
128 else
129 return -1;
130}

135static inline int one_ordered_find_del(one_vec_t * const err, const int val){
137 int bot=0, top=err->wei, mid=0;
138#ifndef NDEBUG
139 if (!top)
140 return 0;
141#endif
142 while(top - bot > 1){
143 mid = (top+bot) >> 1;
144 if (err->vec[mid] <= val)
145 bot = mid;
146 else
147 top = mid;
148 }
149 if ( err->vec[mid] != val)
150 return 0;
151 // ERROR("unexpected! value val=%d not found",val);
152
153 for(int i=mid; i < err->wei; i++)
154 err->vec[i] = err->vec[i+1];
155 err->wei --;
156 return 1;
157}
