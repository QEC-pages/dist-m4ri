1"""Adapter for the Python `dist_m4ri.py` library (`compute_*_distance`).
2
3Configurations:
4 - config='off' (engine=None): Python wrapper over the `dist_m4ri` binary (cache disabled).
5 - config='auto' (engine='auto'): Two-stage portfolio (`dist_m4ri` -> `qec_cert`).
6 - config='cert' (engine='cert'): Direct delegation to `qec_cert` through `dist_m4ri.py`.
7
8Each run executes in a fresh Python subprocess (`py_worker.py`) so process wall time includes
9imports, Python wrapper overhead, and (for circuits) Stim -> DEM conversion.

34def find_qec_cert_root(repo_root: Path, explicit: Optional[Path] = None) -> Optional[Path]:
35 """Locate the read-only qec-cert repository if present."""
36 candidates = []
37 if explicit is not None:
38 candidates.append(explicit)
39 env_repo = os.environ.get("QEC_CERT_REPO")
40 if env_repo:
41 candidates.append(Path(env_repo))
42 candidates.append(repo_root.parent / "qec-cert")
43 try:
44 spec = importlib.util.find_spec("qec_cert")
45 if spec and spec.origin:
46 candidates.append(Path(spec.origin).resolve().parent.parent)
47 except Exception:
48 pass
49 for cand in candidates:
50 if cand and (cand / "qec_cert").is_dir():
51 return cand.resolve()
52 return None

55def find_qec_cert_lib(qec_root: Optional[Path]) -> Optional[Path]:
56 """Locate libqeccert.so if present."""
57 env_lib = os.environ.get("QEC_CERT_LIB")
58 if env_lib and Path(env_lib).is_file():
59 return Path(env_lib).resolve()
60 if qec_root is not None:
61 for rel in ("qec_cert/libqeccert.so", "build/libqeccert.so"):
62 cand = qec_root / rel
63 if cand.is_file():
64 return cand.resolve()
65 return None

68def _py_version_of(mod_path: Path) -> Optional[str]:
69 if not mod_path.is_file():
70 return None
71 for line in mod_path.read_text(encoding="utf-8").splitlines()[:80]:
72 m = re.match(r'^__version__\s*=\s*["\']([^"\']+)["\']', line.strip())
73 if m:
74 return m.group(1)
75 return None
