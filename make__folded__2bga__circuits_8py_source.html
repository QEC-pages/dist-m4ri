32def bits(x: int) -> list[int]:
33 out = []
34 while x:
35 low = x & -x
36 out.append(low.bit_length() - 1)
37 x ^= low

41def echelon(rows: list[int]) -> dict[int, int]:
42 basis: dict[int, int] = {}
43 for r in rows:
44 while r:
45 p = r.bit_length() - 1
46 q = basis.get(p)
47 if q is None:
48 basis[p] = r
49 break
50 r ^= q
51 return basis

54def logical_basis(h_rows: list[int], g_rows: list[int], n: int) -> list[int]:
55 b_h = echelon(h_rows)
56 piv = sorted(b_h)
57 for idx, p in enumerate(piv):
58 r = b_h[p]
59 for q in piv[idx + 1:]:
60 if (b_h[q] >> p) & 1:
61 b_h[q] ^= r
62 ker = []
63 for f in range(n):
64 if f in b_h:
65 continue
66 x = 1 << f
67 for p, r in b_h.items():
68 if (r >> f) & 1:
69 x |= 1 << p
70 ker.append(x)
71 b_g = echelon(g_rows)
72 out = []
73 for x in ker:
74 r = x
75 while r:
76 p = r.bit_length() - 1
77 q = b_g.get(p)
78 if q is None:
79 b_g[p] = r
80 out.append(x)
81 break
82 r ^= q
83 return out

86def group_tables_c9xc9() -> tuple[list[list[int]], list[int], list[tuple[int, int]]]:
87 idx_of = {pt: k for k, pt in enumerate(PAIRS_81_2)}
88 mul = [[idx_of[((i1 + j1) % 9, (i2 + j2) % 9)] for (j1, j2) in PAIRS_81_2] for (i1, i2) in PAIRS_81_2]
89 inv = [idx_of[((-i1) % 9, (-i2) % 9)] for (i1, i2) in PAIRS_81_2]

93def group_tables_cyclic(ell: int) -> tuple[list[list[int]], list[int], list[tuple[int, int]]]:
94 mul = [[(i + j) % ell for j in range(ell)] for i in range(ell)]
95 inv = [(-i) % ell for i in range(ell)]
96 coords_ij = [(i, 0) for i in range(ell)]

100def build_folded_2bga(
101 mul: list[list[int]],
102 inv: list[int],
103 coords_ij: list[tuple[int, int]],
104 a_1based: list[int],
105 b_1based: list[int],
106 order_terms: list[int],
107 basis: str,
108 rounds: int,
109) -> stim.Circuit:
110 ell = len(mul)
111 a = [x - 1 for x in a_1based]
112 b = [x - 1 for x in b_1based]
113 wa, wb = len(a), len(b)
114 w = wa + wb
115 xq = [[mul[inv[g]][al] for g in a] + [ell + mul[al][inv[h]] for h in b] for al in range(ell)]
116 zq = [[ell + mul[g][be] for g in a] + [mul[be][h] for h in b] for be in range(ell)]
117 hx = [sum(1 << q for q in row) for row in xq]
118 hz = [sum(1 << q for q in row) for row in zq]
119
120 xa = [2 * ell + al for al in range(ell)]
121 za = [3 * ell + be for be in range(ell)]
122 data = list(range(2 * ell))
123
124 coords: list[tuple[int, int, int]] = [(0, 0, 0)] * (4 * ell)
125 for al, (i, j) in enumerate(coords_ij):
126 coords[al] = (2 * i + 1, 2 * j, 0)
127 coords[ell + al] = (2 * i, 2 * j + 1, 0)
128 coords[xa[al]] = (2 * i, 2 * j, 1)
129 coords[za[al]] = (2 * i + 1, 2 * j + 1, 2)
130
131 def layer_x(ti: int) -> list[int]:
132 return [v for al in range(ell) for v in (xa[al], xq[al][ti])]
133
134 def layer_z(ti: int) -> list[int]:
135 return [v for be in range(ell) for v in (zq[be][ti], za[be])]

137 def round_body(backward: bool) -> tuple[stim.Circuit, list[tuple[str, int]]]:
138 body = stim.Circuit()
139 blocks = [("X", order_terms), ("Z", order_terms)]
140 meas_all = [("X", al) for al in range(ell)] + [("Z", be) for be in range(ell)]
141 if backward:
142 for typ, ord_t in blocks[::-1]:
143 body.append("RX" if typ == "X" else "R", xa if typ == "X" else za)
144 body.append("TICK")
145 for ti in ord_t[::-1]:
146 body.append("CX", layer_x(ti) if typ == "X" else layer_z(ti))
147 body.append("TICK")
148 body.append("MX", xa)
149 body.append("M", za)
150 body.append("TICK")
151 return body, meas_all
152 body.append("RX", xa)
153 body.append("R", za)
154 body.append("TICK")
155 meas: list[tuple[str, int]] = []
156 for typ, ord_t in blocks:
157 for ti in ord_t:
158 body.append("CX", layer_x(ti) if typ == "X" else layer_z(ti))
159 body.append("TICK")
160 body.append("MX" if typ == "X" else "M", xa if typ == "X" else za)
161 body.append("TICK")
162 meas += [(typ, al) for al in range(ell)]
163 return body, meas
164
165 bodies = {"F": round_body(False), "B": round_body(True)}
166 m_cnt = len(bodies["F"][1])
167 anc_xy = {("X", al): coords[xa[al]][:2] for al in range(ell)}
168 anc_xy.update({("Z", be): coords[za[be]][:2] for be in range(ell)})

170 def detectors(cur: str, prev: str | None) -> stim.Circuit:
171 d = stim.Circuit()
172 pos = {m: idx for idx, m in enumerate(bodies[cur][1])}
173 ppos = {m: idx for idx, m in enumerate(bodies[prev][1])} if prev else {}
174 for m in bodies[cur][1]:
175 if prev is None and m[0] != basis:
176 continue
177 tg = [stim.target_rec(pos[m] - m_cnt)] + ([stim.target_rec(ppos[m] - 2 * m_cnt)] if prev else [])
178 d.append("DETECTOR", tg, [anc_xy[m][0], anc_xy[m][1], 0])
179 d.append("SHIFT_COORDS", [], [0, 0, 1])
180 return d

182 def segment(cur: str, prev: str | None) -> stim.Circuit:
183 return bodies[cur][0] + detectors(cur, prev)
184
185 plan = ["B" if r % 2 == 1 else "F" for r in range(rounds)]
186 c = stim.Circuit()
187 for q, (x, y, typ) in enumerate(coords):
188 c.append("QUBIT_COORDS", [q], [x, y, typ])
189 c.append("R" if basis == "Z" else "RX", data)
190 c.append("TICK")
191 c += segment("F", None)
192 pairs = (len(plan) - 1) // 2
193 if pairs:
194 c.append(stim.CircuitRepeatBlock(pairs, segment("B", "F") + segment("F", "B")))
195 if len(plan) % 2 == 0:
196 c += segment("B", "F")
197 c.append("M" if basis == "Z" else "MX", data)
198 nd = len(data)
199 last = {m: idx for idx, m in enumerate(bodies[plan[-1]][1])}
200 for idx in range(ell):
201 m = (basis, idx)
202 supp = zq[idx] if basis == "Z" else xq[idx]
203 tg = [stim.target_rec(last[m] - m_cnt - nd)] + [stim.target_rec(q - nd) for q in sorted(supp)]
204 c.append("DETECTOR", tg, [anc_xy[m][0], anc_xy[m][1], 0])
205 logs = logical_basis(hx, hz, 2 * ell) if basis == "Z" else logical_basis(hz, hx, 2 * ell)
206 assert len(logs) == 2 * ell - len(echelon(hx)) - len(echelon(hz))
207 for idx, x in enumerate(logs):
208 c.append("OBSERVABLE_INCLUDE", [stim.target_rec(q - nd) for q in bits(x)], idx)
209 return c
